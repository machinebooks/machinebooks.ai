flowchart TD
    Start([Provider.call initiates stream]) --> Read{more events?}
    Read -- "no — stream closed" --> Close([streamDone = true])
    Read -- "yes" --> Type{Event type}

    Type -- "text_delta" --> CT[onContent text]
    CT --> Acc1[accumulated content += text]
    Acc1 --> Yield1[yield text_delta to AgentLoop]
    Yield1 --> Read

    Type -- "thinking_delta" --> CTh[onThinking text]
    CTh --> Acc2[accumulated reasoning += text]
    Acc2 --> Yield2[yield thinking to AgentLoop]
    Yield2 --> Read

    Type -- "content_block_start tool_use" --> NewT[create toolCallState index]
    NewT --> Yield3[onToolCall index, id, name, ""]
    Yield3 --> Read

    Type -- "input_json_delta" --> Acc3[toolCallState idx.arguments += chunk]
    Acc3 --> Read
    Try -- "yes" --> StartT[StreamingToolExecutor.addTool]
    Try -- "no" --> Read
    StartT --> Read

    Type -- "content_block_stop" --> Done[onToolCallDone idx]
    Done --> Try{Complete arguments,<br/>valid schema and scope?}

    Type -- "message_delta usage" --> U[onUsage prompt, completion]
    U --> Read

    Type -- "error" --> Err[onError msg]
    Err --> Close
