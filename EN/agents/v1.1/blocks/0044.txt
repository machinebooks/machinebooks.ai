type AgentStreamEvent =
  | { type: 'text'; text: string }
  | { type: 'thinking'; text: string }
  | { type: 'tool_call_delta'; delta: { index: number; id?: string; name?: string; arguments?: string } }
  | { type: 'tool_call_complete'; index: number }
  | { type: 'usage'; usage: { prompt_tokens: number; completion_tokens: number; model: string } }
  | { type: 'finish'; reason: 'stop' | 'tool_calls' | 'length' | 'filtered' | 'refused' | 'paused' | 'unknown' | 'error'; rawReason?: string };
