@startuml
!theme plain
skinparam backgroundColor transparent
skinparam classAttributeIconSize 0
skinparam classFontStyle bold

interface ContentBlock <<union>> {
  type: 'text' | 'thinking' | 'tool_use' | 'tool_result'
}

class TextBlock {
  type: 'text'
  text: string
}

class ThinkingBlock {
  type: 'thinking'
  thinking: string
}

class ToolUseBlock {
  type: 'tool_use'
  id: string
  name: string
  input: Record<string, unknown>
}

class ToolResultBlock {
  type: 'tool_result'
  tool_use_id: string
  content: string
  is_error?: boolean
}

class Message {
  role: 'user' | 'assistant' | 'system' | 'tool'
  content: string | ContentBlock[]
}

class AssistantMessage {
  role: 'assistant'
  content: ContentBlock[]
  usage?: { prompt_tokens, completion_tokens }
}

ContentBlock <|.. TextBlock
ContentBlock <|.. ThinkingBlock
ContentBlock <|.. ToolUseBlock
ContentBlock <|.. ToolResultBlock
Message <|-- AssistantMessage
Message "1" *-- "many" ContentBlock : content

note right of ToolUseBlock
  Generated by the model.
  The `id` is assigned by the provider
  and persists so it can be matched with
  the corresponding ToolResultBlock.
end note

note right of ToolResultBlock
  Generated by the code after
  executing the tool. The
  `tool_use_id` must match the
  preceding ToolUseBlock.
end note

note bottom of Message
  Simplified historical projection.
  It does not preserve all native blocks
  or thinking signatures.
  The adapter must preserve them separately
  or reject an incompatible continuation.
end note
@enduml
