package "Cancellation: three origins" {
  [User presses cancel] as U
  [Bash sibling fails] as B
  [Parent (subagent) aborts] as P
}

package "StreamingToolExecutor" {
  [cancel(reason)] as CX
  [siblingAbortController] as SAC
  [parentSignal listener] as PSL
}

package "Per tool in flight" {
  [Own AbortController] as AC
  [interruptBehavior\n(cancel | block)] as IB
}

U --> CX
B --> SAC
P --> PSL
CX --> AC : abort if\ninterruptBehavior=cancel
SAC --> AC : abort if\ninterruptBehavior=cancel
PSL --> AC : abort if\ninterruptBehavior=cancel
IB ..> AC : decides if\npropagated
@enduml
