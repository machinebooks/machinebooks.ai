# converged_agent.py — Agent balancing security, operations, and cost
from illustrative_host import Agent, tool
from dataclasses import dataclass

@dataclass
class PlatformDecision:
    action: str
    security_impact: str    # positive, neutral, negative
    cost_impact_usd: float  # monthly incremental cost
    latency_impact_ms: int  # pipeline latency impact
    recommendation: str

@tool
def evaluate_scan_frequency(
    component: str,
    current_frequency: str,
    threat_level: str
) -> PlatformDecision:
    """Evaluates whether to adjust scan frequency considering
    security, cost, and performance."""
    if not isinstance(component,str) or not component or current_frequency not in {"per_commit","daily","weekly"} or threat_level not in {"normal","elevated"}:
        raise ValueError("INCOMPLETE: unknown scan-frequency inputs")
    raise ValueError("REVIEW: usage, rates, latency and coverage evidence required; no optimal decision calculated")

converged_agent = Agent(
    model="claude-sonnet-4-6",
    tools=[evaluate_scan_frequency],
    system="""You are a convergent platform agent.
    Every decision must consider three dimensions:
    1. Security: never degrade the security posture.
    2. Cost: minimize token and compute spending.
    3. Performance: minimize pipeline latency.
    Golden rule: security has veto. If a cost optimization
    degrades security, it is rejected."""
)
