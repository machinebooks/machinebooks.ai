# .github/workflows/policy-check.yml
name: Policy Check
on:
  pull_request:
    paths:
      - 'k8s/**'
      - 'docker/**'
      - 'ai-config/**'

jobs:
  opa-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Checkout policy repo
        uses: actions/checkout@v4
        with:
          repository: our-org/security-policies
          path: policies
          token: ${{ secrets.POLICY_REPO_TOKEN }}

      - name: Install OPA
        run: |
          curl -L -o opa https://openpolicyagent.org/downloads/v0.68.0/opa_linux_amd64_static
          chmod +x opa && sudo mv opa /usr/local/bin/

      - name: Run policy tests
        run: |
          opa test ./policies/policies ./policies/tests -v

      - name: Evaluate Kubernetes manifests
        run: |
          for manifest in k8s/*.yaml; do
            echo "--- Evaluating: $manifest ---"
            result=$(opa eval \
              --data policies/policies/kubernetes/ \
              --data policies/data/ \
              --input "$manifest" \
              "[data.kubernetes.container_security.deny, data.kubernetes.rbac.warn]" \
              --format json)
            if ! python -c 'import json,sys; value=json.loads(sys.argv[1])["result"][0]["expressions"][0]["value"]; sys.exit(0 if value == [[], []] else 1)' "$result"; then
              echo "VIOLATION in $manifest: $result"
              exit 1
            fi
          done

      - name: Evaluate deployment gates
        run: |
          # Generate gate input from pipeline artifacts
          python scripts/generate_gate_input.py > gate_input.json
          result=$(opa eval \
            --data policies/policies/pipeline/ \
            --input gate_input.json \
            "[data.pipeline.deployment_gates.deny, data.pipeline.deployment_gates.warn]" \
            --format json)
          if ! python -c 'import json,sys; value=json.loads(sys.argv[1])["result"][0]["expressions"][0]["value"]; sys.exit(0 if value == [[], []] else 1)' "$result"; then
            echo "Gate failed: $result"
            exit 1
          fi
