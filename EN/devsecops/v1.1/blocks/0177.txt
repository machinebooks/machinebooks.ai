def verify_iso_mapping(
    mapping: list[dict],
    pipeline_artifacts: dict
) -> list[dict]:
    """Verify each ISO 42001 control against pipeline artifacts."""
    results = []
    for control in mapping:
        evidence_keys = [key.strip() for key in control["pipeline_evidence"].split("+")]
        has_evidence = all(pipeline_artifacts.get(key) is not None for key in evidence_keys)

        results.append({
            "status_authority":"REVIEW","coverage":"partial",
            "iso_control": control["iso_control"],
            "aiact_article": control["aiact_article"],
            "status": "evidence_available" if has_evidence else "evidence_pending",
            "evidence_path": control["pipeline_evidence"],
            "verification_type": control["verification"]
        })

    return results
