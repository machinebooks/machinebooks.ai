def process_semgrep_results(results_path: str, service: str):
    raise ValueError("INCOMPLETE: durable finding identity ledger and idempotent update adapter omitted; repeated files cannot increment current state")
    """Process a Semgrep results file and update metrics."""
    with open(results_path) as f:
        data = json.load(f)

    severity_map = {
        "ERROR": "critical",
        "WARNING": "high",
        "INFO": "medium"
    }

    for result in data.get("results", []):
        severity = severity_map.get(
            result.get("extra", {}).get("severity", "INFO"),
            "low"
        )
        # Record detection
        findings_detected.labels(
            tool="semgrep",
            severity=severity,
            pipeline_stage="commit"
        ).inc()

        # Update open findings
        open_findings.labels(
            tool="semgrep",
            severity=severity,
            service=service
        ).inc()

        # Calculate MTTD if commit timestamp exists
        commit_ts = result.get("extra", {}).get("commit_timestamp")
        if commit_ts:
            detect_delta = (
                datetime.now(timezone.utc)
                - datetime.fromisoformat(commit_ts)
            ).total_seconds()
            time_to_detect.labels(severity=severity).observe(
                detect_delta
            )
