# scripts/triage_sca.py
"""
Intelligent SCA finding triage.
Receives Grype results + SBOM and prioritizes with Claude.
"""
import json
import sys
import anthropic

def load_results(grype_path: str, sbom_path: str) -> tuple:
    """Load Grype results and SBOM."""
    with open(grype_path) as f:
        vulnerabilities = json.load(f)
    with open(sbom_path) as f:
        sbom = json.load(f)
    if not isinstance(vulnerabilities,dict) or not isinstance(vulnerabilities.get("matches"),list) or vulnerabilities.get("errors") or not isinstance(sbom,dict) or not isinstance(sbom.get("dependencies"),list):
        raise ValueError("INCOMPLETE: invalid SCA or dependency schema")
    return vulnerabilities, sbom

def build_context(vuln: dict, sbom: dict) -> str:
    if not isinstance(vuln,dict):
        raise ValueError("INCOMPLETE: invalid match")
    artifact, record = vuln.get("artifact"), vuln.get("vulnerability")
    if not isinstance(artifact,dict) or not isinstance(record,dict) or not isinstance(artifact.get("name"),str) or not artifact["name"] or not isinstance(record.get("id"),str) or not record["id"]:
        raise ValueError("INCOMPLETE: missing artifact or advisory identity")
    scores = record.get("cvss", [])
    locations = artifact.get("locations", [])
    fix = record.get("fix", {})
    if not isinstance(scores,list) or not isinstance(locations,list) or not isinstance(fix,dict):
        raise ValueError("INCOMPLETE: invalid nested advisory data")
    values=[]
    for entry in scores:
        metrics=entry.get("metrics") if isinstance(entry,dict) else None
        value=metrics.get("baseScore") if isinstance(metrics,dict) else None
        if type(value) not in (int,float) or not 0 <= value <= 10:
            raise ValueError("INCOMPLETE: invalid CVSS score")
        values.append(value)
    if any(not isinstance(row,dict) or not isinstance(row.get("path"),str) for row in locations):
        raise ValueError("INCOMPLETE: invalid advisory locations")
    versions=fix.get("versions",[])
    if not isinstance(versions,list) or any(not isinstance(v,str) for v in versions):
        raise ValueError("INCOMPLETE: invalid fixed versions")
    context={"status":"REVIEW","coverage":"partial","artifact":artifact,
             "advisory":record,"cvss_scores":values,"paths":[row["path"] for row in locations],
             "supplied_dependents":find_dependents(artifact["name"],sbom)}
    return json.dumps(context,ensure_ascii=False) + "\nPropose priorities only; missing fields, reachability, license compatibility and maintenance are unknown unless evidenced."


def find_dependents(package_name: str, sbom: dict) -> list:
    components=sbom.get("components") if isinstance(sbom,dict) else None
    dependencies=sbom.get("dependencies") if isinstance(sbom,dict) else None
    if not isinstance(components,list) or not isinstance(dependencies,list):
        raise ValueError("INCOMPLETE: missing component or relationship coverage")
    if any(not isinstance(c,dict) or not isinstance(c.get("bom-ref"),str) or not isinstance(c.get("name"),str) for c in components):
        raise ValueError("INCOMPLETE: invalid component reference")
    refs=[c["bom-ref"] for c in components]
    if len(refs)!=len(set(refs)):
        raise ValueError("INCOMPLETE: duplicate component reference")
    targets={c["bom-ref"] for c in components if c["name"]==package_name}
    if len(targets)>1:
        raise ValueError("INCOMPLETE: ambiguous package identity")
    result=[]
    for dep in dependencies:
        if not isinstance(dep,dict) or not isinstance(dep.get("ref"),str) or not isinstance(dep.get("dependsOn"),list) or any(not isinstance(v,str) for v in dep["dependsOn"]):
            raise ValueError("INCOMPLETE: invalid dependency relationship")
        if targets.intersection(dep["dependsOn"]):
            result.append(dep["ref"])
    return result


def triage_vulnerability(client, vuln: dict, sbom: dict) -> dict:
    """Triage an individual vulnerability with Claude."""
    context = build_context(vuln, sbom)

    message = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=1024,
        system="""You are a security analyst specializing in SCA.
Your job is to prioritize vulnerabilities in third-party dependencies
considering real exploitability, not just theoretical CVSS.
Respond in JSON with fields: priority, action, reasoning, effort.
Be concise and precise. Do not invent data not in the context.""",
        messages=[{"role": "user", "content": context}]
    )

    proposal = json.loads(message.content[0].text)
    if not isinstance(proposal, dict):
        raise ValueError("INCOMPLETE: invalid proposal schema")
    if not all(key in proposal for key in ('priority', 'action', 'reasoning', 'effort')):
        raise ValueError("INCOMPLETE: missing proposal fields")
    proposal["status"] = "REVIEW"
    proposal["coverage"] = "partial"
    proposal["origin_verified"] = False
    if proposal.get("priority") not in {"CRITICAL","HIGH","MEDIUM","LOW","ACCEPT"} or any(not isinstance(proposal.get(key),str) for key in ("action","reasoning","effort")):
        raise ValueError("INCOMPLETE: invalid SCA proposal fields")
    return proposal
def main():
    grype_path = sys.argv[1]
    sbom_path = sys.argv[2]

    client = anthropic.Anthropic()  # Reads ANTHROPIC_API_KEY from environment
    vulnerabilities, sbom = load_results(grype_path, sbom_path)

    results = []
    for match in vulnerabilities.get("matches", []):
        triage = triage_vulnerability(client, match, sbom)
        results.append({
            "cve": match["vulnerability"]["id"],
            "package": match["artifact"]["name"],
            "cvss": match["vulnerability"]["severity"],
            "triage": triage
        })

    # Sort by agent priority, not by CVSS
    priority_order = {"CRITICAL": 0, "HIGH": 1, "MEDIUM": 2, "LOW": 3, "ACCEPT": 4}
    results.sort(key=lambda r: priority_order.get(r["triage"]["priority"], 5))

    print(json.dumps(results, indent=2, ensure_ascii=False))

if __name__ == "__main__":
    main()
