from datetime import datetime


class ComplianceReportGenerator:
    """Generates human-readable reports for auditors."""

    def generate_audit_report(
        self,
        assessments: list[ControlAssessment],
        framework: str,
        report_date: datetime,
    ) -> str:
        """Generates a Markdown report for the auditor."""

        known = {ComplianceStatus.COMPLIANT, ComplianceStatus.NON_COMPLIANT,
                 ComplianceStatus.PARTIAL, ComplianceStatus.NOT_ASSESSED}
        if not isinstance(assessments, list) or any(getattr(a,"status",None) not in known for a in assessments):
            return "INCOMPLETE: invalid assessment state"
        compliant = [
            a for a in assessments
            if a.status == ComplianceStatus.COMPLIANT
        ]
        non_compliant = [
            a for a in assessments
            if a.status == ComplianceStatus.NON_COMPLIANT
        ]
        partial = [
            a for a in assessments
            if a.status == ComplianceStatus.PARTIAL
        ]
        not_assessed = [
            a for a in assessments
            if a.status == ComplianceStatus.NOT_ASSESSED
        ]

        if not assessments:
            return "INCOMPLETE: no controls assessed"
        total = len(assessments)
        pct = (len(compliant) / total * 100) if total > 0 else 0

        report = f"""# Evidence Draft — {framework}

**Date:** {report_date.strftime('%Y-%m-%d %H:%M UTC')}
**Evaluator:** Local technical assessment, sufficiency review pending
**Evidence period:** Check the receipts and references for each control

## Executive summary

| Status | Controls | Percentage |
|--------|-----------|------------|
| Compliant | {len(compliant)} | {pct:.1f}% |
| Non-compliant | {len(non_compliant)} | {len(non_compliant)/total*100:.1f}% |
| Partial | {len(partial)} | {len(partial)/total*100:.1f}% |
| Not assessed | {len(not_assessed)} | {len(not_assessed)/total*100:.1f}% |
| **Total** | **{total}** | **100%** |

## Non-compliant controls (require action)

"""
        for a in non_compliant + partial + not_assessed:
            report += f"""### {a.control.control_id} — {a.control.title}

- **Proposed status:** {a.status}
- **Cause:** {a.justification}
- **Evidence evaluated:** {len(a.evidences)}
- **Next review:** {a.next_review.strftime('%Y-%m-%d') if a.next_review else 'N/A'}

"""

        report += "## Compliant controls (with evidence)\n\n"
        for a in compliant:
            evidence_list = ", ".join(
                f"{e.source.value} ({e.timestamp.strftime('%Y-%m-%d')})"
                for e in a.evidences
            )
            report += (
                f"- **{a.control.control_id}** — {a.control.title}: "
                f"{evidence_list}\n"
            )

        return "REVIEW: local evidence draft; conformity and scope not established.\n" + report
