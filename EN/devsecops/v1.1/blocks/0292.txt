# The same agent evaluates both matrices
def ejecutar_compliance_dual(bucket_path: str):
    """Proposes partial technical states for review; not an ENS or AI Act verdict."""
    client = anthropic.Anthropic()
    evidencias = recopilar_evidencias(bucket_path)

    resultados_ens = {}
    for cid, cdef in ENS_CONTROL_MAP.items():
        resultados_ens[cid] = evaluar_control(
            cid, cdef, evidencias, client
        )

    resultados_aiact = {}
    for cid, cdef in AIACT_CONTROL_MAP.items():
        resultados_aiact[cid] = evaluar_control(
            cid, cdef, evidencias, client
        )

    # Generate unified report
    return {"status":"REVIEW", "coverage":"partial", "ENS":resultados_ens, "AI_Act":resultados_aiact, "reason":"Unified report and sufficiency review adapter omitted"}
