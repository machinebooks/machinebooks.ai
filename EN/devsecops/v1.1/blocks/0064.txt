apiVersion: kyverno.io/v1
kind: ClusterPolicy
metadata:
  name: restrict-image-registries
spec:
  validationFailureAction: Enforce
  rules:
    - name: validate-registries
      match:
        any:
          - resources:
              kinds:
                - Pod
      validate:
        message: "Only images from ghcr.io/org/ are allowed"
        pattern:
          spec:
            containers:
              - image: "ghcr.io/org/*:*"
    - name: deny-latest-tag
      match:
        any:
          - resources:
              kinds:
                - Pod
      validate:
        message: "The 'latest' tag is not allowed"
        deny:
          conditions:
            any:
              - key: "{{ request.object.spec.containers[].image }}"
                operator: AnyIn
                value: "*:latest"
