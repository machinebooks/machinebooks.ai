# adaptive_defense_policy.rego — Adaptive policy for threat level
# Evaluated on each pipeline cycle to adjust thresholds
package adaptive_security

import future.keywords.if

valid_input if {
    is_number(input.cve_feed.critical_last_24h)
    input.cve_feed.critical_last_24h >= 0
    is_boolean(input.cve_feed.affects_our_stack)
    is_number(input.github_audit.failed_auth_last_hour)
    input.github_audit.failed_auth_last_hour >= 0
}

# Threat level based on external indicators
elevated if {
    valid_input
    # Critical CVE published in last 24h for stack technologies
    input.cve_feed.critical_last_24h > 0
    input.cve_feed.affects_our_stack == true
}

elevated if {
    valid_input
    # Anomalous increase in repository access attempts
    input.github_audit.failed_auth_last_hour > 10
}

threat_level := "elevated" if { elevated }
default threat_level := "unknown"
threat_level := "normal" if {
    valid_input
    not elevated
    input.github_audit.failed_auth_last_hour <= 10
}
default max_autonomy_level := 3

# Dynamic autonomy level adjustment based on threat
max_autonomy_level := 2 if {
    threat_level == "elevated"
}

max_autonomy_level := 1 if {
    threat_level == "normal"
}

# Under elevated threat, everything goes up one supervision level
deny[msg] if {
    threat_level == "elevated"
    input.finding.proposed_autonomy == 1
    msg := "Elevated threat: finding reclassified to level 2 (supervised)"
}
deny[msg] if {
    not valid_input
    msg := "INCOMPLETE: missing or invalid threat indicators; human-only autonomy"
}
