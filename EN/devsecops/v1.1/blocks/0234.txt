from datetime import datetime, timezone, timedelta


class ComplianceEvaluator:
    """Evaluates the compliance state of each control."""

    # Evidence expires after 30 days without a new execution
    MAX_EVIDENCE_AGE = timedelta(days=30)

    def __init__(self, mappings: list[dict], evidences: list[Evidence]):
        self.mappings = mappings
        self.evidences = evidences

    def evaluate_all(self) -> list[ControlAssessment]:
        """Evaluates all framework controls."""
        assessments = []
        now = datetime.now(timezone.utc)

        for mapping in self.mappings:
            control = Control(
                framework=mapping["framework"],
                control_id=mapping["control_id"],
                title=mapping["title"],
                category=mapping.get("category", "technical"),
                automation_level=mapping.get("automation_level", "partial"),
            )

            # Find relevant evidence for this control
            required_sources = [
                s["type"] for s in mapping.get("evidence_sources", [])
            ]
            relevant = self._find_relevant_evidences(required_sources)

            # Evaluate state
            status, justification = self._assess_control(
                control, required_sources, relevant, now
            )

            assessments.append(ControlAssessment(
                control=control,
                status=status,
                evidences=relevant,
                assessed_at=now,
                assessed_by="pipeline:compliance-evaluator",
                justification=justification,
                next_review=now + self.MAX_EVIDENCE_AGE,
            ))

        return assessments

    def _assess_control(
        self,
        control: Control,
        required_sources: list[str],
        evidences: list[Evidence],
        now: datetime,
    ) -> tuple[ComplianceStatus, str]:
        """Returns NOT_ASSESSED; presence is not legal sufficiency."""

        for evidence in evidences:
            if (not isinstance(evidence.timestamp,datetime) or evidence.timestamp.utcoffset() != timedelta(0)
                    or evidence.timestamp > now or type(evidence.passed) is not bool):
                return ComplianceStatus.NOT_ASSESSED, "INCOMPLETE: evidence time or schema"
        if not required_sources:
            return (
                ComplianceStatus.NOT_ASSESSED,
                "Control without automated evidence sources",
            )

        # Verify evidence exists for each required source
        found_sources = {e.source.value for e in evidences}
        missing = set(required_sources) - found_sources
        if missing:
            return (
                ComplianceStatus.NOT_ASSESSED,
                f"INCOMPLETE: missing evidence: {', '.join(missing)}",
            )

        # Verify evidence is not expired
        stale = [
            e for e in evidences
            if (now - e.timestamp) > self.MAX_EVIDENCE_AGE
        ]
        if stale:
            sources_stale = [e.source.value for e in stale]
            return (
                ComplianceStatus.NOT_ASSESSED,
                f"Expired evidence (>30 days): "
                f"{', '.join(sources_stale)}",
            )

        # Verify all evidence passed
        failed = [e for e in evidences if not e.passed]
        if failed:
            sources_failed = [e.source.value for e in failed]
            return (
                ComplianceStatus.NOT_ASSESSED,
                f"REVIEW: failed technical checks: {', '.join(sources_failed)}",
            )

        return (
            ComplianceStatus.NOT_ASSESSED,
            "REVIEW: presence and freshness do not establish origin or sufficiency",
        )

    def _find_relevant_evidences(
        self, required_sources: list[str]
    ) -> list[Evidence]:
        """Returns all supplied matching-source evidence; no latest selection."""
        return [e for e in self.evidences if e.source.value in required_sources]
