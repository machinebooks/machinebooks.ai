# .gitleaks.toml — Gitleaks configuration for the DevSecOps Pipeline
title = "DevSecOps Pipeline secret scanning rules"

# Custom rule: Anthropic (Claude API) tokens
[extend]
useDefault = true

[[rules]]
id = "anthropic-api-key"
description = "Anthropic API key (Claude)"
regex = '''sk-ant-[a-zA-Z0-9_-]{80,}'''
tags = ["api-key", "anthropic"]
keywords = ["sk-ant-"]

# Custom rule: internal Pipeline tokens
[[rules]]
id = "internal-service-token"
description = "Internal Pipeline service token"
regex = '''PIPELINE_TOKEN_[A-Z0-9]{32,}'''
tags = ["token", "internal"]
keywords = ["PIPELINE_TOKEN_"]

# Custom rule: database connection strings
[[rules]]
id = "database-connection-string"
description = "Connection string with embedded credentials"
regex = '''(?i)(mysql|postgres|mongodb|redis):\/\/[^:]+:[^@]+@[^\/]+'''
tags = ["database", "connection-string"]

# Allow example files and documentation
[allowlist]
paths = [
    '''\.gitleaks\.toml$''',
    '''tests/fixtures/.*\.example$''',
]

# Allow known safe values
regexes = [
    '''EXAMPLE_KEY_[A-Z0-9]+''',
    '''<set-in-vault>''',
    '''sk-ant-xxxx.*''',
]
