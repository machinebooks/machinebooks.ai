from dataclasses import dataclass, field
from typing import Callable

@dataclass
class AgentGuardrails:
    """Guardrails for autonomous agents."""
    max_actions_per_run: int = 10
    max_tokens_per_run: int = 50_000
    allowed_tools: list[str] = field(default_factory=list)
    requires_approval: list[str] = field(default_factory=list)
    db_permissions: list[str] = field(default_factory=list)

    # Execution counters
    _action_count: int = field(default=0, init=False)
    _token_count: int = field(default=0, init=False)

    def check_action(self, tool_name: str) -> bool:
        """Verifies whether the agent can execute the action."""
        # Check action budget
        if self._action_count >= self.max_actions_per_run or self._token_count >= self.max_tokens_per_run:
            raise BudgetExhaustedError(
                f"Agent reached limit of {self.max_actions_per_run} "
                f"actions. Requires manual restart."
            )

        # Check allowed tool
        if tool_name not in self.allowed_tools:
            raise UnauthorizedToolError(
                f"Tool '{tool_name}' is not in the list of "
                f"allowed tools: {self.allowed_tools}"
            )

        # Check if human approval required
        if tool_name in self.requires_approval:
            raise ApprovalRequiredError(
                f"Tool '{tool_name}' requires human approval."
            )

        # Database grants are declarations; execution requires the host's protected adapter.
        if self.db_permissions:
            raise UnauthorizedToolError("REVIEW: database authorization adapter omitted")
        self._action_count += 1
        return True

class BudgetExhaustedError(Exception):
    pass

class UnauthorizedToolError(Exception):
    pass

class ApprovalRequiredError(Exception):
    pass
