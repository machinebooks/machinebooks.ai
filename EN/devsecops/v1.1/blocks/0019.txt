# .github/workflows/surface-discovery.yml
name: Attack Surface Discovery
on:
  push:
    branches: [main]
  schedule:
    - cron: "0 6 * * 1"   # Monday at 06:00 UTC as backup

jobs:
  discover:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install dependencies
        run: pip install anthropic pyyaml claude-agent-sdk

      - name: Run asset inventory
        run: python pipeline_inventory.py > inventory.json
        env:
          PYTHONPATH: .

      - name: Run STRIDE analysis
        run: python stride_classifier.py
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}

      - name: Verify surface changes
        run: python surface_discovery_agent.py
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}

      - name: Upload security artifacts
        uses: actions/upload-artifact@v4
        with:
          name: attack-surface-report
          path: |
            inventory.json
            threat_model.json
