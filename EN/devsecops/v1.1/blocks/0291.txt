# compliance_metrics.py — Prometheus metrics exporter
from prometheus_client import Gauge, start_http_server
import boto3
from datetime import datetime, timedelta, timezone

# Compliance metrics
compliance_controles_conformes = Gauge(
    'ens_draft_controles_conformes',
    'Proposed local-state count; legal conformity pending'
)
compliance_controles_parciales = Gauge(
    'ens_draft_controles_parciales',
    'Proposed partial-state count; review pending'
)
compliance_controles_no_conformes = Gauge(
    'ens_draft_controles_no_conformes',
    'Proposed technical-gap count; no legal verdict'
)
compliance_evidencia_antiguedad_dias = Gauge(
    'ens_draft_evidencia_antiguedad_max_dias',
    'Age in days of the oldest evidence',
    ['control_id']
)
compliance_cobertura_pipeline = Gauge(
    'ens_draft_cobertura_pipeline_pct',
    'Proposed proportion under declared coverage; integration pending'
)

def actualizar_metricas():
    raise ValueError("REVIEW: protected evidence origin and legally reviewed control results are required; this omitted adapter cannot publish conformity metrics")

    """Queries the evidence store and updates metrics."""
    # Read latest compliance report
    resultados = cargar_ultimo_informe()

    if not isinstance(resultados, dict) or not resultados or any(not isinstance(r, dict) or r.get("estado") not in {"conforme", "parcial", "no_conforme"} for r in resultados.values()):
        raise ValueError("INCOMPLETE: no validated control results")
    conformes = sum(
        1 for r in resultados.values()
        if r["estado"] == "conforme"
    )
    parciales = sum(
        1 for r in resultados.values()
        if r["estado"] == "parcial"
    )
    no_conformes = sum(
        1 for r in resultados.values()
        if r["estado"] == "no_conforme"
    )

    compliance_controles_conformes.set(conformes)
    compliance_controles_parciales.set(parciales)
    compliance_controles_no_conformes.set(no_conformes)

    # Calculate evidence age per control
    for control_id, datos in resultados.items():
        if "fecha_ultima_evidencia" in datos:
            dias = (datetime.now(timezone.utc) - datetime.fromisoformat(
                datos["fecha_ultima_evidencia"]
            )).days
            compliance_evidencia_antiguedad_dias.labels(
                control_id=control_id
            ).set(dias)
