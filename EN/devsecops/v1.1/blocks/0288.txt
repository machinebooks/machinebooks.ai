def evaluar_control(control_id: str, control_def: dict,
                    evidencias: dict, client: anthropic.Anthropic):
    """Evaluates an ENS control against available evidence."""
    if not isinstance(evidencias,dict) or not evidencias:
        raise ValueError("INCOMPLETE: no evidence")
    for metadata in evidencias.values():
        if not isinstance(metadata,dict):
            raise ValueError("INCOMPLETE: invalid evidence metadata")
        stamp = datetime.fromisoformat(metadata["fecha"].replace("Z","+00:00"))
        if stamp.utcoffset() != timedelta(0) or stamp > datetime.now(timezone.utc) or type(metadata.get("count")) is not int or metadata["count"] <= 0:
            raise ValueError("INCOMPLETE: invalid evidence time or count")
    artefactos_encontrados = []
    artefactos_faltantes = []

    for artefacto in control_def["artefactos_requeridos"]:
        if artefacto in evidencias:
            artefactos_encontrados.append({
                "nombre": artefacto,
                "fecha_mas_reciente": evidencias[artefacto]["fecha"],
                "total_ejecuciones": evidencias[artefacto]["count"],
                "resumen": evidencias[artefacto]["resumen"]
            })
        else:
            artefactos_faltantes.append(artefacto)

    if artefactos_faltantes or not artefactos_encontrados:
        return {"status":"INCOMPLETE","coverage":"partial","estado": "parcial", "justificacion": "INCOMPLETE: missing evidence",
                "evidencias_referenciadas": [], "acciones_pendientes": artefactos_faltantes}

    # Claude analyzes artifacts and determines compliance
    prompt = f"""Evaluate compliance of ENS control {control_id}
({control_def['nombre']}).

Compliance criteria: {control_def['criterio_cumplimiento']}

Artifacts found:
{json.dumps(artefactos_encontrados, indent=2, ensure_ascii=False)}

Missing artifacts: {artefactos_faltantes}

Propose a draft for REVIEW; retain missing, partial, and pending actions. Do not issue a conformity opinion. Respond in JSON with: estado (conforme/no_conforme/parcial),
justificacion (2-3 sentences), evidencias_referenciadas (list),
acciones_pendientes (list, empty if compliant)."""

    message = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}]
    )
    try:
        proposal = json.loads(message.content[0].text)
        if not isinstance(proposal, dict) or proposal.get("estado") not in {"conforme", "no_conforme", "parcial"}:
            raise ValueError("invalid state")
        if not isinstance(proposal.get("justificacion"), str) or any(not isinstance(proposal.get(k), list) for k in ("evidencias_referenciadas", "acciones_pendientes")):
            raise ValueError("invalid proposal")
        proposal["proposed_estado"] = proposal["estado"]
        proposal["estado"] = "parcial"
        proposal.update(status="REVIEW",coverage="partial",origin_verified=False)
        proposal["acciones_pendientes"].append("REVIEW: authorized sufficiency review required")
        return proposal
    except (ValueError, TypeError, KeyError, IndexError, AttributeError):
        return {"status":"INCOMPLETE","coverage":"partial","estado": "parcial", "justificacion": "INCOMPLETE: invalid model output",
                "evidencias_referenciadas": [], "acciones_pendientes": ["human review"]}
