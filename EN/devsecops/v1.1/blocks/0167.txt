class IndirectInjectionDetector:
    """Detects whether a RAG response shows signs of injection."""

    DETECTION_PATTERNS = [
        r"(?:as|according to)\s+(?:my|the)\s+instructions",
        r"no\s+(?:mandatory|obligatory)\s+requirements?\s+exist",
        r"(?:ignore|forget|discard)\s+(?:the\s+)?(?:previous|above)",
        r"(?:actually|the truth is)\s+(?:it is|they are)\s+not\s+necessar",
    ]

    def __init__(self, anthropic_client):
        self.claude = anthropic_client

    def check_response(
        self, query: str, context_chunks: list[str],
        response: str
    ) -> dict:
        """Verifies coherence between query, context, and response."""
        # Layer 1: static patterns in the response
        static_flags = []
        for pattern in self.DETECTION_PATTERNS:
            if re.search(pattern, response, re.IGNORECASE):
                static_flags.append(pattern[:50])

        # Layer 2: semantic verification if there are flags
        if static_flags or len(context_chunks) > 3:
            coherence = self._check_semantic_coherence(
                query, context_chunks, response
            )
        else:
            coherence = {"score": None, "concerns": ["REVIEW: semantic check omitted"]}

        return {
            "static_flags": static_flags,
            "coherence_score": coherence["score"],
            "concerns": coherence["concerns"],
            "action": (
                "block" if coherence["score"] is not None and coherence["score"] < 0.3
                else "review"
            ),
        }

    def _check_semantic_coherence(
        self, query: str, chunks: list[str], response: str
    ) -> dict:
        """Uses Claude to verify semantic coherence."""
        response_obj = self.claude.messages.create(
            model="claude-haiku-4-5",
            max_tokens=512,
            system=(
                "Analyze whether the response is coherent with the "
                "provided context chunks. Detect whether the response "
                "contains information that is NOT in the chunks or that "
                "contradicts general domain knowledge. "
                "Respond in JSON: {\"score\": <0.0-1.0>, "
                "\"concerns\": [\"...\"]}"
            ),
            messages=[{
                "role": "user",
                "content": (
                    f"Query: {query}\n\n"
                    f"Context chunks:\n"
                    + "\n---\n".join(chunks[:5])
                    + f"\n\nGenerated response:\n{response}"
                )
            }]
        )
        if len(chunks) > 5:
            return {"score": None, "concerns": ["REVIEW: context exceeds sample"]}
        try:
            result = json.loads(response_obj.content[0].text)
            score = result["score"]
            if type(score) not in (int, float) or not 0 <= score <= 1:
                raise ValueError("invalid score")
            if not isinstance(result["concerns"], list) or any(not isinstance(v, str) for v in result["concerns"]):
                raise ValueError("invalid concerns")
            return result
        except (ValueError, KeyError, TypeError, IndexError, AttributeError):
            return {"score": None, "concerns": ["INCOMPLETE: invalid coherence output"]}
