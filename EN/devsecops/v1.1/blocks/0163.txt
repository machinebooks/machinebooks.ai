import hashlib
import re
from dataclasses import dataclass, field
from enum import Enum
from pathlib import Path
from datetime import datetime, timezone

class ClassificationLevel(Enum):
    PUBLIC = "public"
    INTERNAL = "internal"
    CONFIDENTIAL = "confidential"
    RESTRICTED = "restricted"

@dataclass
class DocumentMetadata:
    source_path: str
    uploaded_by: str
    classification: ClassificationLevel
    authorized_groups: list[str]
    sha256: str = ""
    ingested_at: str = ""
    validation_status: str = "pending"
    validation_notes: list[str] = field(default_factory=list)

@dataclass
class ValidationResult:
    passed: bool
    stage: str  # "static" or "semantic"
    findings: list[str]
    risk_score: float  # 0.0 (clean) to 1.0 (poisoned)

class SecureIngestionPipeline:
    """Ingestion pipeline with two-layer validation."""

    # Hidden text and injection payload patterns
    HIDDEN_TEXT_PATTERNS = [
        r'color:\s*#fff\w*.*?font-size:\s*[01]px',  # CSS invisible text
        r'\\textcolor\{white\}',                      # LaTeX white text
        r'<span[^>]*display:\s*none[^>]*>',           # Hidden HTML
        r'ignore\s+(all\s+)?previous\s+instructions',  # Direct injection
        r'system\s*prompt\s*override',                 # System override
        r'you\s+are\s+now\s+a',                       # Classic jailbreak
    ]

    ALLOWED_EXTENSIONS = {".pdf", ".docx", ".md", ".txt", ".html"}
    MAX_FILE_SIZE_MB = 50

    def validate_static(
        self, text: str, metadata: DocumentMetadata
    ) -> ValidationResult:
        """First layer: deterministic validation without LLM."""
        findings = []
        risk = 0.0
        ext = Path(metadata.source_path).suffix.lower()

        # Verify accepted format
        if ext not in self.ALLOWED_EXTENSIONS:
            findings.append(f"Unsupported extension: {ext}")
            return ValidationResult(False, "static", findings, 1.0)

        # Verify size (assuming already-extracted text)
        if len(text.encode("utf-8")) > self.MAX_FILE_SIZE_MB * 1024 * 1024:
            findings.append("Document exceeds maximum size")
            return ValidationResult(False, "static", findings, 1.0)

        # Look for hidden text and injection patterns
        for pattern in self.HIDDEN_TEXT_PATTERNS:
            matches = re.findall(pattern, text, re.IGNORECASE)
            if matches:
                findings.append(
                    f"Suspicious pattern detected: {pattern[:40]}... "
                    f"({len(matches)} occurrences)"
                )
                risk = max(risk, 0.8)

        # Detect anomalous ratio of non-printable characters
        non_printable = sum(
            1 for c in text
            if not c.isprintable() and c not in '\n\r\t'
        )
        ratio = non_printable / max(len(text), 1)
        if ratio > 0.05:
            findings.append(
                f"Non-printable character ratio: {ratio:.2%}"
            )
            risk = max(risk, 0.6)

        # Calculate hash for deduplication and traceability
        metadata.sha256 = hashlib.sha256(
            text.encode("utf-8")
        ).hexdigest()

        passed = risk < 0.5
        return ValidationResult(passed, "static", findings, risk)
