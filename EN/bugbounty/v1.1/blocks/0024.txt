#!/usr/bin/env python3
"""Query DLL Authenticode states without conflating errors and unsigned files."""
import json
import os
import subprocess
from pathlib import Path

def check_signature(dll_path: Path) -> dict:
    command = (
        "$ErrorActionPreference='Stop'; "
        "$s=Get-AuthenticodeSignature -LiteralPath $env:BB_SIGNATURE_PATH; "
        "[pscustomobject]@{status=$s.Status.ToString(); "
        "signer=if($s.SignerCertificate){$s.SignerCertificate.Subject}"
        "else{''}} | ConvertTo-Json -Compress"
    )
    environment = os.environ.copy()
    environment["BB_SIGNATURE_PATH"] = str(dll_path.resolve())
    try:
        result = subprocess.run(
            ["powershell.exe", "-NoProfile", "-NonInteractive",
             "-Command", command],
            env=environment, capture_output=True, text=True, timeout=10,
        )
        if result.returncode != 0:
            return {"path": str(dll_path), "error": "Signature query failed",
                    "returncode": result.returncode}
        response = json.loads(result.stdout)
        if not isinstance(response, dict) or not isinstance(response.get("status"), str):
            raise ValueError("Invalid signature response")
        return {"path": str(dll_path), "name": dll_path.name,
                "status": response["status"], "signer": response.get("signer", ""),
                "size": dll_path.stat().st_size}
    except (OSError, subprocess.TimeoutExpired, ValueError) as error:
        return {"path": str(dll_path), "error": str(error)}

def audit_dll_signatures(app_dir: str) -> dict:
    dlls = list(Path(app_dir).rglob("*.dll"))
    results = {"total": len(dlls), "valid": [], "unsigned": [],
               "other_statuses": [], "errors": []}
    for dll in dlls:
        signature = check_signature(dll)
        if "error" in signature:
            category = "errors"
        elif signature["status"] == "Valid":
            category = "valid"
        elif signature["status"] == "NotSigned":
            category = "unsigned"
        else:
            category = "other_statuses"
        results[category].append(signature)
    results["unsigned_count"] = len(results["unsigned"])
    results["unsigned_ratio_of_total_paths"] = (
        len(results["unsigned"]) / len(dlls) * 100 if dlls else 0
    )
    return results
