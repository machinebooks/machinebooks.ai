// Pseudocode for IOCTL 0x222400 handler (reconstructed)
NTSTATUS HandleMapPhysicalMemory(PIRP Irp) {
    PIO_STACK_LOCATION stack = IoGetCurrentIrpStackLocation(Irp);
    PASIO3_MAP_REQUEST req = Irp->AssociatedIrp.SystemBuffer;

    // No checks (see Chapter 8 for the import analysis)

    PVOID mapped = MmMapIoSpace(
        req->PhysicalAddress,
        req->NumberOfBytes,
        req->CacheType
    );

    // Return virtual address to caller
    *(PVOID*)Irp->AssociatedIrp.SystemBuffer = mapped;
    Irp->IoStatus.Information = sizeof(PVOID);
    Irp->IoStatus.Status = STATUS_SUCCESS;
    IoCompleteRequest(Irp, IO_NO_INCREMENT);
    return STATUS_SUCCESS;
}
