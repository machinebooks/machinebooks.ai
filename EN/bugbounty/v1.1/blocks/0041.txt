#!/usr/bin/env python3
"""
Import table analysis for DLL sideloading.
Classifies each imported DLL by hijacking risk.
"""
import pefile
import os
from pathlib import Path

def analyze_imports_for_hijacking(exe_path: str) -> dict:
    """Analyzes a PE and classifies its imports by hijacking risk."""
    pe = pefile.PE(exe_path)
    exe_dir = Path(exe_path).parent

    results = {
        "executable": Path(exe_path).name,
        "total_imports": 0,
        "hijackable": [],
        "phantom": [],
        "protected": [],
    }

    if not hasattr(pe, 'DIRECTORY_ENTRY_IMPORT'):
        return results

    for entry in pe.DIRECTORY_ENTRY_IMPORT:
        dll_name = entry.dll.decode()
        results["total_imports"] += 1

        # Count exports we'd need to redirect
        export_count = len(entry.imports)

        # Does it exist in the application directory?
        local_exists = (exe_dir / dll_name).exists()

        # Does it exist in System32?
        sys32 = Path(os.environ["SYSTEMROOT"]) / "System32" / dll_name
        sys32_exists = sys32.exists()

        # Classify
        if not sys32_exists and not local_exists:
            # Phantom DLL: doesn't exist anywhere
            results["phantom"].append({
                "dll": dll_name,
                "exports_needed": export_count,
                "risk": "CRITICAL — phantom DLL, trivial to impersonate",
            })
        elif not local_exists and sys32_exists:
            # Sideloading candidate: exists in System32
            # but not in the local directory
            results["hijackable"].append({
                "dll": dll_name,
                "exports_needed": export_count,
                "system_path": str(sys32),
                "risk": "HIGH — proxy DLL candidate",
            })
        else:
            results["protected"].append({
                "dll": dll_name,
                "local_path": str(exe_dir / dll_name) if local_exists else None,
            })

    return results
