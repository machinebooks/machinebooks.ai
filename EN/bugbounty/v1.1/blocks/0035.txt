"""
Hash patching PoC for Wand IDE.
Modifies the ASAR and patches the integrity hash in the executable.

WARNING: For authorized security research only.
"""
import hashlib
from pathlib import Path

def patch_hash(exe_path: str, old_hash: str, new_hash: str):
    """Replaces the integrity hash in the executable."""
    data = bytearray(Path(exe_path).read_bytes())

    old_bytes = old_hash.encode('ascii')
    new_bytes = new_hash.encode('ascii')

    offset = data.find(old_bytes)
    if offset == -1:
        raise ValueError("Original hash not found in binary")

    # Patch
    data[offset:offset + 64] = new_bytes

    # Check for additional occurrences
    second = data.find(old_bytes, offset + 64)
    if second != -1:
        print(f"WARNING: second occurrence at offset {hex(second)}")

    Path(exe_path).write_bytes(bytes(data))
    return offset
