"""
adobe_cdp_enum.py — CDP target enumeration in Adobe Acrobat DC
Discovers the RNA panels exposed by WebView2 with CDP active.
"""
import requests
import json

CDP_HOST = "http://localhost:9222"

def enumerate_targets():
    """List all available CDP targets."""
    resp = requests.get(f"{CDP_HOST}/json")
    targets = resp.json()
    
    print(f"[*] CDP targets found: {len(targets)}\n")
    
    for i, target in enumerate(targets):
        print(f"--- Target {i+1} ---")
        print(f"  Title:       {target.get('title', 'N/A')}")
        print(f"  URL:         {target.get('url', 'N/A')}")
        print(f"  Type:        {target.get('type', 'N/A')}")
        print(f"  WebSocket:   {target.get('webSocketDebuggerUrl', 'N/A')}")
        print()
    
    return targets

if __name__ == "__main__":
    targets = enumerate_targets()
