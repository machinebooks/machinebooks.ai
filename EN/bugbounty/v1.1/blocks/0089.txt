def correlate_crash(checkpoint_file, minidump_dir):
    """Correlates the last BSOD with the IOCTL that caused it."""
    import json, os, glob

    with open(checkpoint_file) as f:
        state = json.load(f)

    # Find the most recent minidump
    dumps = sorted(glob.glob(os.path.join(minidump_dir, "*.dmp")),
                   key=os.path.getmtime, reverse=True)
    if dumps:
        latest = dumps[0]
        print(f"[*] Crash caused by IOCTL {state['current']}")
        print(f"    Buffer size: {state['buf_size']}")
        print(f"    Minidump: {latest}")
        # Use WinDBG or dumpchk.exe to analyze
        return state['current'], latest
    return None, None
