export const fsReadTool = {
  schema: {
    type: "function",
    function: {
      name: "fs_read",
      description: "Reads a file from the active workspace",
      parameters: {
        type: "object",
        properties: {
          path: { type: "string", description: "Path relative to the workspace" },
        },
        required: ["path"],
      },
    },
  },
  handler: async ({ path }, ctx) => {
    if (!ctx.permissions.canRead(path)) {
      throw new Error(`Read denied: ${path}`);
    }
    return await fs.readFile(resolveInWorkspace(path), "utf8");
  },
};
