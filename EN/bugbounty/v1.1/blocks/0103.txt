anim = win32com.client.Dispatch(
    "{756E6C18-79CC-3842-9E47-7C80011D303A}"
)
anim.init(10, 10)

layer1 = anim.CreateLayer()
layer2 = anim.CreateLayer()
anim.AddLayer(layer1)
anim.AddLayer(layer2)

# Clear removes both layers internally
anim.ClearLayers()

# But layer1 and layer2 are still valid references
# Any operation on them is a stale reference
anim.RemoveLayer(layer1)  # Stale reference — memory already freed

# Worse: reuse layer1 after creating new layers
layer3 = anim.CreateLayer()
anim.AddLayer(layer3)
# layer1 may now point to the same memory block as layer3
# Operations on layer1 corrupt layer3
