import re
import json
from pathlib import Path

def extract_system_prompts(extension_js_path):
    """Extracts system prompts from the Copilot Chat extension."""
    content = Path(extension_js_path).read_text(encoding='utf-8')

    # Patterns that contain system prompts
    patterns = [
        r'system["\']:\s*["\'](.+?)["\']',    # system: "prompt..."
        r'systemMessage["\']:\s*["\'](.+?)["\']',
        r'instructions["\']:\s*["\'](.+?)["\']',
        r'You are .+?(?=["\']\s*[,}\]])',       # "You are..." prompts
    ]

    prompts = set()
    for pattern in patterns:
        matches = re.findall(pattern, content, re.DOTALL)
        for match in matches:
            if len(match) > 50:  # Filter short strings
                prompts.add(match[:500])  # Truncate for readability

    return sorted(prompts)
