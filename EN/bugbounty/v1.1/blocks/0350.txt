"""
tg_string_search.py -- Search for debug-related strings
Extracts strings from the Telegram binary and filters by relevant patterns.
"""
import subprocess
import re

def search_debug_strings(binary_path):
    """Search for debug strings in the Telegram binary."""
    # Extract strings from the binary (minimum 8 characters)
    result = subprocess.run(
        ["strings", "-n", "8", binary_path],
        capture_output=True, text=True
    )
    
    patterns = [
        r"debug",
        r"webview.*debug",
        r"devtools",
        r"remote.debugging",
        r"experimental",
        r"options\.json",
    ]
    
    findings = []
    for line in result.stdout.splitlines():
        for pattern in patterns:
            if re.search(pattern, line, re.IGNORECASE):
                findings.append(line.strip())
                break
    
    # Remove duplicates and sort
    unique = sorted(set(findings))
    for f in unique:
        print(f"  [+] {f}")
    
    return unique

# Relevant findings:
# [+] -debug
# [+] Enable webview inspecting
# [+] experimental_options.json
# [+] webview-debug-enabled
