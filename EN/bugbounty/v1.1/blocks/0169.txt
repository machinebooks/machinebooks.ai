import ldclient
from ldclient.config import Config

# SDK key extracted from the Wand bundle
config = Config(sdk_key="sdk-XXXXX-XXXXXXXX")
ldclient.set_config(config)
client = ldclient.get()

# Evaluate all known flags
# Flag names are extracted from the JavaScript bundle
flag_names = [
    "enable-content-filter",
    "require-auth-for-api",
    "beta-agent-mode",
    "enable-telemetry",
    "max-context-window",
    "security-classifier-v2",
]

user = {"key": "anonymous", "anonymous": True}
for flag in flag_names:
    value = client.variation(flag, user, default=None)
    print(f"  {flag}: {value}")
