import winreg

def get_known_dlls():
    """Reads the Known DLLs list from the system registry."""
    known = set()
    try:
        key = winreg.OpenKey(
            winreg.HKEY_LOCAL_MACHINE,
            r"SYSTEM\CurrentControlSet\Control\Session Manager\KnownDLLs"
        )
        i = 0
        while True:
            try:
                name, value, _ = winreg.EnumValue(key, i)
                # Values are DLL names (e.g., "kernel32.dll")
                known.add(value.lower())
                i += 1
            except WindowsError:
                break
        winreg.CloseKey(key)
    except WindowsError:
        # Fallback to hardcoded list if it fails
        known = {"kernel32.dll", "user32.dll", "ntdll.dll"}
    return known
