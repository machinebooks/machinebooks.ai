# Destination approval, citation verification and token coverage require omitted adapters.

def answer_query(query, identity, corpus_adapter, embedding_adapter, filter_adapter, model_adapter):
    results = search_knowledge(query, identity, corpus_adapter, embedding_adapter, filter_adapter, top_k=5)
    if not results:
        return {"answer": "No authorized sources were retrieved; this does not prove that relevant evidence is absent.", "sources": []}
    model_adapter.require_destination_approval(identity, results)
    response = model_adapter.answer_with_sources(query, results)
    if response.get("stop_reason") != "end_turn" or type(response.get("text")) is not str or not response["text"].strip():
        raise ValueError("Incomplete answer")
    if model_adapter.validate_citations(response["text"], results) is not True:
        raise ValueError("Unverified source citations")
    sources = [{**r["metadata"], **{k: r[k] for k in ("doc_id", "doc_hash", "version", "source_uri", "score", "section")}} for r in results]
    return {"answer": response["text"], "sources": sources}
