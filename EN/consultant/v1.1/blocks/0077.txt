@tool
def analyze_job_postings(
    companies: list[str],
    period_days: int = 30
) -> dict:
    """
    Analyzes job postings published by competitor companies.
    Extracts strategy signals: sought profiles, technologies,
    locations, seniority level.
    Only processes information published on public portals.
    """
    signals = {}
    for company in companies:
        postings = _search_public_postings(company, period_days)
        signals[company] = {
            "total_postings": len(postings),
            "predominant_profiles": _classify_profiles(postings),
            "mentioned_technologies": _extract_technologies(postings),
            "expansion_areas": _infer_areas(postings),
            "investment_level": _estimate_investment_level(postings)
        }
    return signals

@tool
def detect_web_changes(
    urls: list[str],
    last_review: str
) -> list[dict]:
    """
    Detects changes on competitors' service pages.
    Compares current content with stored version.
    Only accesses public pages without authentication.
    """
    changes = []
    for url in urls:
        current_content = _get_public_content(url)
        previous_content = _retrieve_cache(url, last_review)
        if current_content != previous_content:
            diff = _generate_semantic_diff(
                previous_content, current_content
            )
            changes.append({
                "url": url,
                "change_type": diff["type"],  # new_service, pricing, team
                "summary": diff["summary"],
                "relevance": diff["relevance"]  # high, medium, low
            })
    return changes
