from anthropic import Anthropic
import json

client = Anthropic()

def generate_initiative_timeline(
    initiative: dict,
    current_maturity: dict,
    regulatory_constraints: list
) -> dict:
    """Generates a realistic timeline for an AI initiative
    in the Spanish public sector, including non-technical dependencies."""

    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=4096,
        system="""You are an AI project planner in the Spanish public
sector. Generate timelines that include:
- Public procurement timelines (minor contract: 1-3 months,
  negotiated without publication: 4-6 months, open: 6-12 months)
- GDPR impact assessment timelines (2-4 months, plus AEPD
  consultation if applicable: 3-6 additional months)
- ENS compliance timelines (medium category: 2-3 months,
  high: 4-8 months)
- AI Act risk classification and documentation (1-3 months)
- Union negotiation if affecting positions (2-6 months)
- Staff training (1-3 months)

IMPORTANT: Technical timelines are usually 30% of the total
timeline. The remaining 70% is regulation, procurement, and people.
Be realistic, not optimistic.""",
        messages=[{
            "role": "user",
            "content": f"""Generate the timeline for this initiative:

INITIATIVE: {json.dumps(initiative, ensure_ascii=False)}

CURRENT MATURITY: {json.dumps(current_maturity, ensure_ascii=False)}

REGULATORY CONSTRAINTS: {json.dumps(regulatory_constraints, ensure_ascii=False)}

Include: phases, estimated duration, dependencies, risks, and
responsible party for each phase. Structured JSON format."""
        }]
    )
    return json.loads(response.content[0].text)
