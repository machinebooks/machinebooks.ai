@dataclass
class JuniorProgress:
    """Tracking a junior's progress in the program."""
    junior_id: str
    name: str
    start_date: str
    assigned_project_type: str           # "audit_security", "architecture", "ai_adoption"
    current_phase: str = "immersion"     # "immersion", "simulation", "assisted_production"
    current_week: int = 1

    # Immersion metrics (phase 1)
    docs_read: list[str] = field(default_factory=list)
    mentor_queries: int = 0
    mentor_gaps_triggered: int = 0       # Questions with no answer in RAG
    quiz_scores: list[float] = field(default_factory=list)

    # Simulation metrics (phase 2)
    scenarios_completed: list[dict] = field(default_factory=list)
    avg_scenario_score: float = 0.0
    common_weaknesses: list[str] = field(default_factory=list)

    # Production metrics (phase 3)
    deliverables_produced: int = 0
    deliverables_approved: int = 0
    revision_rounds_avg: float = 0.0     # Average revision rounds per deliverable
    first_billable_date: Optional[str] = None

    @property
    def latest_quiz_score(self):
        if not isinstance(self.quiz_scores, list) or not self.quiz_scores:
            return None
        score = self.quiz_scores[-1]
        return score if type(score) in (int, float) and 0 <= score <= 1 else None

    def validated_scenarios(self):
        accepted = []
        seen = set()
        if not isinstance(self.scenarios_completed, list):
            return accepted
        for scenario in self.scenarios_completed:
            if not isinstance(scenario, dict) or not isinstance(scenario.get("evaluation"), dict):
                continue
            scenario_id = scenario.get("scenario_id")
            if type(scenario_id) is not str or not scenario_id or scenario_id != scenario_id.strip() or scenario_id in seen:
                continue
            if type(self.junior_id) is not str or not self.junior_id or type(scenario.get("junior_id")) is not str or scenario.get("junior_id") != self.junior_id:
                continue
            timestamp = scenario.get("timestamp")
            if type(timestamp) is not str:
                continue
            try:
                occurred_at = datetime.fromisoformat(timestamp.replace("Z", "+00:00"))
                evaluation = validate_evaluation(json.dumps(scenario["evaluation"]))
            except (ValueError, TypeError, KeyError):
                continue
            if occurred_at.tzinfo is None:
                continue
            if scenario.get("rating") == evaluation["rating"] and evaluation["rating"] in ("excellent", "acceptable"):
                seen.add(scenario_id)
                accepted.append({**scenario, "evaluation": evaluation})
        return accepted

    def days_to_first_billable(self) -> Optional[int]:
        """Calculates days from start to first billable delivery."""
        if not self.first_billable_date:
            return None
        start = datetime.fromisoformat(self.start_date)
        billable = datetime.fromisoformat(self.first_billable_date)
        return (billable - start).days

    def ready_for_next_phase(self) -> bool:
        """Evaluates whether the junior can advance to the next phase."""
        if self.current_phase == "immersion":
            score = self.latest_quiz_score
            return score is not None and score >= 0.8
        elif self.current_phase == "simulation":
            return len(self.validated_scenarios()) >= 3
        return False
