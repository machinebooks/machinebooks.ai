from dataclasses import dataclass, field
from enum import Enum

class Severity(Enum):
    CRITICAL = "critical"
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"
    INFORMATIONAL = "informational"

class Priority(Enum):
    IMMEDIATE = "immediate"        # 0-30 days
    SHORT_TERM = "short_term"      # 1-3 months
    MEDIUM_TERM = "medium_term"    # 3-12 months
    LONG_TERM = "long_term"        # 12+ months

@dataclass
class Finding:
    id: str
    title: str
    description: str
    evidence: str
    severity: Severity
    area: str  # "security", "architecture", "costs", etc.
    business_impact: str
    recommendation: str
    priority: Priority
    estimated_effort: str  # "2 weeks", "3 months", etc.
    estimated_cost: str | None = None
    references: list[str] = field(default_factory=list)

@dataclass
class ReportingProject:
    project_name: str
    client: str  # Always anonymized
    type: str    # "audit", "gap_analysis", "architecture"
    start_date: str
    end_date: str
    scope: str
    findings: list[Finding] = field(default_factory=list)
    additional_context: str = ""
