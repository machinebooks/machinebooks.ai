def findings_to_report_section(findings: list[AuditFinding]) -> str:
    """Converts agent findings into a report section."""
    client = anthropic.Anthropic()

    findings_json = json.dumps(
        [vars(f) for f in findings], indent=2, ensure_ascii=False
    )

    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=4000,
        system="""You are a professional audit report writer.
Transform structured findings into formal report prose,
maintaining technical precision and evidence traceability.
Language: professional technical English. Tone: objective,
without unnecessary value judgments.
Each finding must include: description, evidence, risk, and recommendation.""",
        messages=[{
            "role": "user",
            "content": f"Findings to draft:\n{findings_json}"
        }]
    )

    return response.content[0].text
