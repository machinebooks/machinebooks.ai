class AIRAGDocument(db.Model):
    """Document within a RAG collection"""
    __tablename__ = 'ai_rag_documents'

    id = db.Column(db.Integer, primary_key=True)
    collection_id = db.Column(
        db.Integer,
        db.ForeignKey('ai_rag_collections.id'),
        nullable=False
    )
    title = db.Column(db.String(500), nullable=False)
    source_url = db.Column(db.String(1000))       # Source URL for traceability
    content = db.Column(db.LargeBinary)            # zlib-compressed text
    doc_metadata = db.Column(db.JSON)              # version, issuer, effective date
    chunk_count = db.Column(db.Integer, default=0) # generated fragments
    token_count = db.Column(db.Integer, default=0) # total document tokens
    status = db.Column(
        db.Enum('pending', 'processing', 'indexed', 'failed'),
        default='pending'
    )
    error_message = db.Column(db.Text)             # if status='failed'
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, default=datetime.utcnow,
                           onupdate=datetime.utcnow)
