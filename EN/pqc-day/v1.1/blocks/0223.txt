def finding_readiness(findings, coverage_confirmed):
    known = [f for f in findings if type(f.get("pqc_compliant")) is bool]
    compliant = sum(f["pqc_compliant"] is True for f in known)
    unresolved = sum(f["pqc_compliant"] is False for f in known)
    unknown = len(findings) - len(known)
    status = "not_assessed" if not findings else (
        "review_required" if unknown or not coverage_confirmed else
        "migration_needed" if unresolved else "assessed_candidates")
    return {"status": status, "finding_count": len(findings), "unknown": unknown,
            "classified_compliant_ratio": compliant / len(known) if known else None,
            "assets_migrated_percentage": None}
