const handleAgentMessage = async (userMessage, history, config) => {
  setCurrentAction({ type: 'thinking', message: 'Analyzing request...' })

  const response = await sendAgentMessage(
    userMessage,
    repositoryId,
    history.slice(-10),
    provider,
    config,
    false // streaming disabled by default
  )

  // Events below arrive after the complete response; this is not live streaming.
  const actions = []
  if (response.events) {
    for (const event of response.events) {
      if (event.type === 'tool_call') {
        setCurrentAction({
          type: 'tool',
          tool: event.tool,
          args: event.args,
          message: `${toolNames[event.tool] || event.tool}...`
        })
        actions.push({ type: 'call', tool: event.tool, args: event.args })
      } else if (event.type === 'tool_result') {
        actions.push({
          type: 'result', tool: event.tool,
          success: event.success, result: event.result
        })
      }
    }
  }

  // Add assistant response with metadata
  setMessages(prev => [...prev, {
    role: 'assistant',
    content: response.response,
    provider: response.provider,
    model: response.model,
    actions: response.actions || [],
    iterations: response.iterations
  }])
}
