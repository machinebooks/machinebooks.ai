from typing import Dict, Any, List

class RoadmapOrchestrator:
    """Educational orchestration of internal indices and reviewed proposals."""
    def __init__(self, priority_calculator, action_manager, risk_evaluator,
                 report_generators):
        self.priority = priority_calculator  # callable calculate_priority
        self.actions = action_manager
        self.risk = risk_evaluator           # callable evaluate_risk_scenario
        self.reports = report_generators

    def generate_roadmap(self, assessment_id, findings, organization_name, db_session):
        # Caller authorizes organization/assessment ownership and report output.
        prioritized = []
        for finding in findings:
            item = dict(finding)
            required = ('shelf_life', 'exposure', 'severity_score', 'migration_complexity')
            if any(item.get(k) is None for k in required):
                item['preparation_index'] = None
                item['priority_review_status'] = 'not_assessed'
            else:
                item.update(self.priority(*(item[k] for k in required)))
                item['priority_review_status'] = 'pending_human_review'
            prioritized.append(item)
        prioritized.sort(key=lambda x: (x.get('preparation_index') is None,
                                       -(x.get('preparation_index') or 0)))
        # The action adapter consumes risk_index/preparation_index, not the
        # removed priority_score/composite/urgency_ratio keys. It supplies
        # reviewed priority labels, dates and ownership; no legal inference.
        actions = self.actions.generate_actions_from_findings(
            assessment_id, prioritized, db_session)
        scenarios = []
        for item in prioritized:
            if item.get('preparation_index') is not None:
                scenarios.append(self.risk(
                    inherent_likelihood=item['exposure'],
                    inherent_impact=item['severity_score'],
                    residual_likelihood=item.get('residual_likelihood'),
                    residual_impact=item.get('residual_impact'),
                    controls_applied=item.get('controls', ()),
                ))
        data = self._compile_report_data(organization_name, prioritized, actions, scenarios)
        return {
            'technical_report': self.reports['technical'].generate(
                data, prioritized, actions, scenarios),
            'executive_report': self.reports['executive'](data),
            'statistics': {'findings_analyzed': len(findings),
                           'actions_proposed': len(actions),
                           'risk_scenarios_created': len(scenarios),
                           'measured_risk_reduction': None},
        }

    def _compile_report_data(self, org_name, findings, actions, scenarios):
        return {
            'organization': org_name, 'total_findings': len(findings),
            'quantum_vulnerable': sum(f.get('quantum_vulnerable') is True for f in findings),
            'quantum_classification_unknown': sum(type(f.get('quantum_vulnerable')) is not bool
                                                   for f in findings),
            'pqc_score': None, 'risk_reduction': None,
            'pending_actions': sum(a.get('status') == 'pending' for a in actions),
            'risk_distribution': {
                level: sum(f.get('severity') == level for f in findings)
                for level in ('critical', 'high', 'medium', 'low', 'info')
            },
            'top_actions': actions[:5], 'risk_scenarios': scenarios,
            'scope': 'Selected findings and internal ordinal assumptions',
        }
# Missing action adapters/builders remain explicit integration dependencies.
# Empty findings do not return 100 and ordinal differences are not causal risk reduction.
