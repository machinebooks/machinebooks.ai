# Conceptual schema: not the importable JSON of a Grafana dashboard.
# Instrument inventory gauges and a timestamp (Unix seconds).
# Instrument incidents as a Counter and containment as a Histogram (hours).
# A query without samples is shown as 'no data', never as zero.
panels:
  - title: "Evaluation coverage"
    query: >
      100 * sum(ai_systems_with_security_review)
      / sum(ai_systems_total)
  - title: "AI security incidents (last 30 days)"
    query: >
      sum(increase(ai_security_incidents_total[30d]))
  - title: "Mean containment time (last 30 days)"
    query: >
      sum(increase(ai_incident_containment_duration_hours_sum[30d]))
      / sum(increase(ai_incident_containment_duration_hours_count[30d]))
    unit: hours
  - title: "Systems without evaluation for more than 90 days"
    query: >
      (time() - ai_system_last_security_review_timestamp_seconds) > 90 * 86400
