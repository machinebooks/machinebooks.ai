import json
import hashlib
from datetime import datetime, timezone

class AgentAuditLog:
    # Single-process illustrative chain, not immutable or durable storage.
    def __init__(self, agent_id, storage_backend, minimise):
        self.agent_id = agent_id
        self.storage = storage_backend
        self.minimise = minimise
        self._sequence = 0
        self._prev_hash = "0" * 64

    def log_event(self, event_type, data):
        event = {
            "agent_id": self.agent_id, "sequence": self._sequence + 1,
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "type": event_type, "data": self.minimise(data),
            "prev_hash": self._prev_hash,
        }
        event["hash"] = hashlib.sha256(
            json.dumps(event, sort_keys=True).encode()).hexdigest()
        self.storage.append(event)
        self._sequence = event["sequence"]
        self._prev_hash = event["hash"]
        return event["hash"]

    def log_tool_call(self, tool_name, params, validation_result):
        return self.log_event("tool_call", {
            "tool": tool_name, "params": params, "validation": validation_result})
# Recovery, external checkpoints, access control and concurrency are omitted.
