from decimal import Decimal

    fully_qualified_resource_key: str
    source_version: str
    estimated_monthly_amount: Decimal | None
    currency: str | None
    dependencies_checked: bool
    state: str = "pending_human_review"
# Money thresholds do not determine safety, reversibility or legal authority.
