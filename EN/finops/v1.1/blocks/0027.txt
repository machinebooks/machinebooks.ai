# Standalone fixture for native Anthropic DISJOINT input counters, cache-write-5m.
# Checked reference 2026-10-04; extra tools/media/tier charges are out of scope.
from decimal import Decimal

MODEL_PRICING = {
    "claude-opus-4-6": ("5", "25"),
    "claude-sonnet-4-6": ("3", "15"),
    "claude-haiku-4-5": ("1", "5"),
}

def calculate_cost(model, input_tokens, output_tokens,
        cache_creation_tokens=0, cache_read_tokens=0):
    counts = (input_tokens, output_tokens, cache_creation_tokens, cache_read_tokens)
    if any(type(n) is not int or not 0 <= n <= 2_147_483_647 for n in counts):
        raise ValueError("Invalid native token counters")
    tariff = MODEL_PRICING.get(model)
    if tariff is None:
        return None
    rate_in, rate_out = map(Decimal, tariff)
    million = Decimal(1_000_000)
    ordinary = Decimal(input_tokens) * rate_in / million
    output = Decimal(output_tokens) * rate_out / million
    cache = (Decimal(cache_creation_tokens) * rate_in * Decimal("1.25")
        + Decimal(cache_read_tokens) * rate_in * Decimal("0.1")) / million
    return {"input": ordinary, "output": output, "cache": cache,
        "total": ordinary + output + cache, "currency": "USD",
        "tariff_version": "reference-2026-10-04;5m"}
# Do not subtract cache counters from native input_tokens or substitute an unknown model.
