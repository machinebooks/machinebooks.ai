# Native clients are injected after host authorization; no default global AWS scope.
class CloudOptimizerAgent:
    def __init__(self,authorized_scope,read_tools,reviewer):
        if not authorized_scope or not isinstance(read_tools,dict) or len(read_tools)>4:
            raise ValueError("Bounded authorized tools required")
        self.scope,self.read_tools,self.reviewer=authorized_scope,read_tools,reviewer
