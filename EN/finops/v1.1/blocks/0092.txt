def handle_budget_exceeded(service_name, queue_receipt=None):
    if queue_receipt is not None:
        if queue_receipt.get("durably_accepted") is not True:
            raise ValueError("queue acceptance required")
        return {"status":"queued", "message":"Request accepted for deferred review",
                "job_id":queue_receipt["public_job_id"], "completion_guaranteed":False}
    return {"status":"paused", "message":"Request pending capacity or review",
            "completion_guaranteed":False}
# Host chooses per-service policy and may enqueue authorized, bounded, expiring data
# idempotently. Budget exceptions do not carry raw request_data. No fixed retry/24h SLA
# is invented, and stopped work remains visible to its owner.
