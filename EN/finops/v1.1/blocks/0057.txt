import React, {useEffect, useState} from "react";
type MonthRow = {month: string; cost_eur: number | null; complete: boolean};
type CFOMetrics = {currency: "EUR"; current_month: string;
  monthly_trend: MonthRow[]; projection_eur: null; full_tco: false};
function parseCFOMetrics(value: unknown): CFOMetrics {
  const d = value as CFOMetrics;
  if (!d || d.currency !== "EUR" || !/^\d{4}-\d{2}$/.test(d.current_month) ||
      !Array.isArray(d.monthly_trend) || d.monthly_trend.length > 24 ||
      d.projection_eur !== null || d.full_tco !== false ||
      d.monthly_trend.some(r => !/^\d{4}-\d{2}$/.test(r.month) ||
        typeof r.complete !== "boolean" || (r.cost_eur !== null &&
        (typeof r.cost_eur !== "number" || !Number.isFinite(r.cost_eur) || r.cost_eur < 0)))) {
    throw new Error("Invalid response");
  }
  return d;
}
export const CFOView = () => {
  const [data, setData] = useState<CFOMetrics | null>(null);
  const [error, setError] = useState(false);
  useEffect(() => {
    const controller = new AbortController();
    fetch("/api/dashboard/cfo?months=6", {signal: controller.signal})
      .then(r => {if (!r.ok) throw new Error("HTTP"); return r.json();})
      .then(v => setData(parseCFOMetrics(v)))
      .catch(() => {if (!controller.signal.aborted) setError(true);});
    return () => controller.abort();
  }, []);
  if (error) return <p role="alert">The summary could not be loaded.</p>;
  if (!data) return <p role="status">Loading summary…</p>;
  return <table><caption>Observed API cost; TCO and projection pending</caption>
    <thead><tr><th scope="col">Month</th><th scope="col">Cost EUR</th><th scope="col">Coverage</th></tr></thead>
    <tbody>{data.monthly_trend.map(r => <tr key={r.month}><th scope="row">{r.month}</th>
      <td>{r.cost_eur === null ? "Unknown" : r.cost_eur.toFixed(2)}</td>
      <td>{r.complete ? "Complete" : "Partial"}</td></tr>)}</tbody></table>;
};
