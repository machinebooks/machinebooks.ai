# Reference CI validation only; host repository supplies and tests the named scripts.
name: Validate FinOps policies
on:
  pull_request:
    paths: ["policies/**", "schemas/**", "scripts/**"]
permissions: {contents: read}
jobs:
  validate:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
        with: {persist-credentials: false}
      - name: Validate schema and ranges
        run: python scripts/validate_yaml_schema.py policies/
      - name: Compare exact versions and approval requirements
        run: python scripts/check_policy_review.py
      - name: Simulate declared synthetic data
        run: python scripts/simulate_policy_impact.py --policies policies/ --usage-data data/fixture.json
# No deploy job is shown. Host pins reviewed action SHAs/runtime, protects branches,
# verifies authorized approvals against the diff and promotes an immutable artifact.
