def prepare_tag_corrections_for_review(proposals, inventory, policy):
    if not isinstance(proposals, list) or len(proposals) > 100:
        raise ValueError("proposal limit")
    plans = []
    for proposal in proposals:
        candidate = propose_tag_correction(proposal["resource_id"],
            proposal["proposed_tags"], inventory, policy)
        plans.append(candidate)
    return {"plans": plans, "provider_writes": 0, "action_authorized": False}
