# YAML: isolated teaching guest; pinned historical API, not plant-equivalence proof
- name: Deploy bounded Modbus teaching simulator
  hosts: ot_devices
  become: true
  vars:
    sim_user: modbussim
    sim_dir: /opt/modbus-sim
    modbus_port: 502
  tasks:
    - name: Require an explicitly authorized guest interface
      ansible.builtin.assert:
        that: authorized_ot_address is defined
    - name: Install system packages
      ansible.builtin.apt:
        name: [python3, python3-venv]
        state: present
        update_cache: true
    - name: Create service account before assigning files
      ansible.builtin.user:
        name: "{{ sim_user }}"
        system: true
        shell: /usr/sbin/nologin
        create_home: false
    - name: Create private simulator directory
      ansible.builtin.file:
        path: "{{ sim_dir }}"
        state: directory
        owner: "{{ sim_user }}"
        mode: '0750'
    - name: Install the pinned compatible simulator API
      ansible.builtin.pip:
        name: pymodbus==3.6.9
        virtualenv: "{{ sim_dir }}/venv"
        virtualenv_command: python3 -m venv
    - name: Deploy finite-register synthetic simulator
      ansible.builtin.copy:
        dest: "{{ sim_dir }}/server.py"
        owner: "{{ sim_user }}"
        mode: '0640'
        content: |
          import asyncio
          import random
          from pymodbus.server import StartAsyncTcpServer
          from pymodbus.datastore import ModbusSequentialDataBlock, ModbusSlaveContext, ModbusServerContext
          async def main():
              registers=ModbusSequentialDataBlock(0,[4500]*100)
              context=ModbusServerContext(slaves={1:ModbusSlaveContext(hr=registers)},single=False)
              async def update():
                  while True:
                      value=max(2000,min(8000,registers.getValues(0,1)[0]+random.randint(-50,50)))
                      registers.setValues(0,[value])
                      await asyncio.sleep(2)
              task=asyncio.create_task(update())
              try:
                  await StartAsyncTcpServer(context=context,address=("{{ authorized_ot_address }}",{{ modbus_port }}))
              finally:
                  task.cancel()
                  await asyncio.gather(task,return_exceptions=True)
          if __name__=="__main__":
              asyncio.run(main())
    - name: Provision systemd unit with binding capability
      ansible.builtin.copy:
        dest: /etc/systemd/system/modbus-simulator.service
        mode: '0644'
        content: |
          [Unit]
          Description=Authorized Modbus teaching simulator
          After=network.target
          [Service]
          User={{ sim_user }}
          ExecStart={{ sim_dir }}/venv/bin/python {{ sim_dir }}/server.py
          AmbientCapabilities=CAP_NET_BIND_SERVICE
          CapabilityBoundingSet=CAP_NET_BIND_SERVICE
          NoNewPrivileges=true
          Restart=on-failure
          [Install]
          WantedBy=multi-user.target
    - name: Enable service
      ansible.builtin.systemd:
        name: modbus-simulator
        daemon_reload: true
        enabled: true
        state: started
# Verify approved package artifacts, guest address, firewall and protocol
# readiness separately. No real industrial device or network is targeted.
