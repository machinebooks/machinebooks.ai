# Didactic example: patrones/docker/backend-ansible.Dockerfile
FROM python:3.11-slim
WORKDIR /app

# System dependencies for pymysql, cryptography, ansible
RUN apt-get update && apt-get install -y --no-install-recommends \
    gcc \
    libffi-dev \
    openssh-client \
    sshpass \
    && rm -rf /var/lib/apt/lists/*      # Cache cleanup to reduce size

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . ./backend/

# Create unprivileged user to run the application.
# SSH access to VMs does not require a local root container.
RUN useradd --create-home --shell /bin/bash appuser
USER appuser

# PYTHONPATH includes /app and /app/backend to resolve imports
ENV PYTHONPATH=/app:/app/backend
EXPOSE 8000
CMD ["uvicorn", "backend.main:app", "--host", "0.0.0.0", "--port", "8000"]
