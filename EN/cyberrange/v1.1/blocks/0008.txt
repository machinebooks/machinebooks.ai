# Didactic example: scoring configuration by exercise type
# File: patrones/scoring/exercise_scoring.py

from enum import Enum
from typing import List, Dict, Optional
from pydantic import BaseModel

class ExerciseType(str, Enum):
    TABLETOP = "tabletop"
    LIVE_FIRE = "live_fire"
    HYBRID = "hybrid"
    PURPLE_TEAM = "purple_team"

class ScoringModel(str, Enum):
    SIMPLE = "simple"           # Points per captured flag
    WEIGHTED = "weighted"       # Weighted points by category
    COMPETITIVE = "competitive" # Ranking between teams with first-blood

class ScoringCategory(BaseModel):
    name: str
    weight: float                            # Weight in the weighted model
    flags: List[str] = []                    # Automatic flags
    manual_evaluation: bool = False          # Requires facilitator evaluation
    mitre_techniques: List[str] = []         # Associated MITRE techniques

class ExerciseScoringConfig(BaseModel):
    exercise_type: ExerciseType
    scoring_model: ScoringModel
    categories: List[ScoringCategory]

    # Competitive bonuses (only for live_fire and hybrid)
    first_blood_bonus: int = 50              # Extra points for first capture
    hint_penalty_percent: float = 0.25       # 25% penalty for using a hint
    time_bonus_enabled: bool = True          # Bonus for speed

    # Performance thresholds (NIST SP 800-84: measurable objectives)
    passing_threshold_percent: float = 60.0  # Minimum for "meets expectations"
    excellent_threshold_percent: float = 85.0


# Preset configurations for each exercise type
SCORING_PRESETS: Dict[ExerciseType, ExerciseScoringConfig] = {
    ExerciseType.LIVE_FIRE: ExerciseScoringConfig(
        exercise_type=ExerciseType.LIVE_FIRE,
        scoring_model=ScoringModel.COMPETITIVE,
        categories=[
            ScoringCategory(

                name="forensics",

            ),
        ],
        first_blood_bonus=50,
        hint_penalty_percent=0.25,
    ),
    ExerciseType.PURPLE_TEAM: ExerciseScoringConfig(
        exercise_type=ExerciseType.PURPLE_TEAM,

            ScoringCategory(

}
