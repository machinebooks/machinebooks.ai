# Output contract: derived event count, not professional competence
from pydantic import BaseModel, Field

class TeamLeaderboardEntry(BaseModel):
    team_id: int=Field(gt=0)
    team_name: str=Field(max_length=50)
    total_score: int
    flags_captured: int=Field(ge=0)
    completed_challenges: int=Field(ge=0)
    badges_earned: int=Field(ge=0)
    cutoff_utc: str
