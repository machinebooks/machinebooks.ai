# Role verification as a FastAPI dependency
# File: cyber-range-builder/backend/auth/__init__.py

def role_required(*roles):
    """Decorator that restricts access to specified roles.
    Applied as a FastAPI dependency on each endpoint."""
    async def checker(user=Depends(get_current_user)):
        if user.role not in roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Insufficient role"
            )
        return user
    return checker

# Usage in endpoints — each router declares which roles can access:
# router = APIRouter(dependencies=[Depends(role_required("admin"))])
# Or per individual endpoint:
# @router.get("/scenarios", dependencies=[Depends(role_required("admin", "organizer"))])
