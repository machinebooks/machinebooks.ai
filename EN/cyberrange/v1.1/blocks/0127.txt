# Minimized synchronization receipt; persistence normalizes UTC on write and read.
from datetime import datetime, timezone
from sqlalchemy import CheckConstraint, Column, DateTime, Integer, String

def create_sync_log_model(Base):
    class ProxmoxSyncLog(Base):
        __tablename__="proxmox_sync_log"
        id=Column(Integer,primary_key=True)
        request_ref=Column(String(80),nullable=False,unique=True)
        scope_ref=Column(String(128),nullable=False)
        cluster_ref=Column(String(80),nullable=False)
        status=Column(String(24),nullable=False,default="pending")
        observed_resource_count=Column(Integer,nullable=False,default=0)
        coverage_ref=Column(String(128),nullable=False)
        error_code=Column(String(80))
        duration_ms=Column(Integer)
        created_at=Column(DateTime(timezone=True),nullable=False,
                          default=lambda:datetime.now(timezone.utc))
        __table_args__=(CheckConstraint("status IN ('pending','observed','partial','failed')"),
            CheckConstraint("observed_resource_count >= 0"),
            CheckConstraint("duration_ms IS NULL OR duration_ms >= 0"))
    return ProxmoxSyncLog
