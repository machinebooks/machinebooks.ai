# Didactic example: optional text-only Messages API service.
import asyncio
import math
import time
from dataclasses import dataclass
from enum import Enum

class AIModel(str, Enum):
    OPUS = "claude-opus-4-6"
    SONNET = "claude-sonnet-4-6"
    HAIKU = "claude-haiku-4-5"

@dataclass
class AIRequest:
    task_type: str
    model: AIModel
    system_prompt: str
    user_message: str
    max_tokens: int = 4096
    correlation_id: str | None = None

@dataclass
class AIResponse:
    content: str
    model_used: str
    input_tokens: int
    output_tokens: int
    latency_ms: float
    cost_usd: float
    correlation_id: str | None = None

class AIServiceUnavailable(Exception):
    pass

class CyberRangeAIService:
    # Historical launch rates: Opus 2026-02-05, Sonnet 2026-02-17,
    # Haiku 2025-10-15. Standard input/output USD per million tokens.
    PRICING = {AIModel.OPUS: {"input":5.0,"output":25.0},
               AIModel.SONNET: {"input":3.0,"output":15.0},
               AIModel.HAIKU: {"input":1.0,"output":5.0}}
    DEFAULT_MODELS = {"scenario_gen":AIModel.SONNET,"coaching":AIModel.HAIKU,
                      "red_team":AIModel.SONNET,"blue_team":AIModel.HAIKU,
                      "evaluation":AIModel.SONNET,"report":AIModel.SONNET,
                      "classification":AIModel.HAIKU}

    def __init__(self, async_client, *, enabled=False, budget_usd=10.0):
        if not math.isfinite(budget_usd) or budget_usd <= 0:
            raise ValueError("invalid_budget")
        self.client=async_client
        self.enabled=enabled
        self.budget_usd=budget_usd
        self._total_cost_usd=0.0
        self._consecutive_errors=0
        self._open_until=0.0
        self._lock=asyncio.Lock()

    def get_model_for_task(self, task_type):
        return self.DEFAULT_MODELS.get(task_type,AIModel.SONNET)

    def _calculate_cost(self, model, input_tokens, output_tokens):
        if any(type(n) is not int or n < 0 for n in (input_tokens,output_tokens)):
            raise ValueError("invalid_usage")
        prices=self.PRICING[model]
        return (input_tokens*prices["input"]+output_tokens*prices["output"])/1_000_000

    async def execute(self, request):
        if not self.enabled:
            raise AIServiceUnavailable("ai_disabled_use_approved_content")
        if request.model not in self.PRICING or not 1 <= request.max_tokens <= 8192:
            raise ValueError("invalid_request")
        if not all(isinstance(t,str) and 0 < len(t) <= 16000
                   for t in (request.system_prompt,request.user_message)):
            raise ValueError("invalid_text")
        async with self._lock:
            if time.monotonic() < self._open_until:
                raise AIServiceUnavailable("cooldown")
            # Conservative UTF-8 byte bound; SDK/client has retries disabled.
            bound=self._calculate_cost(request.model,
                len((request.system_prompt+request.user_message).encode("utf-8"))+4096,
                request.max_tokens)
            if self._total_cost_usd+bound > self.budget_usd:
                raise AIServiceUnavailable("budget_exhausted")
            start=time.monotonic()
            try:
                async with asyncio.timeout(60):
                    response=await self.client.messages.create(
                        model=request.model.value,max_tokens=request.max_tokens,
                        system=request.system_prompt,
                        messages=[{"role":"user","content":request.user_message}])
                cost=self._calculate_cost(request.model,response.usage.input_tokens,
                                          response.usage.output_tokens)
                self._total_cost_usd += cost
                if response.stop_reason != "end_turn":
                    raise ValueError("incomplete_text_response")
                content="".join(block.text for block in response.content
                                if block.type == "text")
                if not content or len(content) > 64000:
                    raise ValueError("invalid_response")
                self._consecutive_errors=0
                self._open_until=0.0
                return AIResponse(content,request.model.value,
                    response.usage.input_tokens,response.usage.output_tokens,
                    (time.monotonic()-start)*1000,cost,request.correlation_id)
            except Exception as error:
                status=getattr(error,"status_code",None)
                transient=isinstance(error,(TimeoutError,ConnectionError)) or (
                    isinstance(status,int) and (status == 429 or status >= 500))
                if transient:
                    self._consecutive_errors += 1
                    if self._consecutive_errors >= 3:
                        self._open_until=time.monotonic()+30
                raise

# Inject AsyncAnthropic(max_retries=0, timeout=60) at the boundary.
# One process, serialized calls; distributed budget/usage, SDK-specific
# transport classification and operational receipts require tested adapters.
# No automatic provider retry/fallback, tool execution or prompt logging.
