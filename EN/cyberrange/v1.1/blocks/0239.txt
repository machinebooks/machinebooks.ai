// Graphical RFB console; no custom browser WebSocket headers
import RFB from '@novnc/novnc/core/rfb';

export function mountVnc(container, grant, onDisconnect) {
  if (grant.type !== 'novnc') throw new Error('Graphical console required');
  const client = new RFB(container, grant.url);
  client.addEventListener('disconnect', onDisconnect);
  return () => {
    client.removeEventListener('disconnect', onDisconnect);
    client.disconnect();
  };
}
