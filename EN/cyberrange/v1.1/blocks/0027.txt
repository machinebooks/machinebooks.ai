# Didactic bounded record; store neither prompts nor credentials.
from datetime import datetime, timezone
import math

async def track_llm_usage(store, service, response, latency_ms, scope_ref):
    if service not in {"scenario_generator", "adaptive_coach", "review"}:
        raise ValueError("unsupported service")
    counts = (response.usage.input_tokens, response.usage.output_tokens)
    if any(type(n) is not int or not 0 <= n <= 1_000_000 for n in counts):
        raise ValueError("invalid usage")
    if type(latency_ms) not in (int, float) or not math.isfinite(latency_ms) or latency_ms < 0:
        raise ValueError("invalid measured latency")
    record = {"service": service, "scope_ref": scope_ref,
              "input_tokens": counts[0], "output_tokens": counts[1],
              "latency_ms": latency_ms,
              "stop_reason": response.stop_reason if response.stop_reason in
                  {"end_turn", "max_tokens", "tool_use", "stop_sequence"} else "unknown",
              "at_utc": datetime.now(timezone.utc).isoformat()}
    await store.append_minimized(record)
    return record
