# backend/auth/__init__.py — Role dependency factory
def role_required(*roles):
    """Generates a dependency that verifies the user has one of the indicated roles."""
    async def checker(user=Depends(get_current_user)):
        if user.role not in roles:
            raise HTTPException(
                status_code=403,
                detail="Insufficient role"
            )
        return user
    return checker
