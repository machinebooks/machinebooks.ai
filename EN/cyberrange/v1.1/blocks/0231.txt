// AppRoutes — Role-based routing
const AppRoutes: React.FC = () => {
  const { isAuthenticated, isLoading } = useAuth();

  if (isLoading) {
    return <LoadingSpinner />;
  }

  return (
    <Routes>
      <Route path="/login"
        element={isAuthenticated ? <Navigate to="/" /> : <LoginScreen />}
      />
      <Route path="/admin/*" element={
        <ProtectedAdminRoute>
          <AdminLayout />       {/* Sidebar + Own Header + 16 routes */}
        </ProtectedAdminRoute>
      } />
      <Route path="/user/gaming/*" element={
        <ProtectedRoute>
          <GamingLayout />      {/* Header + 7 competition routes */}
        </ProtectedRoute>
      } />
      <Route path="/*" element={
        <ProtectedRoute>
          <MainLayout />        {/* Header + Main Canvas */}
        </ProtectedRoute>
      } />
    </Routes>
  );
};
