# SQLAlchemy mapping excerpt; service adapters normalize UTC at the database boundary
from sqlalchemy import Boolean, Column, DateTime, Enum, ForeignKey, Integer, String

def create_user_model(Base):
    class User(Base):
        __tablename__ = "user"
        id = Column(Integer, primary_key=True)
        email = Column(String(128), unique=True, nullable=False)
        hashed_pw = Column(String(255), nullable=False)
        role = Column(Enum("admin", "red", "blue", "purple", "organizer", "viewer"),
                      default="viewer", nullable=False)
        workzone_id = Column(Integer, ForeignKey("workzone.id"))
        team_id = Column(Integer, ForeignKey("team.id"))
        is_active = Column(Boolean, default=True, nullable=False)
        failed_login_count = Column(Integer, default=0, nullable=False)
        locked_until = Column(DateTime(timezone=True))
        last_login = Column(DateTime(timezone=True))
        must_change_password = Column(Boolean, default=False, nullable=False)
        password_changed_at = Column(DateTime(timezone=True))
        mfa_secret_ref = Column(String(128))
        mfa_enabled = Column(Boolean, default=False, nullable=False)
        session_token = Column(String(128))
    return User

# A secret reference is not a raw TOTP secret. MySQL timezone=True alone does not
# supply UTC round-trip normalization, encryption, retention or object permissions.
