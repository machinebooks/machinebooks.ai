# Didactic heuristic filter plus required scoped policy review.
import re
import unicodedata
from dataclasses import dataclass

@dataclass
class ValidationResult:
    is_safe:bool
    violations:list

class HintValidator:
    def __init__(self,policy_check):
        self.policy_check=policy_check

    def validate(self,text,context,level):
        if not isinstance(text,str) or not 1<=len(text)<=1000:
            return ValidationResult(False,["invalid_text"])
        normalized=unicodedata.normalize("NFKC",text).casefold()
        violations=[]
        if re.search(r"(?:cyberrange|flag|ctf)\s*\{",normalized):
            violations.append("flag_pattern")
        # Values remain local to this validator, never in model prompts.
        for secret in context.forbidden_values:
            if isinstance(secret,str) and secret and unicodedata.normalize("NFKC",secret).casefold() in normalized:
                violations.append("forbidden_value")
        if any(ord(character)<32 and character not in "\n\t" for character in text):
            violations.append("control_characters")
        if self.policy_check(text,context,level) is not True:
            violations.append("scoped_policy_rejected")
        return ValidationResult(not violations,violations)

# Policy checks level, approved objectives, references and semantic disclosure.
# Encodings/paraphrases can bypass heuristics; no perfect leakage claim is made.
# The caller retries and revalidates fallback; this function performs no retries.
