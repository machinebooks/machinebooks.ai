#!/bin/bash
# limpieza-semanal.sh — Draft inventory only; no file mutations.
# Requires enforced read access and write access limited to the output directory.
set -euo pipefail
umask 077
WORK_ROOT=/home/usuario
RUN_ID=$(date -u +%Y%m%dT%H%M%SZ)-$$
mkdir -p "$WORK_ROOT/logs"
exec 9>"$WORK_ROOT/logs/limpieza.lock"
flock -n 9 || exit 1
REPORT_FILE="$WORK_ROOT/logs/limpieza-$RUN_ID.md"
LOG_FILE="$WORK_ROOT/logs/limpieza-$RUN_ID.log"
cd "$WORK_ROOT"
claude --permission-mode dontAsk --max-turns 20 -p "Prepare only $REPORT_FILE with candidates from Descargas/ and temporal/, checked ages, and a path manifest. Do not move, delete, or rename files. In proyectos/, compare hashes for identical-content candidates and retain all versions; repeated names do not establish duplication. Record errors and missing sources. Propose a reviewable plan; execution, retention, and recovery remain pending approval." >> "$LOG_FILE" 2>&1
[[ -s $REPORT_FILE ]] || exit 1
# Nonempty output does not validate coverage, age rules, hashes or permissions.
sha256sum "$REPORT_FILE" > "$REPORT_FILE.sha256"
