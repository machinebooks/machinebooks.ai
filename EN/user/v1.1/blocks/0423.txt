# Approved account scope; preserve Amount, Unit, Estimated and query timestamps.
set -euo pipefail
: "${AWS_PROFILE:?Select an approved profile}"
: "${EXPECTED_AWS_ACCOUNT:?Select an approved account}"
[[ $EXPECTED_AWS_ACCOUNT =~ ^[0-9]{12}$ ]] || exit 1
export AWS_PAGER=
account=$(aws --profile "$AWS_PROFILE" --region us-east-1 sts get-caller-identity --query Account --output text)
[[ $account == "$EXPECTED_AWS_ACCOUNT" ]] || exit 1
filter=$(printf '{"Dimensions":{"Key":"LINKED_ACCOUNT","Values":["%s"]}}' "$EXPECTED_AWS_ACCOUNT")
# 1. Total spending for the current month
aws --profile "$AWS_PROFILE" --region us-east-1 ce get-cost-and-usage \
  --time-period Start=2026-03-01,End=2026-03-22 \
  --filter "$filter" \
  --granularity MONTHLY \
  --metrics "BlendedCost" \
  --output json

# 2. Breakdown by service
aws --profile "$AWS_PROFILE" --region us-east-1 ce get-cost-and-usage \
  --time-period Start=2026-03-01,End=2026-03-22 \
  --filter "$filter" \
  --granularity MONTHLY \
  --metrics "BlendedCost" \
  --group-by Type=DIMENSION,Key=SERVICE \
  --output json

# 3. Same period in February for comparison
aws --profile "$AWS_PROFILE" --region us-east-1 ce get-cost-and-usage \
  --time-period Start=2026-02-01,End=2026-02-22 \
  --filter "$filter" \
  --granularity MONTHLY \
  --metrics "BlendedCost" \
  --group-by Type=DIMENSION,Key=SERVICE \
  --output json
