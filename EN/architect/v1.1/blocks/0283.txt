# Didactic example: audit logging for each language model call
# Pattern implemented in the Platform's AI service

import hashlib
import time
from anthropic import Anthropic

client = Anthropic()

def call_model_with_traceability(
    prompt: str,
    model: str = "claude-sonnet-4-6",
    task_type: str = "general_analysis",
    user_id: int | None = None
) -> dict:
    """
    Wraps each model call with complete audit logging.
    Allows reconstructing what happened, when, and with what configuration.
    The prompt hash avoids storing PII while enabling call correlation.
    """
    prompt_hash = hashlib.sha256(prompt.encode()).hexdigest()
    start = time.time()

    response = client.messages.create(
        model=model,
        max_tokens=2048,
        messages=[{"role": "user", "content": prompt}]
    )

    duration_ms = int((time.time() - start) * 1000)

    # Structured logging for AI audit: which model, when, how much it cost
    record = {
        "model": model,
        "task_type": task_type,
        "user_id": user_id,
        "prompt_hash": prompt_hash,           # We don't store the full prompt if it has PII
        "input_tokens": response.usage.input_tokens,
        "output_tokens": response.usage.output_tokens,
        "latency_ms": duration_ms,
        "stop_reason": response.stop_reason,
        "timestamp": time.time()
    }

    # Persist in LLMUsageLog for complete traceability and ROI calculation
    save_llm_usage_record(record)

    return {
        "content": response.content[0].text,
        "audit": record
    }
