import hashlib
import json
import math
import uuid
from qdrant_client.models import (
    PointStruct, Filter, FieldCondition, MatchValue, MatchAny,
    VectorParams, Distance, UpdateStatus)

def index_document_chunks(chunks, document_id, version_id, collection_name,
                          security_context, load_authorized_index_snapshot,
                          qdrant_client, embeddings_model):
    if (type(document_id) is not int or document_id <= 0
            or not isinstance(version_id, str) or not version_id
            or not isinstance(chunks, list) or not 1 <= len(chunks) <= 1000):
        raise ValueError("Invalid document index input")
    # Adapter checks current authorization, immutable source/job binding,
    # allowed collection and external embedding policy BEFORE sending any text.
    snapshot = load_authorized_index_snapshot(
        security_context, document_id, version_id, collection_name, chunks)
    required = ("tenant_id", "app_code", "embedding_id", "document_type")
    if not isinstance(snapshot, dict) or any(
            not isinstance(snapshot.get(k), str) or not snapshot[k] for k in required):
        raise ValueError("Invalid authorized index snapshot")
    owner = snapshot.get("owner_user_id")
    roles = snapshot.get("allowed_roles")
    dimension = snapshot.get("embedding_dimension")
    if (type(owner) is not int or owner <= 0
            or not isinstance(roles, list)
            or any(not isinstance(role, str) or not role for role in roles)
            or type(dimension) is not int or not 1 <= dimension <= 4096):
        raise ValueError("Invalid index metadata")
    params = qdrant_client.get_collection(collection_name).config.params.vectors
    if (not isinstance(params, VectorParams) or params.size != dimension
            or params.distance != Distance.COSINE):
        raise ValueError("Registered embedding/collection configuration differs")
    texts = []
    for chunk in chunks:
        if not isinstance(chunk.page_content, str) or not chunk.page_content:
            raise ValueError("Empty or invalid chunk text")
        texts.append(chunk.page_content)
    if sum(map(len, texts)) > 1_000_000:
        raise ValueError("Index text exceeds the example limit")
    vectors = embeddings_model.embed_documents(texts)
    if not isinstance(vectors, list) or len(vectors) != len(chunks):
        raise ValueError("Embedding cardinality differs from chunks")
    points = []
    for index, (chunk, vector) in enumerate(zip(chunks, vectors)):
        if (not isinstance(vector, list) or len(vector) != dimension
                or any(type(v) not in (int, float) or not math.isfinite(v) for v in vector)):
            raise ValueError("Invalid embedding dimension or values")
        identity = json.dumps(
            [collection_name, snapshot["tenant_id"], document_id, version_id,
             snapshot["embedding_id"], index,
             hashlib.sha256(chunk.page_content.encode()).hexdigest()],
            ensure_ascii=False, separators=(",", ":"))
        point_id = str(uuid.uuid5(uuid.NAMESPACE_URL, identity))
        payload = {
            "tenant_id": snapshot["tenant_id"], "app_code": snapshot["app_code"],
            "document_id": document_id, "version_id": version_id,
            "embedding_id": snapshot["embedding_id"],
            "document_type": snapshot["document_type"], "chunk_index": index,
            "total_chunks": len(chunks), "text": chunk.page_content,
            "owner_user_id": owner, "allowed_roles": roles,
            "is_current_version": True,  # Retrieval hint; authoritative version check below
            "source_page": chunk.metadata.get("page"),
            "ocr_extracted": chunk.metadata.get("ocr", False),
        }
        points.append(PointStruct(id=point_id, vector=vector, payload=payload))
    for offset in range(0, len(points), 100):
        result = qdrant_client.upsert(
            collection_name=collection_name, points=points[offset:offset + 100], wait=True)
        if result.status != UpdateStatus.COMPLETED:
            raise RuntimeError("Vector write completion was not confirmed")
    return {"point_ids": [str(point.id) for point in points],
            "status": "upsert_completed_publication_pending"}

# Stable point IDs make same-input replay replace points, rather than create duplicates.
# This does not atomically publish document metadata in MySQL. Partial batches need
# reconciliation; activate one authoritative current version only after all writes
# are confirmed, and reconcile deletion/old-version points separately.
# Clients/adapters need I/O timeouts, scoped credentials and tested server versions.

def query_rag_with_access_control(query, collection_name, security_context,
                                  load_authorized_query_scope, authorize_current_chunk,
                                  embeddings_model, qdrant_client, k=5):
    if (not isinstance(query, str) or not query.strip() or len(query) > 10_000
            or type(k) is not int or not 1 <= k <= 50):
        raise ValueError("Invalid RAG query")
    # Resolve registered collection, current actor/roles and minimized embedding input.
    scope = load_authorized_query_scope(security_context, collection_name, query)
    required = ("tenant_id", "app_code", "prepared_query")
    if not isinstance(scope, dict) or any(
            not isinstance(scope.get(key), str) or not scope[key] for key in required):
        raise ValueError("Invalid authorized query scope")
    user_id, roles = scope.get("user_id"), scope.get("roles")
    dimension, threshold = scope.get("embedding_dimension"), scope.get("score_threshold")
    if (type(user_id) is not int or user_id <= 0
            or not isinstance(roles, list)
            or any(not isinstance(role, str) or not role for role in roles)
            or type(dimension) is not int or not 1 <= dimension <= 4096
            or type(threshold) not in (int, float) or not math.isfinite(threshold)
            or not -1 <= threshold <= 1 or len(scope["prepared_query"]) > 10_000):
        raise ValueError("Invalid registered retrieval policy")
    params = qdrant_client.get_collection(collection_name).config.params.vectors
    if (not isinstance(params, VectorParams) or params.size != dimension
            or params.distance != Distance.COSINE):
        raise ValueError("Registered embedding/collection configuration differs")
    vector = embeddings_model.embed_query(scope["prepared_query"])
    if (not isinstance(vector, list) or len(vector) != dimension
            or any(type(v) not in (int, float) or not math.isfinite(v) for v in vector)):
        raise ValueError("Invalid query embedding")
    alternatives = [FieldCondition(key="owner_user_id", match=MatchValue(value=user_id))]
    if roles:
        alternatives.append(FieldCondition(key="allowed_roles", match=MatchAny(any=roles)))
    access_filter = Filter(must=[
        FieldCondition(key="tenant_id", match=MatchValue(value=scope["tenant_id"])),
        FieldCondition(key="app_code", match=MatchValue(value=scope["app_code"])),
        FieldCondition(key="is_current_version", match=MatchValue(value=True)),
        Filter(should=alternatives),
    ])
    hits = qdrant_client.query_points(
        collection_name=collection_name, query=vector, query_filter=access_filter,
        limit=k, with_payload=True, score_threshold=threshold).points
    result = []
    for hit in hits:
        payload = hit.payload or {}
        doc_id, version = payload.get("document_id"), payload.get("version_id")
        if type(doc_id) is not int or not isinstance(version, str):
            raise ValueError("Invalid stored source identity")
        # Checks current tenant/object ACL, published version and processed state.
        if authorize_current_chunk(security_context, doc_id, version) is not True:
            continue
        if not isinstance(payload.get("text"), str):
            raise ValueError("Invalid stored chunk")
        result.append({
            "text": payload["text"], "document_id": doc_id, "version_id": version,
            "document_type": payload.get("document_type"), "score": hit.score,
            "source_page": payload.get("source_page"),
        })
    return result

# Payload filtering reduces candidates; it is not the authority for current ACL.
# Role and owner IDs are server-resolved, not request claims. query_points is used
# with the installed client API; cosine thresholds are task-specific, not probabilities.
