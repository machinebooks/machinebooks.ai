# Didactic example: patterns/middleware/pipeline.py
# Middleware order — order matters

@app.before_request
def middleware_pipeline():
    # 1. Inject request-id for traceability (X-Request-ID)
    g.request_id = request.headers.get('X-Request-ID', str(uuid4()))

    # 2. Record start timestamp for timing
    g.request_start_time = time.time()

    # 3. Verify JWT and extract claims
    verify_jwt_in_request()

    # 4. platform_guard: verify permissions by AppRole
    check_platform_permissions(request.endpoint)

    # 5. Rate limiting by operation type (Redis)
    enforce_rate_limit(get_jwt_identity(), request.endpoint)

@app.after_request
def add_security_headers(response):
    # OWASP headers: strict CSP, HSTS with preload, X-Frame-Options
    response.headers['Content-Security-Policy'] = "default-src 'self'"
    response.headers['Strict-Transport-Security'] = 'max-age=31536000; includeSubDomains; preload'
    response.headers['X-Frame-Options'] = 'DENY'
    response.headers['X-Content-Type-Options'] = 'nosniff'

    # Timing logging
    duration_ms = (time.time() - g.request_start_time) * 1000
    logger.info("request_completed", request_id=g.request_id, duration_ms=duration_ms)
    return response
