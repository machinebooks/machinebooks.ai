ANALYTICS_SYSTEM_PROMPT = """
You are a specialized business analysis assistant.
You have access to up-to-date company data in the following areas:
- Business KPIs: ARR, MRR, revenue, pipeline
- F1-F5 funnel analysis with conversion rates
- Market scoring by sector and service line
- Client portfolio with ABC segmentation
- Profitability by client, service, and vertical
- Competitor profiles with SWOT analysis

IMPORTANT: Only answer questions about available business data.
Do not generate data that doesn't exist in the provided context.
When you don't have sufficient data, state it explicitly.
If the question requires more up-to-date data, suggest when it will be updated.
"""

class AnalyticsChatService:
    def __init__(self, client: anthropic.Anthropic):
        self.client = client

    def answer_query(self, query: str, user_id: int) -> str:
        """
        Answers an analytical question combining RAG context
        with pre-calculated data from analytics_db.
        """
        query = query[:2000].strip()  # Length limit to prevent abuse
        # In production, inform the user if the question was truncated
        # 1. Retrieve semantic context from Qdrant (relevant documents)
        rag_context = self._retrieve_rag_context(query)

        # 2. For questions requiring numerical data, execute
        #    predefined queries based on detected intent
        intent = self._classify_intent(query)
        structured_data = self._fetch_structured_data(intent, query)

        # 3. Build the complete context for Claude
        context_parts = []
        if rag_context:
            context_parts.append(f"## Relevant documents\n{rag_context}")
        if structured_data:
            context_parts.append(f"## Business data\n{structured_data}")

        context = "\n\n".join(context_parts)

        # 4. Invoke Claude with specific context
        response = self.client.messages.create(
            model="claude-sonnet-4-6",  # Sonnet for quality/cost balance
            max_tokens=1500,
            system=ANALYTICS_SYSTEM_PROMPT,
            messages=[
                {
                    "role": "user",
                    "content": f"{context}\n\n---\nQuestion: {query}"
                }
            ]
        )

        # 5. Log usage for governance and costs
        _log_llm_usage(
            service='analytics_chat',
            model='claude-sonnet-4-6',
            input_tokens=response.usage.input_tokens,
            output_tokens=response.usage.output_tokens,
            user_id=user_id
        )

        return response.content[0].text
