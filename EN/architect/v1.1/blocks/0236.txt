# .github/workflows/ci.yml
# Continuous integration workflow: lint, type-check, and tests

name: Continuous Integration

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

jobs:
  lint-backend:
    name: Backend static analysis
    runs-on: ubuntu-latest

    steps:
      - name: Clone repository
        uses: actions/checkout@v4

      - name: Set up Python 3.11
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: "pip"  # Cache pip dependencies between runs

      - name: Install analysis dependencies
        run: |
          pip install flake8 flake8-bugbear flake8-docstrings
          pip install bandit  # Static security analysis

      - name: Run flake8
        run: |
          # Exclude Alembic migrations (auto-generated)
          flake8 backend/ --exclude=database/migrations/ \
            --max-line-length=120 \
            --extend-ignore=E203,W503

      - name: Run security analysis
        run: |
          # bandit detects common security patterns: hardcoded secrets,
          # unsafe subprocess usage, dangerous deserialization
          bandit -r backend/ -ll --exclude=backend/tests/ -q

  test-backend:
    name: Backend tests
    runs-on: ubuntu-latest
    needs: lint-backend  # Only run if lint passes

    services:
      # Temporary MySQL for integration tests
      mysql:
        image: mysql:8.0
        env:
          MYSQL_ROOT_PASSWORD: test_password  # Ephemeral test credentials — in production use ${{ secrets.* }}
          MYSQL_DATABASE: test_db
        ports:
          - 3306:3306
        options: >-
          --health-cmd="mysqladmin ping"
          --health-interval=10s
          --health-timeout=5s
          --health-retries=5

      # Temporary Redis for Celery tests
      redis:
        image: redis:7-alpine
        ports:
          - 6379:6379

    steps:
      - name: Clone repository
        uses: actions/checkout@v4

      - name: Set up Python 3.11
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: "pip"

      - name: Install backend dependencies
        run: |
          cd backend
          pip install -r requirements.txt
          pip install pytest pytest-cov pytest-flask

      - name: Run migrations on test database
        env:
          DATABASE_URL: mysql+pymysql://root:test_password@localhost/test_db
        run: |
          cd backend
          flask db upgrade

      - name: Run tests with coverage
        env:
          FLASK_ENV: testing
          DATABASE_URL: mysql+pymysql://root:test_password@localhost/test_db
          REDIS_URL: redis://localhost:6379/0
          SECRET_KEY: test-secret-key-no-production
        run: |
          cd backend
          pytest tests/ -v --cov=. --cov-report=xml \
            --cov-fail-under=40  # Minimum coverage threshold

  lint-frontend:
    name: Frontend static analysis
    runs-on: ubuntu-latest

    steps:
      - name: Clone repository
        uses: actions/checkout@v4

      - name: Set up Node.js 20
        uses: actions/setup-node@v4
        with:
          node-version: "20"
          cache: "npm"
          cache-dependency-path: frontend/package-lock.json

      - name: Install dependencies
        run: |
          cd frontend
          npm ci  # ci is more reproducible than install in automated environments

      - name: Run ESLint
        run: |
          cd frontend
          npm run lint

      - name: Verify TypeScript compilation
        run: |
          cd frontend
          npm run type-check
