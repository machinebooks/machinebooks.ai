# Location block for the streaming endpoint
location /ai/chat/stream {
    proxy_pass http://ai_service:8001;

    # Critical for SSE: disable buffering
    proxy_buffering off;
    proxy_cache off;

    # Keep connection open during stream
    proxy_read_timeout 300s;
    proxy_send_timeout 300s;

    # Headers needed for SSE
    proxy_set_header Connection '';
    proxy_http_version 1.1;
    chunked_transfer_encoding on;
}
