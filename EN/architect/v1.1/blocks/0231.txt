# Didactic example: patterns/nginx/nginx.conf
# Upstream: the three React applications served from their Nginx containers
upstream frontend_ops {
    server nginx_ops:80;
}
upstream frontend_analytics {
    server nginx_analytics:80;
}
upstream frontend_admin {
    server nginx_admin:80;
}

# Upstream: the backend services
upstream backend_api {
    server backend:5000;
}
upstream ai_service_api {
    server ai_service:8001;
}

server {
    listen 443 ssl http2;
    server_name plataforma.ejemplo.com;

    ssl_certificate     /etc/nginx/certs/fullchain.pem;
    ssl_certificate_key /etc/nginx/certs/privkey.pem;
    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_ciphers         ECDHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES256-GCM-SHA384;

    # Upload limit for PDF documents and Excel files
    client_max_body_size 100M;
    client_body_timeout  120s;

    # Gzip compression for JSON responses and static assets
    gzip on;
    gzip_types application/json text/html application/javascript text/css;
    gzip_min_length 1024;

    # Main App at root
    location / {
        proxy_pass http://frontend_ops;
        proxy_set_header Host $host;
    }

    # Analytics App at /analytics
    location /analytics {
        proxy_pass http://frontend_analytics;
        rewrite ^/analytics(/.*)$ $1 break;
    }

    # Admin App at /admin
    location /admin {
        proxy_pass http://frontend_admin;
        rewrite ^/admin(/.*)$ $1 break;
    }

    # Backend API
    location /api/ {
        proxy_pass http://backend_api;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_read_timeout 120s;   # Generation endpoints can take time
    }

    # AI service SSE endpoints: Claude streaming
    # CRITICAL: disable buffering so streaming reaches the client
    location /api/ai/stream {
        proxy_pass http://ai_service_api;
        proxy_buffering off;              # No buffering for SSE
        proxy_cache off;
        proxy_set_header Connection '';   # HTTP/1.1 for keep-alive
        proxy_http_version 1.1;
        proxy_read_timeout 300s;          # Streaming can last several minutes
        chunked_transfer_encoding on;
    }
}
