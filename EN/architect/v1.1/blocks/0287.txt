# Didactic example: patterns/models/document.py
# Document model with versioning and AI enrichment

class Document(db.Model):
    __bind_key__ = 'operations'
    __tablename__ = 'documents'

    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(500), nullable=False)
    doc_type = db.Column(db.String(50))  # 12 types: requirements_document, technical_proposal,
                                          #           cv_profile, catalog, meeting_minutes...

    # Document versioning
    parent_version_id = db.Column(db.Integer, db.ForeignKey('documents.id'))
    is_current_version = db.Column(db.Boolean, default=True)
    version_number = db.Column(db.Integer, default=1)

    # AI processing status
    processing_status = db.Column(db.String(20), default='pending')
    # pending → processing → processed → failed

    # Automatic enrichment by Claude
    ai_summary = db.Column(db.Text)        # Summary generated by claude-sonnet-4-6
    ai_keywords = db.Column(db.JSON)       # Extracted keywords
    ai_entities = db.Column(db.JSON)       # Recognized entities (companies, technologies)

    # Integrity
    file_hash = db.Column(db.String(64))   # SHA-256 of the original content
    file_size_bytes = db.Column(db.BigInteger)
