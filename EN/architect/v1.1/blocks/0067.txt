# Bounded educational SSE core; no Flask g is accessed after request completion.
import json


def prepare_copilot_response(ctx, message, session_id, mode,
                             authorize_session, check_input, minimize_for_provider,
                             stream_authorized_response, check_output,
                             max_output_chars=100000):
    if ctx is None or not isinstance(message,str) or not message.strip():
        raise ValueError("Authenticated context and message required")
    if len(message) > 10000:
        raise ValueError("Message too long")
    if not isinstance(session_id,str) or not 1 <= len(session_id) <= 128:
        raise ValueError("Server-authorized session identifier required")
    if not isinstance(mode,str) or mode not in {"auto","chat_rag","agent_tools","orchestrate"}:
        raise ValueError("Invalid operation mode")
    if type(max_output_chars) is not int or max_output_chars < 1:
        raise ValueError("Positive output budget required")
    if authorize_session(ctx,session_id,mode) is not True:
        raise PermissionError("Session or mode not authorized")
    guarded = check_input(ctx,message)
    if guarded.action not in {"ALLOW","SANITIZE"}:
        raise PermissionError("Input rejected")
    safe_input = message if guarded.action == "ALLOW" and guarded.filtered_text is None else guarded.filtered_text
    if not isinstance(safe_input,str) or not safe_input.strip():
        raise ValueError("Valid filtered input required")
    safe_input = minimize_for_provider(ctx,safe_input)
    if not isinstance(safe_input,str) or not safe_input.strip():
        raise ValueError("Minimized input required")

    def events():
        fragments=[]
        size=0
        for fragment in stream_authorized_response(ctx,session_id,mode,safe_input):
            if not isinstance(fragment,str):
                raise TypeError("Unexpected model fragment")
            size+=len(fragment)
            if size > max_output_chars:
                raise ValueError("Output budget exceeded")
            fragments.append(fragment)
        output=check_output(ctx,"".join(fragments))
        if output.action not in {"ALLOW","SANITIZE"}:
            raise PermissionError("Output rejected")
        if not isinstance(output.filtered_text,str) or len(output.filtered_text) > max_output_chars:
            raise ValueError("Valid filtered output required")
        payload=json.dumps({"type":"response","content":output.filtered_text})
        yield f"data: {payload}\n\n"
        yield 'data: {"type":"done"}\n\n'
    return events()
# HTTP adapters catch preflight errors and authorize before returning Response(events).
# Full response buffering enables whole-output filters; token-by-token delivery is absent.
# Disconnects, time budgets, external effects and audit delivery require additional tests.
