from datetime import datetime, timezone
from sqlalchemy import Index
from extensions import db

class AuditLog(db.Model):
    """Audit log for business and security actions.

    Separate from technical logs — serves for compliance, security
    audits, and incident response, not for debugging.
    """
    __tablename__ = 'audit_logs'
    __bind_key__ = 'platform_core'

    id = db.Column(db.Integer, primary_key=True)

    # Who — can be null if action occurs before authentication
    user_id = db.Column(db.Integer, db.ForeignKey('users.id'), nullable=True)
    session_id = db.Column(db.String(100), nullable=True)

    # What
    action = db.Column(db.String(100), nullable=False)  # Predefined constant
    resource_type = db.Column(db.String(50), nullable=True)  # 'document', 'proposal', 'user'
    resource_id = db.Column(db.String(100), nullable=True)

    # Context
    details = db.Column(db.JSON, nullable=True)  # Additional action metadata
    ip_address = db.Column(db.String(45), nullable=True)  # IPv4 and IPv6
    user_agent = db.Column(db.String(500), nullable=True)

    # Classification
    severity = db.Column(db.String(20), default='INFO')  # INFO, WARNING, CRITICAL

    # When
    created_at = db.Column(db.DateTime, default=lambda: datetime.now(timezone.utc), nullable=False)

    # Composite indexes for the most frequent queries in the audit panel
    __table_args__ = (
        Index('idx_audit_user_action', 'user_id', 'action'),
        Index('idx_audit_timestamp_severity', 'created_at', 'severity'),
        Index('idx_audit_resource', 'resource_type', 'resource_id'),
    )
