# Didactic example: patterns/chat/history_manager.py

def estimate_tokens(text: str) -> int:
    # Claude uses its own tokenizer; tiktoken (OpenAI) would give incorrect counts.
    # Practical approximation: 1 token ~ 4 characters in Spanish.
    return len(text) // 4

def manage_history(messages: list[dict], budget: int) -> list[dict]:
    """Manages conversation history within the budget."""
    total_tokens = sum(estimate_tokens(m["content"]) for m in messages)

    if total_tokens <= budget:
        return messages  # Fits completely, do not truncate

    # Preserve first message + last N messages
    first_message = messages[0]
    recent_window = 6  # Last 3 turns (user + assistant)

    if len(messages) <= recent_window + 1:
        return messages  # Too few messages, do not compress

    middle_messages = messages[1:-recent_window]

    # Generate summary of intermediate messages with Haiku
    summary = compress_to_summary(middle_messages)

    return [
        first_message,
        {"role": "user", "content": f"[Summary of previous conversation: {summary}]"},
        {"role": "assistant", "content": "Understood, I continue with that context."},
        *messages[-recent_window:]
    ]
