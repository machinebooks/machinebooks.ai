class HumanBaselineConfig(db.Model):
    """Human baseline configuration per task type.

    These values are NOT estimates — they come from real measurements
    of the manual process prior to Platform implementation.
    They can be updated from the Admin panel when conditions change.
    """
    __tablename__ = 'human_baseline_configs'
    __bind_key__ = 'platform_core'

    id = db.Column(db.Integer, primary_key=True)

    task_type = db.Column(db.String(100), unique=True, nullable=False)
    # 'document_analysis', 'proposal_generation', 'cv_analysis',
    # 'business_intelligence_report', 'opportunity_search'

    # Baseline time in minutes (measured from the actual previous process)
    human_baseline_minutes = db.Column(db.Float, nullable=False)
    # document_analysis: 480 min (8 hours senior analyst)
    # proposal_generation: 2400 min (40 hours, multiple profiles)
    # cv_analysis: 45 min (45 minutes per profile)

    # Hourly cost of the human profile who would perform this task
    human_hourly_cost_eur = db.Column(db.Float, nullable=False)
    # Includes total employee cost: salary + social charges

    description = db.Column(db.Text, nullable=True)
    is_active = db.Column(db.Boolean, default=True)
    updated_at = db.Column(db.DateTime, onupdate=lambda: datetime.now(timezone.utc))


class TaskCompletionLog(db.Model):
    """Log of tasks completed with AI with ROI calculation.

    Every time a user completes a task with AI, it's recorded
    here with the savings calculation. The Admin dashboard aggregates
    these records to show accumulated ROI.
    """
    __tablename__ = 'task_completion_logs'
    __bind_key__ = 'platform_core'

    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, db.ForeignKey('users.id'), nullable=False)

    task_type = db.Column(db.String(100), nullable=False)
    baseline_id = db.Column(
        db.Integer, db.ForeignKey('human_baseline_configs.id'), nullable=True
    )

    # Actual AI task time
    ai_duration_seconds = db.Column(db.Float, nullable=False)

    # Derived calculations (calculated at registration time)
    time_saved_minutes = db.Column(db.Float, nullable=True)
    money_saved_eur = db.Column(db.Float, nullable=True)

    # AI model used (for per-provider analysis)
    ai_model = db.Column(db.String(100), nullable=True)
    ai_provider = db.Column(db.String(50), nullable=True)

    # Additional context
    context = db.Column(db.JSON, nullable=True)
    completed_at = db.Column(db.DateTime, default=lambda: datetime.now(timezone.utc))

    __table_args__ = (
        Index('idx_task_completion_type_date', 'task_type', 'completed_at'),
        Index('idx_task_completion_user', 'user_id', 'completed_at'),
    )
