class LLMUsageLog(db.Model):
    """Schema for recorded LLM invocations.
    Only actual recorded invocations are represented; missing usage is not zero."""
    __tablename__ = 'llm_usage_logs'
    __bind_key__ = 'platform_core'

    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, nullable=True)
    # Nullable: some calls are from system tasks, without a user
    service_type = db.Column(db.String(50), nullable=False)
    provider = db.Column(db.String(30), nullable=False)
    model_name = db.Column(db.String(100), nullable=False)

    # Token counters according to the API response
    prompt_tokens = db.Column(db.Integer, nullable=True)
    completion_tokens = db.Column(db.Integer, nullable=True)
    total_tokens = db.Column(db.Integer, nullable=True)

    # Cost calculated at the time of the call using LLMModelPricing
    estimated_cost_eur = db.Column(db.Numeric(18, 8), nullable=True)
    latency_ms = db.Column(db.Integer, nullable=True)
    status = db.Column(db.String(20), default='success')
    # Values: success, error, timeout, rate_limited, fallback_used

    # Traceability without storing the full prompt
    prompt_hash = db.Column(db.String(64))
    # SHA-256 of the prompt: allows correlation without storing sensitive content

    created_at = db.Column(db.DateTime, default=lambda: datetime.now(timezone.utc), nullable=False, index=True)

    __table_args__ = (
        db.Index('idx_usage_user_date', 'user_id', 'created_at'),
        db.Index('idx_usage_service', 'service_type', 'created_at'),
        db.Index('idx_usage_provider_model', 'provider', 'model_name', 'created_at'),
    )
