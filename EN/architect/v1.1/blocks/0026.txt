# Historical-version Compose excerpt: 14 services, not a private application release.
# Image versions describe the case; assess supported versions before deployment.
services:
  # ======================================================
  # DATA LAYER — Network membership is not data authorization
  # ======================================================
  mysql:
    image: mysql:8.0
    environment:
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD}
      MYSQL_DATABASE: ${MYSQL_DATABASE}
    volumes:
      - mysql_data:/var/lib/mysql
      - ./database/init:/docker-entrypoint-initdb.d  # 3 schemas at initialization
    mem_limit: 4g
    networks: [data_net]

  redis:
    image: redis:7-alpine
    command: redis-server --requirepass ${REDIS_PASSWORD} --appendonly yes
    volumes:
      - redis_data:/data
    mem_limit: 512m
    networks: [data_net, broker_net]

  qdrant:
    image: qdrant/qdrant:v1.7.4
    volumes:
      - qdrant_data:/qdrant/storage
    mem_limit: 8g
    networks: [data_net]

  meilisearch:
    image: getmeili/meilisearch:v1.6
    environment:
      MEILI_MASTER_KEY: ${MEILI_MASTER_KEY}
    volumes:
      - meili_data:/meili_data
    mem_limit: 2g
    networks: [data_net]

  # ======================================================
  # APPLICATION LAYER — Accessible from backend_net
  # ======================================================
  backend:
    build: ./backend
    environment:
      DATABASE_URL: ${DATABASE_URL:?Provide a valid URL with encoded credentials}
      REDIS_URL: ${REDIS_URL:?Provide a valid broker URL with encoded credentials}
      AI_SERVICE_URL: http://ai_service:8000
      AI_SERVICE_API_KEY: ${AI_SERVICE_API_KEY}
    depends_on: [mysql, redis]
    mem_limit: 2g
    networks: [backend_net, data_net]

  ai_service:
    build: ./ai_service
    environment:
      ANTHROPIC_API_KEY: ${ANTHROPIC_API_KEY}
      QDRANT_URL: http://qdrant:6333
    depends_on: [qdrant, redis, backend]
    mem_limit: 4g
    networks: [backend_net, data_net]

  # ======================================================
  # WORKER LAYER — Same build as backend, different queues
  # ======================================================
  celery_default:
    build: ./backend
    command: celery -A celery_app worker -Q default,priority,sync,crm -c 4
    depends_on: [mysql, redis]
    networks: [backend_net, data_net]

  celery_ai:
    build: ./backend
    command: celery -A celery_app worker -Q ai,documents -c 2
    depends_on: [mysql, redis, ai_service]
    networks: [backend_net, data_net]

  celery_automation:
    build: ./backend
    command: celery -A celery_app worker -Q automation -c 1
    depends_on: [redis]
    networks: [backend_net, rpa_net, broker_net]

  celery_beat:
    build: ./backend
    command: celery -A celery_app beat --loglevel=info
    networks: [backend_net, data_net]

  # ======================================================
  # FRONTEND LAYER — Independent static builds
  # ======================================================
  frontend:
    build: ./frontend
    networks: [frontend_net]

  analytics_frontend:
    build: ./analytics/frontend
    networks: [frontend_net]

  admin_frontend:
    build: ./admin/frontend
    networks: [frontend_net]

  # ======================================================
  # REVERSE PROXY — Single entry point
  # ======================================================
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    depends_on: [backend, ai_service, frontend]
    networks: [frontend_net, backend_net]

# ======================================================
# NETWORKS — Review all reachable paths and service authorization
# ======================================================
networks:
  frontend_net:    # Nginx and frontends; only published ports expose entry points
    driver: bridge
  backend_net:     # Members can communicate with other members
    driver: bridge
  data_net:        # Data services and explicitly connected application services
    driver: bridge
  rpa_net:         # Selenium Grid — isolated from data
    driver: bridge
  broker_net:      # Redis and the automation worker; no MySQL membership
    driver: bridge

volumes:
  mysql_data:
  redis_data:
  qdrant_data:
  meili_data:
