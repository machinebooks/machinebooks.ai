# Partial CRM adapter: update_tokens accepts plaintext and encrypts exactly once.
# Current tenant/user authorization and serialized refresh with reread are required
# in the repository adapter; this class alone does not implement those contracts.
# Didactic example: CRM client with OAuth2 and automatic refresh
# Pattern: backend/services/crm/crm_client.py
# Credentials loaded from environment variables via current_app.config

import requests
from datetime import datetime, timedelta, timezone
from models.crm import CRMCredentials
from utils.crypto import decrypt_value

class CRMClient:
    """HTTP client for the corporate CRM with automatic OAuth2 token management."""

    def __init__(self, user_id: int):
        self.user_id = user_id
        self._credentials = None

    def _get_valid_token(self) -> str:
        """Returns a valid token, refreshing if necessary."""
        creds = CRMCredentials.query.filter_by(user_id=self.user_id).first()
        if not creds:
            raise PermissionError("User without configured CRM credentials")

        # Refresh if the token expires in less than 5 minutes
        # Historical DB timestamps are naive UTC; aware values convert to UTC.
        expiry = creds.token_expires_at
        expiry = expiry.replace(tzinfo=timezone.utc) if expiry.tzinfo is None else expiry.astimezone(timezone.utc)
        if expiry < datetime.now(timezone.utc) + timedelta(minutes=5):
            self._refresh_token(creds)

        return decrypt_value(creds.access_token_encrypted)

    def _refresh_token(self, creds: CRMCredentials):
        """Requests a new token using the stored refresh_token."""
        # CRM_BASE_URL, CRM_CLIENT_ID, and CRM_CLIENT_SECRET: loaded
        # from current_app.config or environment variables
        response = requests.post(
            f"{CRM_BASE_URL}/oauth/token",
            data={
                "grant_type": "refresh_token",
                "refresh_token": decrypt_value(creds.refresh_token_encrypted),
                "client_id": CRM_CLIENT_ID,
                "client_secret": CRM_CLIENT_SECRET,
            },
            timeout=10
        )
        response.raise_for_status()
        token_data = response.json()

        # Update encrypted credentials in database
        creds.update_tokens(
            access_token=token_data["access_token"],
            refresh_token=token_data.get("refresh_token", decrypt_value(creds.refresh_token_encrypted)),
            expires_in=token_data["expires_in"]
        )

    def get_cases(self, modified_since: datetime = None) -> list:
        """Retrieves cases from the CRM, optionally filtered by modification date."""
        token = self._get_valid_token()
        params = {}
        if modified_since:
            params["modified_since"] = modified_since.isoformat()

        response = requests.get(
            f"{CRM_BASE_URL}/api/cases",
            headers={"Authorization": f"Bearer {token}"},
            params=params,
            timeout=30
        )
        response.raise_for_status()
        return response.json().get("cases", [])
