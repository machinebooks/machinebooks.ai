# Partial Celery adapter: server repositories resolve current job/credential scope.
from celery import shared_task
from celery.utils.log import get_task_logger
logger=get_task_logger(__name__)

@shared_task(name="tasks.automation.run_portal_sync",bind=True,base=AutomationTask,
             queue="automation",max_retries=3,default_retry_delay=60,
             acks_late=True,soft_time_limit=1800,time_limit=2100)
def run_portal_sync(self,task_config):
    task_id=self.request.id
    context=resolve_authorized_sync_job(task_id,task_config)
    if not isinstance(context,dict) or type(context.get("user_id")) is not int or context["user_id"]<1:
        raise PermissionError("Current job scope required")
    if operation_deadline_open(task_id) is not True:raise TimeoutError("Job deadline exceeded")
    # Vault adapter checks tenant/user/credential purpose before decrypting.
    credentials=load_authorized_credentials(context)
    challenge_store=load_authorized_challenge_store(context)

    try:
        with PortalBot(task_id,credentials,challenge_store,context["user_id"],context) as bot:
            if not bot.login():raise PermissionError("Portal login failed")
            projects=bot.extract_project_list()
            if persist_sync_result(context,task_id,projects) is not True:
                raise RuntimeError("Durable sync acknowledgement required")
            return {"status":"completed","projects_synced":len(projects)}

    except RetryablePortalError as exc:
        # Adapter classifies retryable reads and reconciles ambiguous write effects.
        if self.request.retries>=self.max_retries:
            record_sync_failure(context,task_id,type(exc).__name__)
            raise RuntimeError("Portal sync retry limit exceeded") from None
        logger.warning("Portal sync retry",extra={"task_id":task_id,"error_type":type(exc).__name__})
        raise self.retry(exc=RuntimeError("Retryable portal read failure"),countdown=60) from None
    except Exception as exc:
        record_sync_failure(context,task_id,type(exc).__name__)
        logger.error("Portal sync failed",extra={"task_id":task_id,"error_type":type(exc).__name__})
        raise RuntimeError("Portal sync failed") from None
# RetryablePortalError and all repositories are application-owned dependencies.
# Three retries is this task's policy; Chapter 8's 180 belongs to another task.
# Failure recording is idempotent, uses protected job references and omits raw errors.
