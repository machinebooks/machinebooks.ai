# Add in the server block, alongside the location configurations
# Security headers for all frontends

# Prevents the browser from inferring the content's MIME type
add_header X-Content-Type-Options "nosniff" always;

# Basic protection against clickjacking
add_header X-Frame-Options "SAMEORIGIN" always;

# Force HTTPS on future requests (only if TLS is enabled)
# add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

# Referrer policy: only send the origin, not the full URL
add_header Referrer-Policy "strict-origin-when-cross-origin" always;

# Content Security Policy: content source restriction
# This CSP is tailored to the Platform's specific content
add_header Content-Security-Policy "
  default-src 'self';
  script-src 'self';
  style-src 'self' 'unsafe-inline';
  img-src 'self' data: blob:;
  connect-src 'self' https://api.example.com;
  font-src 'self';
  object-src 'none';
  frame-ancestors 'self'
" always;
