def _seed_rbac_fixtures():
    """Creates the minimum RBAC structure needed for tests."""
    # Operations app with its 6 roles
    ops_app = App(name="Operations", code="operations")
    db.session.add(ops_app)
    db.session.flush()

    roles_permissions = {
        "admin":    {"clients": ["read","write","delete"], "exports": ["read","write"]},
        "manager":  {"clients": ["read","write"], "exports": ["read","write"]},
        "analyst":  {"clients": ["read"], "exports": ["read"]},
        "viewer":   {"clients": ["read"], "exports": []},
        "editor":   {"clients": ["read","write"], "exports": []},
        "readonly": {"clients": ["read"], "exports": []},
    }

    for role_name, permissions in roles_permissions.items():
        role = AppRole(
            app_id=ops_app.id,
            name=role_name,
            permissions=permissions,
            is_admin=(role_name == "admin")
        )
        db.session.add(role)
        user = User(
            email=f"test_{role_name}@example.com",
            name=f"Test {role_name.capitalize()}"
        )
        user.set_password("Test1234!")
        db.session.add(user)
        # Associate user with role and app (UserAppMembership) — omitted for brevity

    db.session.commit()
