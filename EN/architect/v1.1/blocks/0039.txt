# Illustrative three-table migration; DDL locking and atomicity depend on the engine.
from alembic import op
import sqlalchemy as sa
from datetime import datetime, timezone

TABLES = ("opportunities", "cv_profiles", "alerts")


def upgrade():
    # A migration timestamp is not the original deletion timestamp.
    changed_at = datetime.now(timezone.utc).replace(tzinfo=None)
    for name in TABLES:
        op.add_column(name, sa.Column("deleted_at", sa.DateTime, nullable=True))
        table = sa.table(name, sa.column("is_deleted", sa.Boolean),
                         sa.column("deleted_at", sa.DateTime))
        op.get_bind().execute(
            table.update().where(table.c.is_deleted.is_(True))
            .values(deleted_at=changed_at)
        )
        # Verify dependencies/indexes and recovery before dropping old data.
        op.drop_column(name, "is_deleted")


def downgrade():
    for name in TABLES:
        op.add_column(name, sa.Column("is_deleted", sa.Boolean,
                                     nullable=False, server_default=sa.false()))
        table = sa.table(name, sa.column("is_deleted", sa.Boolean),
                         sa.column("deleted_at", sa.DateTime))
        op.get_bind().execute(
            table.update().values(is_deleted=table.c.deleted_at.is_not(None))
        )
        # Boolean state is restored; deletion timestamps are irreversibly discarded.
        op.drop_column(name, "deleted_at")
