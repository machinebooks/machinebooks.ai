# A leased lock coordinates overlap; expiry does not stop an old owner.
import logging
import secrets
from redis.exceptions import RedisError
from contextlib import contextmanager

RELEASE_LOCK = """
if redis.call('GET',KEYS[1]) == ARGV[1] then
    return redis.call('DEL',KEYS[1])
end
return 0
"""


class TaskAlreadyRunningError(RuntimeError):
    pass


@contextmanager
def task_lock(client,lock_id,expire=3600):
    if not isinstance(lock_id,str) or not lock_id or len(lock_id) > 128:
        raise ValueError("Bounded lock identifier required")
    if type(expire) is not int or expire < 1:
        raise ValueError("Positive lease duration required")
    key=f"celery_lock:{lock_id}"
    token=secrets.token_hex(16)
    if client.set(key,token,nx=True,ex=expire) is not True:
        raise TaskAlreadyRunningError("Task lease already held")
    primary_error = None
    try:
        yield token
    except BaseException as exc:
        primary_error = exc
        raise
    finally:
        # Atomic comparison prevents an expired owner deleting a successor's lease.
        try:
            client.eval(RELEASE_LOCK,1,key,token)
        except RedisError:
            if primary_error is None:
                raise
            logging.getLogger(__name__).error("Lease cleanup unconfirmed")
# Renewal, fencing, durable idempotency and external-effect reconciliation are separate.
