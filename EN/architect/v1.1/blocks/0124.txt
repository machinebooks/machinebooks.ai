import hashlib
import math
import time
import pymupdf as fitz
from PIL import Image
from langchain_core.documents import Document as LangChainDocument
from langchain_text_splitters import RecursiveCharacterTextSplitter

class DocumentProcessor:
    MAX_FILE_BYTES = 20_000_000
    MAX_PAGES = 500
    MAX_TEXT_CHARS = 1_000_000
    MAX_PIXELS = 8_000_000

    def __init__(self, load_authorized_blob, ocr_page, office_loaders,
                 needs_ocr=None):
        self._load = load_authorized_blob
        self._ocr = ocr_page
        self._office = dict(office_loaders)
        self._needs_ocr = needs_ocr or (
            lambda page, text: not text.strip()
            or (len(text.strip()) < 50 and bool(page.get_images())))
        # Units are CHARACTERS here. Token budgeting is a separate adapter step.
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=800, chunk_overlap=150,
            separators=["\n\n", "\n", ". ", " ", ""])

    def compute_hash(self, content):
        if not isinstance(content, bytes) or len(content) > self.MAX_FILE_BYTES:
            raise ValueError("Invalid bounded file content")
        return hashlib.sha256(content).hexdigest()

    def load_document(self, document_id, security_context):
        if type(document_id) is not int or document_id <= 0:
            raise ValueError("Invalid document identifier")
        # Callback authorizes current actor/document/version and reads a bounded,
        # immutable upload; MIME is server-verified, not trusted from a request.
        record = self._load(security_context, document_id, self.MAX_FILE_BYTES)
        if (not isinstance(record, dict) or not isinstance(record.get("content"), bytes)
                or not isinstance(record.get("version_id"), str) or not record["version_id"]
                or not isinstance(record.get("tenant_id"), str) or not record["tenant_id"]):
            raise ValueError("Invalid authorized document snapshot")
        content = record["content"]
        if not content or len(content) > self.MAX_FILE_BYTES:
            raise ValueError("Document exceeds the example size limit")
        base = {"document_id": document_id, "version_id": record["version_id"],
                "tenant_id": record["tenant_id"], "file_hash": self.compute_hash(content)}
        mime = record.get("mime_type")
        if mime == "application/pdf":
            if not content.startswith(b"%PDF-"):
                raise ValueError("PDF signature mismatch")
            return self._load_pdf(content, base)
        loader = self._office.get(mime)
        if not callable(loader):
            raise ValueError("No authorized parser for this MIME type")
        documents = loader(content, max_characters=self.MAX_TEXT_CHARS, timeout=60)
        if not isinstance(documents, list) or len(documents) > 5000:
            raise ValueError("Invalid parser output")
        total = 0
        for document in documents:
            if not isinstance(document, LangChainDocument):
                raise TypeError("Parser must return LangChain documents")
            total += len(document.page_content)
            if total > self.MAX_TEXT_CHARS:
                raise ValueError("Extracted text exceeds the example limit")
            document.metadata = {**document.metadata, **base}
        return documents

    def _load_pdf(self, content, metadata):
            result = []
            total = 0
            deadline = time.monotonic() + 120
            with fitz.open(stream=content, filetype="pdf") as pdf:
                if pdf.needs_pass or not 1 <= len(pdf) <= self.MAX_PAGES:
                    raise ValueError("Unsupported protected or oversized PDF")
                for index, page in enumerate(pdf):
                    if time.monotonic() >= deadline:
                        raise TimeoutError("PDF extraction deadline reached")
                    text = page.get_text()
                    use_ocr = self._needs_ocr(page, text)
                    if type(use_ocr) is not bool:
                        raise TypeError("OCR decision must be boolean")
                    if use_ocr:
                        area = page.rect.width * page.rect.height
                        if not math.isfinite(area) or area <= 0:
                            raise ValueError("Invalid PDF page geometry")
                        zoom = min(300 / 72, math.sqrt((self.MAX_PIXELS - 10_000) / area))
                        pixmap = page.get_pixmap(matrix=fitz.Matrix(zoom, zoom),
                                                colorspace=fitz.csRGB, alpha=False)
                        if pixmap.width * pixmap.height > self.MAX_PIXELS:
                            raise ValueError("OCR raster exceeds the example limit")
                        with Image.frombytes("RGB", (pixmap.width, pixmap.height),
                                             pixmap.samples) as image:
                            ocr_text = self._ocr(image, lang="spa+eng", timeout=30)
                        if not isinstance(ocr_text, str):
                            raise TypeError("OCR adapter must return text")
                        # Retain selectable text in mixed pages; OCR may duplicate it.
                        text = "\n".join(p for p in (text.strip(), ocr_text.strip()) if p)
                    if time.monotonic() >= deadline:
                        raise TimeoutError("PDF extraction deadline reached")
                    total += len(text)
                    if total > self.MAX_TEXT_CHARS:
                        raise ValueError("Extracted text exceeds the example limit")
                    if text.strip():
                        result.append(LangChainDocument(
                            page_content=text, metadata={**metadata, "page": index + 1,
                                                         "ocr": use_ocr}))
            return result

    # load_authorized_blob, Office parsers and OCR are required adapters.
    # A pytesseract.image_to_string adapter can enforce timeout=30 per page.
    # Signature/length checks are not malware screening or a parser sandbox.
    # Native rendering needs an isolated runner with CPU/memory limits; the cooperative
    # deadline cannot interrupt a native call. The page heuristic cannot prove completeness.
