import anthropic
import sqlparse
from typing import Optional

ALLOWED_TABLES = {
    "opportunities", "proposals", "clients",
    "pipeline_stages", "revenue_entries"
}

ANALYTICS_SCHEMA_CONTEXT = """
Available tables (read-only):
- opportunities(id, title, value_eur, stage, created_at, closed_at, client_id)
- proposals(id, opportunity_id, type, status, created_at, submitted_at)
- clients(id, name, sector, tier, account_manager)
- pipeline_stages(id, name, order_num, conversion_rate_avg)
- revenue_entries(id, client_id, amount_eur, period, type)
"""

async def generate_analytics_query(
    user_question: str,
    client: anthropic.Anthropic
) -> Optional[str]:
    """
    Generates safe SQL from a natural language question.
    Includes validation before returning the query.
    """
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=500,
        system=f"""You are a SQL generator for analytics.
        {ANALYTICS_SCHEMA_CONTEXT}

        Rules:
        - Generate ONLY SELECT. Never INSERT, UPDATE, DELETE, DROP.
        - Use only tables from the list.
        - Add LIMIT 1000 if there is no explicit LIMIT.
        - Return ONLY the SQL, without explanations.""",
        messages=[{"role": "user", "content": user_question}]
    )

    sql = response.content[0].text.strip()

    # Validate that SQL is a SELECT and only accesses allowed tables
    if not validate_analytics_sql(sql):
        return None

    return sql

def validate_analytics_sql(sql: str) -> bool:
    """
    Basic security validation: only SELECT, only allowed tables.
    Not a substitute for a read-only DB layer.
    """
    parsed = sqlparse.parse(sql)
    if not parsed:
        return False

    stmt = parsed[0]
    stmt_type = stmt.get_type()

    # Only SELECT allowed
    if stmt_type != 'SELECT':
        return False

    # Verify no tables outside the whitelist
    # (simplified implementation, production uses a more robust parser)
    sql_lower = sql.lower()
    for token in stmt.flatten():
        if token.ttype is sqlparse.tokens.Name:
            if token.value.lower() not in ALLOWED_TABLES and \
               token.value.lower() not in {'id', 'and', 'or', 'not', 'in', 'is', 'null'}:
                # Heuristic check: if it looks like a disallowed table name
                if any(keyword in sql_lower for keyword in ['from', 'join']):
                    return False

    return True
