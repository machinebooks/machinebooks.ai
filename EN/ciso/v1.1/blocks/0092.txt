# Historical route name retained; this endpoint supplies internal checklist progress only.
# Dependencies must authenticate, authorize role/licence and provide trusted corporate_id.
# Didactic example: routes/ai_governance.py
# Conformity assessment endpoint with weighted scoring

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session

router = APIRouter(prefix="/api/v1/ai-governance", tags=["AI Governance"])


@router.get("/records/{record_id}/conformity-score")
def get_conformity_score(
    record_id: int,
    db: Session = Depends(get_db),
    current_user = Depends(get_current_user)
):
    # Seven selected checkpoints provide internal progress, not legal conformity.
    record = db.query(AIGovernanceRecord).filter_by(
        id=record_id,
        corporate_id=current_user.corporate_id,
        is_deleted=False
    ).first()

    if not record:
        raise HTTPException(status_code=404, detail="AI record not found")

    if record.risk_level != AIRiskLevel.HIGH:
        return {
            "record_id": record_id,
            "risk_level": record.risk_level.value,
            "message": "This internal seven-control checklist is not used for this level. Applicable obligations require separate review.",
            "applicable": False
        }

    # Get all assessments for the record
    assessments = db.query(ConformityAssessment).filter_by(
        ai_record_id=record_id, corporate_id=current_user.corporate_id, is_deleted=False
    ).order_by(ConformityAssessment.evaluation_date.is_(None),
               ConformityAssessment.evaluation_date.desc(), ConformityAssessment.id.desc()).all()

    # Build map of checkpoint -> most recent assessment
    checkpoint_map = {}
    for assessment in assessments:
        checkpoint_map.setdefault(assessment.checkpoint, assessment)

    total_score = 0.0
    checkpoint_details = []
    blocking_issues = []

    for checkpoint in ConformityCheckpoint:
        weight = CHECKPOINT_WEIGHTS[checkpoint]
        assessment = checkpoint_map.get(checkpoint)

        if not assessment or assessment.status == CheckpointStatus.NOT_STARTED:
            score = 0.0
            status = "not_started"
        elif assessment.status == CheckpointStatus.PASSED:
            score = 1.0
            status = "passed"
        elif assessment.status == CheckpointStatus.CONDITIONAL:
            score = 0.7  # Meets with conditions: 70% of weight
            status = "conditional"
        elif assessment.status == CheckpointStatus.IN_PROGRESS:
            score = 0.3  # In evaluation: recognizes the effort
            status = "in_progress"
        elif assessment.status == CheckpointStatus.FAILED:
            score = 0.0
            status = "failed"
            blocking_issues.append({
                "checkpoint": checkpoint.value,
                "article": _get_article_reference(checkpoint),
                "findings": assessment.findings
            })

        else:
            raise ValueError("Unknown checkpoint status")

        weighted_score = score * weight
        total_score += weighted_score
        checkpoint_details.append({
            "checkpoint": checkpoint.value,
            "article": _get_article_reference(checkpoint),
            "status": status,
            "weight": weight,
            "score": round(score * 100),
            "weighted_score": round(weighted_score * 100, 1),
            "evaluation_date": assessment.evaluation_date.isoformat() if assessment and assessment.evaluation_date else None,
            "evaluator_id": assessment.evaluator_id if assessment else None,
            "iso42001_controls": assessment.iso42001_controls if assessment else None
        })

    conformity_score = round(total_score * 100, 1)

    # Determine overall status
    if blocking_issues:
        overall_status = "issues_identified"
    elif conformity_score >= 90:
        overall_status = "review_complete"
    elif conformity_score >= 70:
        overall_status = "review_partial"
    else:
        overall_status = "review_pending"

    return {
        "record_id": record_id,
        "system_name": record.name,
        "risk_level": record.risk_level.value,
        "conformity_score": conformity_score,
        "assessment_scope": "Internal checklist; not legal conformity or deployment approval",
        "overall_status": overall_status,
        "checkpoints": checkpoint_details,
        "blocking_issues": blocking_issues,
        "recommendations": _generate_recommendations(checkpoint_details, blocking_issues),
        "last_full_assessment": max((a.evaluation_date for a in assessments if a.evaluation_date), default=None),
    }

def _get_article_reference(checkpoint: ConformityCheckpoint) -> str:
    mapping = {
        ConformityCheckpoint.RISK_MANAGEMENT: "Art. 9",
        ConformityCheckpoint.DATA_GOVERNANCE: "Art. 10",
        ConformityCheckpoint.TECHNICAL_DOCS: "Art. 11",
        ConformityCheckpoint.RECORD_KEEPING: "Art. 12",
        ConformityCheckpoint.TRANSPARENCY: "Art. 13",
        ConformityCheckpoint.HUMAN_OVERSIGHT: "Art. 14",
        ConformityCheckpoint.ACCURACY_ROBUSTNESS: "Art. 15",
    }
    return mapping.get(checkpoint, "N/A")


def _generate_recommendations(details: list, blocking: list) -> list[str]:
    recommendations = []
    for issue in blocking:
        recommendations.append(f"Review blocking issue: {issue['checkpoint']} ({issue['article']}).")
    for detail in details:
        if detail["status"] == "not_started":
            recommendations.append(f"Start review: {detail['checkpoint']} ({detail['article']}).")
        elif detail["status"] == "conditional":
            recommendations.append(f"Review conditions: {detail['checkpoint']} ({detail['article']}).")
    return recommendations
