# Didactic model based on Art. 30.1 GDPR
# Didactic example: patterns/privacy/data_processing_activity.py

from sqlalchemy import Column, String, Text, JSON, Boolean, DateTime, UniqueConstraint, Enum as SAEnum
from sqlalchemy.orm import relationship
import enum

class LegalBasis(enum.Enum):
    """Art. 6.1 GDPR — Legal bases for processing."""
    CONSENT = "consent"                          # Art. 6.1.a
    CONTRACT = "contract"             # Art. 6.1.b
    LEGAL_OBLIGATION = "legal_obligation"         # Art. 6.1.c
    VITAL_INTERESTS = "vital_interest"           # Art. 6.1.d
    PUBLIC_INTEREST = "public_interest"            # Art. 6.1.e
    LEGITIMATE_INTEREST = "legitimate_interest"    # Art. 6.1.f

class DataProcessingActivity(BaseModel):
    """Didactic controller ROPA draft, not a complete activation validator.

    BaseModel supplies integer id, uuid, corporate_id and row_version.
    Tenant context requires authorization and scoped queries at every boundary.
    Validate required Art. 30.1 information before activating a draft.
    """
    __tablename__ = "data_processing_activities"
    __table_args__ = (UniqueConstraint("corporate_id", "code"),)
    code = Column(String(64))
    joint_controller_contacts = Column(JSON)
    representative_name = Column(String(255))
    representative_contact = Column(String(255))
    # Applicable joint controllers, representative and DPO need contact details.
    dpia_review_status = Column(String(32), default="pending_review", nullable=False)

    # Operational name; the following contacts support Art. 30.1.a.
    name = Column(String(255), nullable=False,
                  comment="Name of the processing activity")
    controller_name = Column(String(255), nullable=False,
                             comment="Name of the data controller")
    controller_contact = Column(String(255),
                                comment="Controller contact details")
    joint_controller = Column(String(255),
                              comment="Joint controller, if applicable")
    dpo_contact = Column(String(255),
                         comment="DPO contact details")

    # --- Art. 30.1.b — Purposes of processing ---
    purposes = Column(JSON, nullable=False,
                      comment="Array of processing purposes")
    legal_basis = Column(SAEnum(LegalBasis), nullable=False,
                         comment="Legal basis per Art. 6.1 GDPR")
    legal_basis_detail = Column(Text,
                                comment="Detail: specific law, contract, or interest")

    # --- Art. 30.1.c — Categories of data subjects and data ---
    data_subject_categories = Column(JSON,
        comment="Categories of data subjects: employees, clients, suppliers...")
    personal_data_categories = Column(JSON,
        comment="Categories of data: identifiers, financial, health...")

    # Art. 9 — Special categories of data
    special_categories = Column(Boolean, nullable=True, default=None,
        comment="Does it include Art. 9 data? Health, biometric, ideology...")
    special_categories_detail = Column(Text,
        comment="If special_categories=True, which categories and Art. 9.2 basis")

    # --- Art. 30.1.d — Recipients ---
    recipients = Column(JSON,
        comment="Categories of data recipients")

    # --- Art. 30.1.e — International transfers ---
    international_transfers = Column(Boolean, nullable=True, default=None,
        comment="Are there transfers outside the EEA?")
    transfer_countries = Column(JSON,
        comment="Destination transfer countries")
    transfer_safeguards = Column(String(100),
        comment="Chapter V mechanism: Art. 45 adequacy, Art. 46 safeguards or another applicable route")

    # --- Art. 30.1.f — Retention periods ---
    retention_period = Column(String(255),
        comment="Envisaged time limit for erasure")
    retention_criteria = Column(Text,
        comment="Retention criteria when the period is not fixed")

    # --- Art. 30.1.g — Security measures (Art. 32) ---
    security_measures = Column(JSON,
        comment="Description of technical and organizational measures")

    # --- Operational fields (not required by Art. 30 but necessary) ---
    status = Column(String(20), default="draft",
        comment="Status: draft, active, under_review, archived")
    department = Column(String(100),
        comment="Department responsible for the processing")
    last_review_date = Column(DateTime,
        comment="Date of last record review")

    # Relationships
    risks = relationship("ProcessingRisk", back_populates="processing_activity")
    evidences = relationship("ProcessingEvidence", back_populates="processing_activity")
