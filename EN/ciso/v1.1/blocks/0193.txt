from datetime import datetime, timezone
from sqlalchemy import Column, Integer, BigInteger, String, Float, Numeric
from sqlalchemy import Boolean, Text, DateTime, Index, ForeignKey, ForeignKeyConstraint, UniqueConstraint, JSON
from sqlalchemy.orm import relationship
from app.database import Base

class LLMUsageLog(Base):
    """Optional monitoring projection of authorized metering records."""
    __tablename__ = 'llm_usage_logs'

    id = Column(BigInteger, primary_key=True, autoincrement=True)

    corporate_id = Column(BigInteger, ForeignKey('corporates.id'), nullable=False, index=True)

    # Who and from where
    user_id = Column(BigInteger, nullable=True)
    service_type = Column(String(50), nullable=False, index=True)
    # E.g.: 'privacy_agent', 'risk_agent', 'compliance_agent',
    #        'rag_query', 'copilot_chat', 'report_writer'

    # Business context (which GRC entity originates the call)
    entity_type = Column(String(50), nullable=True)  # 'dpia', 'risk_analysis'
    entity_id = Column(Integer, nullable=True)

    # Provider and model
    provider_type = Column(String(50), nullable=False)  # 'anthropic', 'ollama'
    model_name = Column(String(100), nullable=False)     # 'claude-sonnet-4-6'

    # Tokens consumed
    prompt_tokens = Column(Integer, nullable=True)
    completion_tokens = Column(Integer, nullable=True)
    total_tokens = Column(Integer, nullable=True)

    # Explicit USD projection only; missing meter/catalog/FX is unknown.
    cost_input = Column(Numeric(18, 8), nullable=True)
    cost_output = Column(Numeric(18, 8), nullable=True)
    cost_total = Column(Numeric(18, 8), nullable=True)
    cost_currency = Column(String(3), nullable=True)
    usage_status = Column(String(40), nullable=False)
    meter_record_id = Column(BigInteger, nullable=True)
    fx_reference = Column(String(200), nullable=True)

    # Performance
    latency_ms = Column(Integer, nullable=True)
    is_cached = Column(Boolean, default=False)
    is_streaming = Column(Boolean, default=False)

    # Result
    success = Column(Boolean, nullable=True)
    error_type = Column(String(100), nullable=True)
    # No exception text, prompt or response is stored.

    # Approved operational correlation; not complete legal audit evidence
    request_id = Column(String(100), nullable=True, index=True)
    prompt_key = Column(String(100), nullable=True)
    rag_collection = Column(String(100), nullable=True)
    rag_results_count = Column(Integer, nullable=True)

    created_at = Column(DateTime(timezone=True),
                        default=lambda: datetime.now(timezone.utc), index=True)

    # Composite indexes for frequent queries
    __table_args__ = (
        ForeignKeyConstraint(['corporate_id', 'user_id'],
                             ['users.corporate_id', 'users.id']),
        UniqueConstraint('corporate_id', 'id', name='uq_usage_tenant_id'),
        Index('idx_usage_service_date', 'corporate_id', 'service_type', 'created_at'),
        Index('idx_usage_provider_date', 'corporate_id', 'provider_type', 'created_at'),
        Index('idx_usage_model_date', 'corporate_id', 'model_name', 'created_at'),
    )
