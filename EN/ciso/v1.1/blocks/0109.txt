# Protected paths authenticate before trusting tenant context.
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import JSONResponse
from fastapi import HTTPException
from app.security.auth_middleware import authenticate_request, bearer_scheme

class TenantMiddleware(BaseHTTPMiddleware):
    EXEMPT_ROUTES = {("POST", "/api/v1/auth/login"), ("GET", "/health"), ("HEAD", "/health")}

    async def dispatch(self, request: Request, call_next):
        if (request.method, request.url.path) in self.EXEMPT_ROUTES:
            return await call_next(request)
        try:
            credentials = await bearer_scheme(request)
            identity = await authenticate_request(request, credentials)
            # Required identity adapter normalizes all auth methods to this server-owned schema.
            if not isinstance(identity, dict):
                raise ValueError("Invalid current identity")
            cid, uid, roles = identity.get("corporate_id"), identity.get("user_id"), identity.get("roles")
            if (type(cid) is not int or cid <= 0 or type(uid) is not int or uid <= 0
                    or not isinstance(roles, list) or len(roles) > 100
                    or any(not isinstance(role, str) or not 1 <= len(role) <= 100 for role in roles)):
                raise ValueError("Invalid current identity scope")
        except HTTPException as exc:
            status = exc.status_code if exc.status_code in (401, 403) else 503
            return JSONResponse(status_code=status, content={"detail": "Access unavailable"})
        except Exception:
            return JSONResponse(status_code=503, content={"detail": "Identity unavailable"})
        request.state.current_identity = identity
        request.state.corporate_id, request.state.user_id, request.state.roles = cid, uid, roles
        return await call_next(request)
# CORS preflight runs outside this middleware; login uses its own atomic attempt limits.
# A valid identity still requires current operation/object permissions and same-tenant queries.
