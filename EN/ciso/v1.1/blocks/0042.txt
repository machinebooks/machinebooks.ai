# Didactic example: patterns/compliance/models.py (continued)

class ComplianceControl(BaseModel):
    """Individual control within a framework."""
    __tablename__ = "compliance_controls"

    id = Column(BigInteger, primary_key=True)
    framework_id = Column(
        BigInteger,
        nullable=False
    )
    parent_id = Column(
        BigInteger,
        nullable=True  # NULL = root control or category
    )
    code = Column(String(50), nullable=False)      # "mp.com.2", "A.8.24"
    name = Column(String(300), nullable=False)
    description = Column(Text)
    guidance = Column(Text)  # Implementation guidance
    level = Column(Integer, default=0)  # Hierarchical level: 0=category, 1=control

    # Compliance status (auditor/CISO evaluation)
    compliance_status = Column(
        SAEnum(ComplianceStatus),
        default=ComplianceStatus.NOT_ASSESSED
    )
    # Implementation status (technical progress)
    implementation_status = Column(
        SAEnum(ImplementationStatus),
        default=ImplementationStatus.NOT_STARTED
    )
    # Non-applicability justification
    not_applicable_justification = Column(Text)

    # For ENS: differentiated requirements by category
    # {"BASICA": "applies", "MEDIA": "applies+", "ALTA": "reinforced"}
    level_requirements = Column(JSON, nullable=True)

    # Relationships
    framework = relationship("ComplianceFramework", back_populates="controls")
    parent = relationship(
        "ComplianceControl",
        remote_side="(ComplianceControl.corporate_id, ComplianceControl.framework_id, ComplianceControl.id)",
        backref="children",
        viewonly=True
    )
    evidences = relationship(
        "Evidence",
        back_populates="control",
        cascade="all, delete-orphan"
    )

    __table_args__ = (
        UniqueConstraint("corporate_id", "id", name="uq_control_tenant_id"),
        UniqueConstraint("corporate_id", "framework_id", "id", name="uq_control_framework_id"),
        UniqueConstraint("corporate_id", "framework_id", "code", name="uq_framework_control"),
        ForeignKeyConstraint(["corporate_id", "framework_id"],
            ["compliance_frameworks.corporate_id", "compliance_frameworks.id"]),
        ForeignKeyConstraint(["corporate_id", "framework_id", "parent_id"],
            ["compliance_controls.corporate_id", "compliance_controls.framework_id", "compliance_controls.id"]),
    )
