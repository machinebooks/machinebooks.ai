# Partial gateway contract, not a copy of the private framework ORM.
# The gateway must scope every relation and return only authorized controls.
# has_current_reviewed_evidence must reflect review, expiry and applicability;
# the boolean is not inferred here from the mere existence of a document.
import json
import math

class ComplianceAgent(BaseAgent):
    TOOLS = ["query_framework", "evaluate_control_gap", "search_rag"]

    def __init__(self, llm_service, db_session, *, corporate_id, tools, model):
        if type(corporate_id) is not int or corporate_id < 1:
            raise ValueError("A verified corporate context is required")
        if not isinstance(model, str) or not model:
            raise ValueError("An authorized model is required")
        super().__init__("compliance", llm_service, db_session)
        self.corporate_id = corporate_id
        self.tools = tools
        self.model = model

    def gather_data(self, params: dict) -> dict:
        framework_id = params.get("framework_id")
        if type(framework_id) is not int or framework_id < 1:
            raise ValueError("framework_id must be a positive integer")
        framework = self.tools.query_framework(
            filters={"id": framework_id, "corporate_id": self.corporate_id,
                     "is_deleted": False})
        if not isinstance(framework, dict):
            raise ValueError("Framework not found in the authorized scope")
        controls = framework.get("controls")
        if not isinstance(controls, list) or not all(isinstance(c, dict) for c in controls):
            raise ValueError("The normalized control collection is required")
        total = len(controls)
        reviewed = sum(c.get("has_current_reviewed_evidence") is True for c in controls)
        return {
            "framework": framework,
            "gaps": self.tools.evaluate_control_gap(
                corporate_id=self.corporate_id, framework_id=framework_id),
            "stats": {
                "total_controls": total, "with_current_reviewed_evidence": reviewed,
                "evidence_coverage_pct": round(reviewed / total * 100, 1) if total else None,
                "scope": "authorized_selected_controls_not_legal_conformity",
            },
            "normative_context": self.tools.search_rag(
                corporate_id=self.corporate_id,
                query="Framework requirements and evidence review",
                corpus="compliance", top_k=6),
        }

    def analyze(self, gathered: dict, params: dict) -> dict:
        context = json.dumps(gathered, ensure_ascii=False, allow_nan=False)
        if len(context) > 30000:
            raise ValueError("Reduce the authorized context")
        response = self.llm_service.invoke(
            model=self.model, system=(
                'Draft a gap review using the framework sources, '
                'scope and edition. Reviewed evidence coverage is an'
                ' internal metric for this sample; it establishes '
                'neither legal conformance nor certification. Identify'
                ' questions and review steps without inventing '
                'evidence, work durations or definitive conclusions.'
            ),
            messages=[{"role": "user", "content": context}],
            max_tokens=4096, temperature=0.2,
        )
        self.total_tokens += response.usage.total_tokens
        cost = getattr(response, "cost", None)
        currency = getattr(response, "cost_currency", None)
        self.cost_known = (type(cost) in (int, float) and math.isfinite(cost)
                           and cost >= 0 and isinstance(currency, str) and bool(currency))
        if self.cost_known:
            self.total_cost += cost
        return {
            "gap_analysis": response.content, "stats": gathered["stats"],
            "model_used": response.model, "tokens": response.usage.total_tokens,
            "cost": cost if self.cost_known else None,
                "cost_currency": currency if self.cost_known else None,
        }

    def generate_output(self, analysis: dict, params: dict) -> dict:
        return {"type": "compliance_gap_review", **analysis,
                "status": "draft", "requires_human_review": True}
