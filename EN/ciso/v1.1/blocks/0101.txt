from ldap3 import Server, Connection, ALL, SUBTREE, Tls
import ssl

LDAP_SERVER = os.environ.get("LDAP_SERVER", "ldaps://ad.example.com:636")
LDAP_BASE_DN = os.environ.get("LDAP_BASE_DN", "DC=example,DC=com")
LDAP_USER_FILTER = os.environ.get("LDAP_USER_FILTER", "(sAMAccountName={username})")
LDAP_BIND_DN = os.environ.get("LDAP_BIND_DN")  # Service account
LDAP_BIND_PASSWORD = os.environ.get("LDAP_BIND_PASSWORD")
LDAP_GROUP_FILTER = os.environ.get("LDAP_GROUP_FILTER", "")  # Access group

async def authenticate_ldap(username: str, password: str, db) -> Optional[User]:
    """Authenticates against Active Directory with user bind.

    Flow:
    1. Connect with service account (bind DN)
    2. Search user by sAMAccountName
    3. Attempt bind with user credentials
    4. If bind succeeds, user is authenticated
    5. Create or update local user for GRC permissions
    """
    tls_config = Tls(validate=ssl.CERT_REQUIRED, version=ssl.PROTOCOL_TLSv1_2)
    server = Server(LDAP_SERVER, use_ssl=True, tls=tls_config, get_info=ALL)

    try:
        service_conn = Connection(server, user=LDAP_BIND_DN, password=LDAP_BIND_PASSWORD, auto_bind=True)
        search_filter = LDAP_USER_FILTER.replace("{username}", username)
        service_conn.search(search_base=LDAP_BASE_DN, search_filter=search_filter,
                           search_scope=SUBTREE, attributes=["cn", "mail", "memberOf", "distinguishedName"])

        if not service_conn.entries:
            return None

        user_entry = service_conn.entries[0]
        user_dn = str(user_entry.distinguishedName)

        # Verify access group membership (if configured)
        if LDAP_GROUP_FILTER:
            member_of = [str(g) for g in user_entry.memberOf]
            if LDAP_GROUP_FILTER not in member_of:
                return None  # User exists but does not have GRC access

        user_conn = Connection(server, user=user_dn, password=password)
        if not user_conn.bind():
            return None

        user_conn.unbind()
        service_conn.unbind()
        return await sync_ldap_user(db, username, user_entry)
