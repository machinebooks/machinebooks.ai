from datetime import datetime, date
from sqlalchemy import BigInteger, Boolean, Date, Enum, Float, ForeignKey, String, Text, UniqueConstraint, ForeignKeyConstraint
from sqlalchemy.dialects.mysql import DATETIME, JSON
from sqlalchemy.orm import Mapped, mapped_column
from app.models.base import BaseModel
import enum


class SourceType(str, enum.Enum):
    """Type of monitored regulatory source."""
    OFFICIAL_JOURNAL = "official_journal"   # BOE, OJEU
    REGULATOR = "regulator"                 # AEPD, CNMC
    STANDARDS_BODY = "standards_body"       # ISO, CEN/CENELEC
    SECTOR_AUTHORITY = "sector_authority"   # CCN-CERT, ENISA


class CheckFrequency(str, enum.Enum):
    """Source query frequency."""
    HOURLY = "hourly"
    DAILY = "daily"
    WEEKLY = "weekly"


class UpdateStatus(str, enum.Enum):
    """State machine for detected updates."""
    NEW = "new"                    # Detected, pending analysis
    ANALYZING = "analyzing"        # Analysis in progress (human or AI)
    ANALYZED = "analyzed"          # Analyzed, pending decision
    DISMISSED = "dismissed"        # No relevant impact
    REQUIRES_ACTION = "requires_action"  # Controls need modification


class RegulatorySource(BaseModel):
    """Monitored regulatory source.

    Each source represents an authority or official gazette
    that publishes regulation relevant to the organization.
    """
    __tablename__ = "regulatory_sources"

    name: Mapped[str] = mapped_column(
        String(200), nullable=False,
        comment="Name: BOE, OJEU, AEPD, ENISA, CCN-CERT"
    )
    source_type: Mapped[SourceType] = mapped_column(
        Enum(SourceType), nullable=False,
        default=SourceType.OFFICIAL_JOURNAL
    )
    url: Mapped[str | None] = mapped_column(
        String(1000), nullable=True,
        comment="Base URL to monitor"
    )
    country: Mapped[str | None] = mapped_column(
        String(10), nullable=True,
        comment="Jurisdiction code: ES or EU; EU is not a country."
    )
    check_frequency: Mapped[CheckFrequency] = mapped_column(
        Enum(CheckFrequency), nullable=False,
        default=CheckFrequency.DAILY
    )
    last_checked_at: Mapped[datetime | None] = mapped_column(
        DATETIME(fsp=6), nullable=True
    )
    is_active: Mapped[bool] = mapped_column(
        Boolean, default=True, nullable=False
    )
    __table_args__ = (
        UniqueConstraint("corporate_id", "id"),
        UniqueConstraint("corporate_id", "name"),
    )
