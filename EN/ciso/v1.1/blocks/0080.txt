from typing import Callable, Any
from dataclasses import dataclass, field


@dataclass
class ToolDefinition:
    """Formal definition of a tool for agents."""
    name: str
    description: str
    handler: Callable[..., Any]
    parameters: dict            # JSON Schema for parameters
    allowed_agents: list[str]   # Agents that can use it
    requires_auth: bool = True  # Whether it needs user context
    audit_level: str = "full"   # full | summary | none


class ToolRegistry:
    """Centralized catalog of tools for agents.

    Controls which tool each agent can use,
    validates parameters before execution, and records
    each invocation for auditing.
    """

    def __init__(self, validate_params, authorize_tool, record_audit):
        if not all(callable(fn) for fn in
                   (validate_params, authorize_tool, record_audit)):
            raise TypeError("Validation, authorization and audit are required")
        self._validate_params = validate_params
        self._authorize_tool = authorize_tool
        self._record_audit = record_audit
        self._tools: dict[str, ToolDefinition] = {}

    def register(self, tool: ToolDefinition) -> None:
        """Registers a tool in the catalog."""
        if tool.audit_level not in {"full", "summary", "none"}:
            raise ValueError("Invalid audit level")
        if tool.name in self._tools:
            raise ValueError("Duplicate tool name")
        self._tools[tool.name] = tool

    def invoke(self, tool_name: str, agent_name: str,
               params: dict, user_context: dict) -> Any:
        """Invokes a tool with validation and auditing."""
        tool = self._tools.get(tool_name)
        if not tool:
            raise ValueError(f"Tool '{tool_name}' not registered")

        # Verify the agent has permission
        if agent_name not in tool.allowed_agents:
            raise PermissionError(
                f"Agent '{agent_name}' not authorized "
                f"for tool '{tool_name}'"
            )

        if not isinstance(params, dict) or "user_context" in params:
            raise ValueError("Invalid tool parameters")
        if tool.requires_auth and not user_context:
            raise PermissionError("Authenticated context is required")
        if not self._authorize_tool(tool_name, agent_name, user_context):
            raise PermissionError("Tool access denied")
        self._validate_params(params, tool.parameters)
        event = {"tool": tool_name, "agent": agent_name,
                 "audit_level": tool.audit_level}
        if tool.audit_level != "none":
            self._record_audit({**event, "status": "started"}, user_context)
        try:
            result = tool.handler(**params, user_context=user_context)
        except Exception as exc:
            if tool.audit_level != "none":
                self._record_audit({**event, "status": "failed",
                                    "error_type": type(exc).__name__}, user_context)
            raise
        if tool.audit_level != "none":
            self._record_audit({**event, "status": "completed"}, user_context)
        return result

    def get_tools_for_agent(self, agent_name: str) -> list[dict]:
        """Returns the tools available for an agent.

        Messages API input_schema shape; an SDK adapter is required.
        """
        return [
            {
                "name": t.name,
                "description": t.description,
                "input_schema": t.parameters
            }
            for t in self._tools.values()
            if agent_name in t.allowed_agents
        ]
