# Server block for mTLS (port 443 with client verification)
server {
    listen 443 ssl;
    server_name grc.example.com;

    # Server certificate
    ssl_certificate     /etc/nginx/ssl/server.crt;
    ssl_certificate_key /etc/nginx/ssl/server.key;

    # Robust TLS configuration
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256;
    ssl_prefer_server_ciphers on;

    # --- mTLS: client certificate verification ---
    ssl_client_certificate /etc/nginx/ssl/trusted-client-ca.pem;

    # "optional" allows clients without certificates to access (they will use JWT)
    # "on" would require a certificate from everyone — only for classified deployments
    ssl_verify_client optional;

    # Certificate chain depth (root CA + intermediates)
    ssl_verify_depth 3;

    # OCSP stapling for server revocation verification
    ssl_stapling on;
    ssl_stapling_verify on;

    location / {
        proxy_pass http://backend:8000;

        # Pass client certificate data to the backend
        proxy_set_header X-Client-Cert       $ssl_client_cert;
        proxy_set_header X-Client-Verify     $ssl_client_verify;
        proxy_set_header X-Client-S-DN       $ssl_client_s_dn;
        proxy_set_header X-Client-I-DN       $ssl_client_i_dn;
        proxy_set_header X-Client-Serial     $ssl_client_serial;

        # Standard headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
