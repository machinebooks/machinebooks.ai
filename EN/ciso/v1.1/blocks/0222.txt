from datetime import datetime, timezone
from celery import shared_task
from app.models.policy_awareness import SecurityPolicy, PolicyStatus
from app.worker.policy_jobs import authorized_policy_review_job

@shared_task(name="policy_awareness.check_review_dates", queue="maintenance")
def check_policy_review_dates(job_id):
    context = authorized_policy_review_job(job_id)
    now = datetime.now(timezone.utc)
    with context.session_scope() as db:
        policies = db.query(SecurityPolicy).filter(
            SecurityPolicy.corporate_id == context.corporate_id,
            SecurityPolicy.is_deleted.is_(False),
            SecurityPolicy.status == PolicyStatus.PUBLISHED,
            SecurityPolicy.review_due_at < now.replace(tzinfo=None),
        ).limit(101).all()
        if len(policies) > 100:
            raise ValueError("Policy review batch exceeds its bound")
        for policy in policies:
            context.recheck_policy_review_permission(policy)
            # Flag review while preserving the current publication/approval/version.
            policy.extra_data = {**(policy.extra_data or {}), "review_pending": True}
            policy.updated_by = context.user_id
            context.outbox.review_reminder_once(
                db, corporate_id=context.corporate_id, policy_id=policy.id,
                version=policy.policy_version, due_at=policy.review_due_at,
                approved_owner=context.authorized_policy_owner(policy))
        db.commit()
    return {"review_candidates": len(policies), "checked_at": now.isoformat()}
# MySQL timestamps use the declared UTC boundary; no client-selected recipient/text.
# Weekly scheduling can delay detection; urgent review/changes require an event-driven path.
# A timer requests review, never approves new text or automatically withdraws a binding policy.
# Outbox/ownership/authorization, retention and actual delivery need verification.
