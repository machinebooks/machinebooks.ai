def seed_normative_corpus(
    indexer: RAGIndexer,
    db_session,
    collection_id: int,
    corporate_id: int,
    collection_name: str = "normative_local",
    embedding_dimensions: int = 768,
):
    """Loads the base regulatory corpus. Idempotent: only indexes what is
    missing or has changed (detection via SHA-256 file hash)."""

    # 1. Ensure the collection exists in Qdrant

    if (type(collection_id) is not int or collection_id <= 0
            or type(corporate_id) is not int or corporate_id <= 0):
        raise ValueError("Invalid collection or tenant identifier")
    collection = db_session.query(RAGCollection).filter_by(
        id=collection_id, corporate_id=corporate_id, is_deleted=False,
    ).one_or_none()
    if (collection is None or collection.name != collection_name
            or collection.embedding_dimensions != embedding_dimensions):
        raise ValueError("Collection metadata does not match authorized tenant")

    indexer.ensure_collection(collection_name, embedding_dimensions)

    for doc_spec in NORMATIVE_CORPUS:
        file_path = doc_spec["file"]
        current_hash = compute_file_hash(file_path)

        # 2. Check if already indexed with the same hash
        existing = db_session.query(RAGDocument).filter_by(
            title=doc_spec["title"],
            collection_id=collection_id,
            corporate_id=corporate_id, is_deleted=False,
        ).one_or_none()

        if existing and existing.file_hash == current_hash:
            continue  # Already indexed, no changes → skip

        if existing and existing.file_hash != current_hash:
            # Document changed → delete old chunks and re-index
            indexer.delete_document_chunks(
                collection_name, doc_spec["title"], corporate_id)
            existing.status = "processing"

        # 3. Extract text and split into chunks
        text = extract_text(file_path)
        chunks = chunk_text(text, chunk_size=512, overlap=64)

        # 4. Index in Qdrant
        indexed = indexer.index_document(
            collection_name=collection_name,
            chunks=chunks,
            metadata={**doc_spec, "corporate_id": corporate_id,
                      "file_hash": current_hash},
        )

        # 5. Register in MySQL
        if not existing:
            rag_doc = RAGDocument(
                collection_id=collection_id,
                corporate_id=corporate_id,
                title=doc_spec["title"],
                source_type=doc_spec["source_type"],
                source_authority=doc_spec.get("authority"),
                source_url=doc_spec.get("source_url"),
                file_path=file_path,
                file_hash=current_hash,
                total_chunks=indexed,
                status="indexed",
            )
            db_session.add(rag_doc)
        else:
            existing.file_hash = current_hash
            existing.total_chunks = indexed
            existing.status = "indexed"

        db_session.commit()
