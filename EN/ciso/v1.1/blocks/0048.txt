# Historical Agents SDK tool; adapters enforce authorized tenant/edition and source access.
from agents import function_tool, RunContextWrapper

@function_tool
async def evaluate_control_gap(ctx: RunContextWrapper, framework_code: str,
                               control_code: str | None = None,
                               ens_category: str | None = None) -> dict:
    trusted = ctx.context
    # Trusted server context supplies async adapters owning their sessions.
    # Neither the model nor the tool arguments choose a tenant or actor.
    trusted.authorize("compliance", "read")
    if framework_code not in trusted.authorized_framework_editions:
        raise PermissionError("Framework is outside the authorized catalog")
    report = await trusted.gap_query(
        corporate_id=trusted.corporate_id,
        framework_code=framework_code,
        framework_version=trusted.authorized_framework_editions[framework_code],
        control_code=control_code, ens_category=ens_category,
        max_age_days=trusted.evidence_max_age_days,
    )
    # The query adapter runs build_gap_report under the same authorization,
    # filters the optional control, and returns at most 100 reviewed rows.
    if len(report.get("gaps", [])) > 100:
        raise ValueError("Narrow the query or use authorized pagination")
    context = await trusted.rag_query(
        corporate_id=trusted.corporate_id, framework_code=framework_code,
        framework_version=report["framework_version"],
        query=control_code or framework_code, top_k=3,
    )
    return {"gap_report": report, "normative_context": context,
            "requires_human_review": True}
