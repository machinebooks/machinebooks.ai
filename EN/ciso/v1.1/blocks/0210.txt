async def seed_regulatory_sources(db, corporate_id, user_id, policy):
    policy.authorize_seed(corporate_id, user_id)
    svc = RegulatoryWatchService(db, corporate_id)
    created = 0
    for source_data in DEFAULT_SOURCES:
        data = dict(source_data)
        data["url"] = await policy.approve_source_url(corporate_id, user_id, data["url"])
        data["is_active"] = False  # Activation requires the reviewed source/fetch configuration.
        if not await svc.source_exists_scoped(data["name"]):
            await svc.create_source(data, user_id)
            created += 1
    await db.flush()
    return created
# Caller owns transaction/rollback; unique corporate_id/name resolves concurrent seeds.
# Sources differ: BOE/DOUE publish binding texts, guides need their own authority/scope review.
# Public URL does not grant unrestricted corpus rights or prove a deployed fetch/analysis pipeline.
