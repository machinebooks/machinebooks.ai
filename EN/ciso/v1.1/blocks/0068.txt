# Didactic example: embedding generation and indexing in Qdrant
from qdrant_client import QdrantClient
from qdrant_client.models import (
    Distance, VectorParams, PointStruct, Filter, FieldCondition, MatchValue
)
import uuid

class RAGIndexer:
    """Indexes regulatory documents in Qdrant with configurable embeddings."""

    def __init__(self, qdrant_url: str, embedding_service):
        self.client = QdrantClient(url=qdrant_url)
        self.embedding = embedding_service  # Local (Ollama) or cloud (Azure)

    def ensure_collection(self, name: str, dimensions: int):
        """Creates the collection if it does not exist. Idempotent."""
        if type(dimensions) is not int or dimensions <= 0:
            raise ValueError("Invalid vector dimensions")
        collections = [c.name for c in self.client.get_collections().collections]
        if name not in collections:
            self.client.create_collection(
                collection_name=name,
                vectors_config=VectorParams(
                    size=dimensions,
                    distance=Distance.COSINE,
                ),
            )

        else:
            params = self.client.get_collection(name).config.params.vectors
            if (not isinstance(params, VectorParams)
                    or params.size != dimensions
                    or params.distance != Distance.COSINE):
                raise ValueError("Incompatible collection vector configuration")

    def index_document(
        self,
        collection_name: str,
        chunks: list[dict],
        metadata: dict,  # regulation, article, authority, date...
    ) -> int:
        """Generates embeddings for each chunk and inserts them in Qdrant.
        Returns the number of indexed chunks."""
        corporate_id = metadata["corporate_id"]
        if type(corporate_id) is not int or corporate_id <= 0:
            raise ValueError("Invalid tenant identifier")
        document_key = metadata["title"]
        version_hash = metadata["file_hash"]
        points = []
        for chunk in chunks:
            # Generate embedding for the chunk text
            vector = self.embedding.encode(chunk["text"])

            point = PointStruct(
                id=str(uuid.uuid5(uuid.NAMESPACE_URL,
                    f"{corporate_id}:{collection_name}:{document_key}:"
                    f"{version_hash}:{chunk['index']}")),
                vector=vector,
                payload={
                    "corporate_id": corporate_id,
                    "file_hash": version_hash,
                    "text": chunk["text"],
                    "chunk_index": chunk["index"],
                    "word_count": chunk["word_count"],
                    # Source document metadata — key for traceability
                    "regulation": metadata.get("regulation", "unknown"),
                    "article": metadata.get("article", ""),
                    "section": metadata.get("section", ""),
                    "authority": metadata.get("authority", ""),
                    "publication_date": metadata.get("publication_date", ""),
                    "document_title": metadata.get("title", ""),
                    "source_url": metadata.get("source_url", ""),
                },
            )
            points.append(point)

        # Upsert in batches of 100 to avoid saturating Qdrant
        batch_size = 100
        for i in range(0, len(points), batch_size):
            batch = points[i:i + batch_size]
            self.client.upsert(
                collection_name=collection_name,
                points=batch,
            )

        return len(points)

    def delete_document_chunks(self, collection_name: str, document_title: str,
                               corporate_id: int):
        """Deletes all chunks of a document before re-indexing."""
        if type(corporate_id) is not int or corporate_id <= 0:
            raise ValueError("Invalid tenant identifier")
        self.client.delete(
            collection_name=collection_name,
            points_selector=Filter(
                must=[
                    FieldCondition(key="corporate_id",
                                   match=MatchValue(value=corporate_id)),
                    FieldCondition(
                        key="document_title",
                        match=MatchValue(value=document_title),
                    )
                ]
            ),
        )
