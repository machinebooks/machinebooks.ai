#!/usr/bin/env bash
# Only an explicitly authorized disposable test deployment and benign login fixture.
set -euo pipefail
: "${AUTHORIZED_TEST_BASE_URL:?Set the authorized HTTPS test URL}"
: "${LOGIN_TEST_PAYLOAD:?Provide a benign invalid-login JSON fixture}"
BASE_URL="${AUTHORIZED_TEST_BASE_URL%/}"
case "$BASE_URL" in https://*) ;; *) echo "HTTPS is required"; exit 1 ;; esac
HEADERS=$(mktemp)
CLEAN=$(mktemp)
trap 'rm -f "$HEADERS" "$CLEAN"' EXIT
STATUS=$(curl --proto '=https' --max-time 15 --silent --show-error \
  --dump-header "$HEADERS" --output /dev/null --write-out '%{http_code}' "$BASE_URL")
[ "$STATUS" = "200" ] || { echo "Unexpected HTTP status"; exit 1; }
tr -d '\r' < "$HEADERS" > "$CLEAN"
# Selected properties of the chapter's static CSP; no browser enforcement claim.
grep -Eiq "^content-security-policy:.*script-src[[:space:]]+'self'[[:space:]]*;" "$CLEAN"
grep -Eiq '^strict-transport-security:.*max-age=31536000([;[:space:]]|$)' "$CLEAN"
grep -Eiq '^x-frame-options:[[:space:]]*DENY[[:space:]]*$' "$CLEAN"
RATE_LIMIT_SEEN=0
for i in $(seq 1 8); do
  STATUS=$(printf '%s' "$LOGIN_TEST_PAYLOAD" | curl --proto '=https' --max-time 15 \
    --silent --show-error --output /dev/null --write-out '%{http_code}' \
    --request POST "$BASE_URL/api/v1/auth/login" \
    --header 'Content-Type: application/json' --data-binary @-)
  if [ "$STATUS" = "429" ]; then RATE_LIMIT_SEEN=1; break; fi
  [ "$STATUS" = "401" ] || { echo "Unexpected authentication status"; exit 1; }
done
[ "$RATE_LIMIT_SEEN" = "1" ] || { echo "Expected rate limit was not observed"; exit 1; }
echo "Scoped check passed"
# Assumes the reviewed 5-attempt account threshold and compatible login DTO.
# Fresh nonces/HTML linkage, policy behavior, TLS, roles and recovery need other tests.
# This neither establishes remote CI/deployment nor replaces a professional penetration test.
