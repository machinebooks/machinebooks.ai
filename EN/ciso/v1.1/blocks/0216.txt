# Continuation of policy model imports.
from sqlalchemy import UniqueConstraint, CheckConstraint

class SecurityCultureScore(BaseModel):
    __tablename__ = "pa_security_culture_scores"
    department: Mapped[str] = mapped_column(String(255), nullable=False)
    period: Mapped[str] = mapped_column(String(20), nullable=False)
    overall_score: Mapped[float | None] = mapped_column(Float, nullable=True)
    phishing_score: Mapped[float | None] = mapped_column(Float, nullable=True)
    training_score: Mapped[float | None] = mapped_column(Float, nullable=True)
    policy_compliance_score: Mapped[float | None] = mapped_column(Float, nullable=True)
    incident_reporting_score: Mapped[float | None] = mapped_column(Float, nullable=True)
    breakdown: Mapped[dict | None] = mapped_column(JSON, nullable=True)
    __table_args__ = (
        UniqueConstraint("corporate_id", "department", "period"),
        CheckConstraint("overall_score IS NULL OR (overall_score >= 0 AND overall_score <= 100)"),
    )
# Score weights/denominators, known-data coverage and validated period are case policy.
# Missing components remain NULL unless a disclosed partial-score method is approved.
# 100-click_rate is meaningful only for a validated comparable measured cohort/window.
# Department aggregation is not anonymity and does not prove culture, effectiveness or conformity.
