# Continuation of the source model imports; publication, entry into force and application dates differ.
class RegulationType(str, enum.Enum):
    REGULATION = "regulation"
    DIRECTIVE = "directive"
    LAW = "law"
    GUIDANCE = "guidance"
    OTHER = "other"


class RegulatoryUpdate(BaseModel):
    """Detected regulatory update.

    Records each relevant publication with classification
    metadata, impact analysis, and affected frameworks.
    """
    __tablename__ = "regulatory_updates"

    source_id: Mapped[int] = mapped_column(
        BigInteger,
        nullable=False, index=True
    )
    title: Mapped[str] = mapped_column(String(1000), nullable=False)
    summary: Mapped[str | None] = mapped_column(Text, nullable=True)
    original_url: Mapped[str | None] = mapped_column(
        String(2000), nullable=True
    )
    publication_date: Mapped[date | None] = mapped_column(
        Date, nullable=True
    )
    effective_date: Mapped[date | None] = mapped_column(
        Date, nullable=True,
        comment="Effective date — may differ from publication"
    )
    regulation_type: Mapped[RegulationType] = mapped_column(
        Enum(RegulationType), nullable=False,
        default=RegulationType.REGULATION
    )
    sector: Mapped[str | None] = mapped_column(
        String(100), nullable=True,
        comment="energy|finance|telecom|health|all"
    )
    status: Mapped[UpdateStatus] = mapped_column(
        Enum(UpdateStatus), nullable=False,
        default=UpdateStatus.NEW
    )
    relevance_score: Mapped[float | None] = mapped_column(
        Float, nullable=True,
        comment="Relevance score 0-100"
    )
    # --- AI analysis fields ---
    ai_summary: Mapped[str | None] = mapped_column(
        Text, nullable=True,
        comment="LLM-generated summary"
    )
    ai_impact_analysis: Mapped[str | None] = mapped_column(
        Text, nullable=True,
        comment="LLM-generated impact analysis"
    )
    affected_frameworks: Mapped[list[str] | None] = mapped_column(
        JSON, nullable=True,
        comment='["ISO27001", "NIS2", "DORA"]'
    )
    affected_controls: Mapped[list[dict] | None] = mapped_column(
        JSON, nullable=True,
        comment='[{"framework": "NIS2", "control_id": "5.1", "impact": "high"}]'
    )
    assigned_to: Mapped[str | None] = mapped_column(
        String(300), nullable=True
    )
    publication_key: Mapped[str] = mapped_column(String(64), nullable=False)
    application_schedule: Mapped[list[dict] | None] = mapped_column(JSON, nullable=True)
    __table_args__ = (
        ForeignKeyConstraint(["corporate_id", "source_id"],
                             ["regulatory_sources.corporate_id", "regulatory_sources.id"]),
        UniqueConstraint("corporate_id", "source_id", "publication_key"),
    )
# Ingestion supplies a stable reviewed publication/revision key; hash is not authenticity.
# Application schedule entries carry obligation/jurisdiction/date/source and human review.
# LLM affected_controls/deadlines are drafts, never authority for state changes or actions.
