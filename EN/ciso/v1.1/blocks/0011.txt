from pydantic import BaseModel, Field, model_validator
from app.models.privacy import LegalBasis, TransferSafeguard

class ProcessingActivityCreate(BaseModel):
    # Draft fields: Article 6 legal basis is operational information, not Art. 30.1.
    name: str = Field(min_length=3, max_length=255)
    controller_name: str = Field(min_length=3)
    purposes: list[str] = Field(min_length=1)
    legal_basis: LegalBasis
    controller_contact: str | None = None
    joint_controller_contacts: list[str] = Field(default_factory=list)
    representative_contact: str | None = None
    representative_applicable: bool = False
    dpo_contact: str | None = None
    dpo_applicable: bool = False
    data_subject_categories: list[str] = Field(default_factory=list)
    personal_data_categories: list[str] = Field(default_factory=list)
    recipients: list[str] = Field(default_factory=list)
    special_categories: bool = False
    special_categories_detail: list[str] = Field(default_factory=list)
    article9_condition: str | None = None
    international_transfers: bool = False
    transfer_countries: list[str] = Field(default_factory=list)
    transfer_safeguards: TransferSafeguard | None = None
    retention_period: str | None = None
    retention_criteria: str | None = None
    security_measures: list[str] = Field(default_factory=list)

    @model_validator(mode="after")
    def validate_conditions(self):
        if self.special_categories and (
            not self.special_categories_detail or not self.article9_condition
        ):
            raise ValueError("Special categories require detail and an Article 9 condition")
        if self.international_transfers and not self.transfer_countries:
            raise ValueError("Transfer destinations must be identified")
        return self

class ProcessingActivityActivate(ProcessingActivityCreate):
    controller_contact: str = Field(min_length=1)
    data_subject_categories: list[str] = Field(min_length=1)
    personal_data_categories: list[str] = Field(min_length=1)
    security_measures: list[str] = Field(min_length=1)

    @model_validator(mode="after")
    def validate_activation(self):
        if self.dpo_applicable and not self.dpo_contact:
            raise ValueError("DPO contact is required when applicable")
        if self.representative_applicable and not self.representative_contact:
            raise ValueError("Representative contact is required when applicable")
        if not self.retention_period and not self.retention_criteria:
            raise ValueError("Record erasure periods where possible or justify their absence")
        # Empty recipient categories can mean no recipients; review must confirm it.
        # Chapter V assessment and safeguard evidence are separate from destinations.
        return self
