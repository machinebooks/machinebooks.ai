# gateway/app/services/prompt_service.py (synthetic: public interface)
class PromptService:
    async def get_active(self, key: str, *, db: AsyncSession) -> AiPrompt:
        """Returns the active prompt for a given key."""
        result = await db.execute(
            select(AiPrompt).where(
                AiPrompt.key == key,
                AiPrompt.status == "active",
            )
        )
        return result.scalar_one_or_none()

    async def get_variant(self, key: str, user_id: int, *, db: AsyncSession) -> AiPrompt:
        """Returns the prompt considering sticky A/B testing."""
        active = await self.get_active(key, db=db)
        if active and active.ab_variant:
            variant = self._select_variant(key, user_id)
            return await self._get_variant(db, key, variant) or active
        return active

    async def render(self, key: str, variables: dict, user_id: int = None, *, db: AsyncSession) -> str:
        """Resolves and renders a prompt with its variables."""
        prompt = await self.get_variant(key, user_id or 0, db=db)
        if prompt is None:
            raise PromptNotFound(f"Prompt '{key}' not found or inactive")
        content = render_template(prompt.content, variables)
        # Increment popularity
        await db.execute(
            update(AiPrompt).where(AiPrompt.id == prompt.id).values(popularity=AiPrompt.popularity + 1)
        )
        return content

    async def create_draft(self, key: str, content: str, *, db: AsyncSession, **kwargs) -> AiPrompt:
        """Creates a draft with the next version."""
        last = await db.execute(
            select(func.max(AiPrompt.version)).where(AiPrompt.key == key)
        )
        version = (last.scalar() or 0) + 1
        draft = AiPrompt(key=key, version=version, content=content, status="draft", **kwargs)
        db.add(draft)
        return draft

    async def activate(self, prompt_id: int, *, db: AsyncSession) -> AiPrompt:
        """Activates a draft, deprecating the previous active one."""
        draft = await db.get(AiPrompt, prompt_id)
        if draft.status != "draft":
            raise ValueError("Only a draft can be activated")
        # Deprecate existing actives with the same key
        actives = await db.execute(
            select(AiPrompt).where(AiPrompt.key == draft.key, AiPrompt.status == "active")
        )
        for active in actives.scalars():
            active.status = "deprecated"
        draft.status = "active"
        return draft
