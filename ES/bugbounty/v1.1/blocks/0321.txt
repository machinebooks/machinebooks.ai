#!/usr/bin/env python3
"""
WebView2 MassTest: verificación de exposición CDP simultánea.
Detecta qué aplicaciones WebView2 exponen targets en el puerto CDP.
SOLO PARA USO EN LABORATORIO DE INVESTIGACIÓN AUTORIZADO.
"""

import json
import urllib.request
import subprocess
import time
import os

CDP_PORT = 9222
APPS_TO_TEST = {
    "Teams": r"%LOCALAPPDATA%\Microsoft\Teams\current\Teams.exe",
    "Outlook": r"%LOCALAPPDATA%\Microsoft\Olk\olk.exe",
    "Acrobat": r"%ProgramFiles%\Adobe\Acrobat DC\Acrobat\Acrobat.exe",
}

def check_cdp_targets() -> list:
    """Consulta el endpoint CDP y devuelve los targets activos."""
    try:
        url = f"http://127.0.0.1:{CDP_PORT}/json"
        with urllib.request.urlopen(url, timeout=3) as resp:
            return json.loads(resp.read().decode())
    except Exception:
        return []

def run_masstest():
    """Ejecuta la prueba de exposición CDP en múltiples aplicaciones."""
    print("[*] WebView2 MassTest — verificación de CDP multiplex")
    print(f"[*] Puerto CDP: {CDP_PORT}")
    print()
    
    # Verificar que la variable de entorno está configurada
    env_val = os.environ.get("WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS", "")
    if "--remote-debugging-port" not in env_val:
        print("[!] Variable de entorno no configurada. Abortando.")
        return
    
    # Esperar a que las aplicaciones arranquen (en prueba manual)
    print("[*] Verificando targets CDP activos...")
    targets = check_cdp_targets()
    
    if not targets:
        print("[!] No se encontraron targets CDP activos.")
        return
    
    # Clasificar targets por aplicación
    apps_found = {}
    for t in targets:
        url = t.get("url", "")
        title = t.get("title", "")
        
        if "teams" in url.lower() or "teams" in title.lower():
            app = "Teams"
        elif "outlook" in url.lower() or "mail" in title.lower():
            app = "Outlook"
        elif "acrobat" in url.lower() or "adobe" in url.lower():
            app = "Acrobat"
        else:
            app = "Otro"
        
        if app not in apps_found:
            apps_found[app] = []
        apps_found[app].append(t)
    
    # Resumen
    print(f"\n[+] {len(targets)} targets en {len(apps_found)} aplicaciones:\n")
    for app, app_targets in apps_found.items():
        print(f"  {app}: {len(app_targets)} targets")
        for at in app_targets:
            print(f"    - {at.get('title', 'Sin título')[:50]}")
    
    # Verificación de multiplex
    if len(apps_found) > 1:
        print(f"\n[!!] MULTIPLEX CONFIRMADO: {len(apps_found)} aplicaciones")
        print("     expuestas en el mismo puerto CDP.")

if __name__ == "__main__":
    run_masstest()
