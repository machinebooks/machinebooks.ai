import ctypes
from ctypes import wintypes
import struct

# Constantes de la API de Windows
CRYPTPROTECT_UI_FORBIDDEN = 0x01

class DATA_BLOB(ctypes.Structure):
    _fields_ = [
        ("cbData", wintypes.DWORD),
        ("pbData", ctypes.POINTER(ctypes.c_byte))
    ]

def decrypt_dpapi_blob(encrypted_blob: bytes, entropy: bytes) -> bytes:
    """Descifra un blob DPAPI usando la entropy de SessionP."""
    # Cabecera DPAPI estandar: 20 bytes
    # MobaXterm almacena el blob sin cabecera; hay que prependerla
    DPAPI_HEADER = bytes.fromhex(
        "01000000D08C9DDF0115D1118C7A00C04FC297EB"
    )
    full_blob = DPAPI_HEADER + encrypted_blob

    blob_in = DATA_BLOB()
    blob_in.cbData = len(full_blob)
    blob_in.pbData = (ctypes.c_byte * len(full_blob))(*full_blob)

    entropy_blob = DATA_BLOB()
    entropy_blob.cbData = len(entropy)
    entropy_blob.pbData = (ctypes.c_byte * len(entropy))(*entropy)

    blob_out = DATA_BLOB()

    result = ctypes.windll.crypt32.CryptUnprotectData(
        ctypes.byref(blob_in),
        None,                          # descripcion (no necesaria)
        ctypes.byref(entropy_blob),    # entropy de SessionP
        None,                          # reservado
        None,                          # prompt (no UI)
        CRYPTPROTECT_UI_FORBIDDEN,
        ctypes.byref(blob_out)
    )

    if result:
        size = blob_out.cbData
        data = bytes((ctypes.c_byte * size).from_address(
            ctypes.addressof(blob_out.pbData.contents)
        ))
        # Liberar memoria asignada por CryptUnprotectData
        ctypes.windll.kernel32.LocalFree(blob_out.pbData)
        return data  # SHA-512 de la contraseña maestra (64 bytes)
    else:
        error = ctypes.GetLastError()
        print(f"[-] CryptUnprotectData fallo: error {error}")
        return None
