# Script de detección: verificar si hay un servidor CDP activo
# Ejecutar periódicamente o como servicio de monitorización
$ports = @(9222, 9229, 9230, 9231)  # Puertos comunes de depuración

foreach ($port in $ports) {
    try {
        $tcp = New-Object System.Net.Sockets.TcpClient
        $tcp.Connect("127.0.0.1", $port)
        if ($tcp.Connected) {
            Write-Warning "ALERTA: Puerto CDP $port activo en localhost"
            
            # Consultar el endpoint para identificar la aplicación
            $response = Invoke-RestMethod `
                -Uri "http://127.0.0.1:$port/json/version" `
                -TimeoutSec 3 -ErrorAction SilentlyContinue
            
            if ($response) {
                Write-Warning "  Browser: $($response.Browser)"
                Write-Warning "  User-Agent: $($response.'User-Agent')"
            }
            $tcp.Close()
        }
    }
    catch {
        # Puerto no activo — situación normal
    }
}

# Verificar la variable de entorno
$envVar = [Environment]::GetEnvironmentVariable(
    'WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS', 'User')

if ($envVar) {
    Write-Warning "ALERTA: WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS = $envVar"
}
