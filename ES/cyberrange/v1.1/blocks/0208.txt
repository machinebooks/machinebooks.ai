# Bounded heuristic signals, not calibrated confidence or competence
from collections import Counter
from datetime import datetime, timedelta, timezone

class StallDetector:
    def __init__(self,inactivity_minutes=10,window_minutes=5,repetitions=3):
        if any(type(value) is not int or value<1 for value in (inactivity_minutes,window_minutes,repetitions)):
            raise ValueError("positive bounds required")
        self.inactivity_minutes=inactivity_minutes
        self.window_minutes=window_minutes
        self.repetitions=repetitions
    def evaluate(self,actions,last_action_at,failed_flag_attempts=0,now=None):
        now=now or datetime.now(timezone.utc)
        if now.tzinfo is None or last_action_at.tzinfo is None or last_action_at>now:
            raise ValueError("ordered aware timestamps required")
        cutoff=now-timedelta(minutes=self.window_minutes)
        recent=[]
        for action in actions:
            stamp=action["timestamp"]
            if stamp.tzinfo is None or stamp>now:
                raise ValueError("invalid action timestamp")
            if cutoff<=stamp<=now and str(action["command"]).strip():
                recent.append(str(action["command"]).strip().split()[0].lower())
        patterns=[]
        if (now-last_action_at).total_seconds()>=self.inactivity_minutes*60:
            patterns.append("inactivity")
        if Counter(recent).most_common(1) and Counter(recent).most_common(1)[0][1]>=self.repetitions:
            patterns.append("repetition")
        if len(recent)>=6 and len(set(recent))<=3:
            patterns.append("circular_pattern")
        if type(failed_flag_attempts) is not int or failed_flag_attempts<0:
            raise ValueError("invalid failed attempt count")
        if failed_flag_attempts>=3:
            patterns.append("failed_flag_attempts")
        return {"possible_stall":bool(patterns),"patterns":patterns,
                "window_minutes":self.window_minutes,"calibrated_probability":None}

# Bound and authorize input at collection. An empty/missing action stream
# may indicate telemetry loss; signals require instructor/participant context.
