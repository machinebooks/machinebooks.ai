// Didactic ES/EN provider; dictionary is validated at the application boundary.
import React,{createContext,useContext,useEffect,useState} from 'react';
type Language='es'|'en';
type TranslationValue={language:Language;setLanguage:(value:Language)=>void;t:(key:string)=>string};
const TranslationContext=createContext<TranslationValue|null>(null);
export const TranslationProvider=({children,dictionaries}:
  {children:React.ReactNode;dictionaries:Record<Language,Record<string,unknown>>})=>{
  const [language,setLanguage]=useState<Language>('es');
  useEffect(()=>{
    let saved:string|null=null;
    try{saved=localStorage.getItem('language');}catch{}
    const browser=navigator.language.split('-')[0];
    setLanguage(saved==='es'||saved==='en'?saved:browser==='es'?'es':'en');
  },[]);
  const changeLanguage=(value:Language)=>{
    if(value!=='es'&&value!=='en')return;
    setLanguage(value);
    try{localStorage.setItem('language',value);}catch{}
  };
  const t=(key:string)=>{
    let value:unknown=dictionaries[language];
    for(const part of key.split('.')){
      if(!value||typeof value!=='object'||!Object.prototype.hasOwnProperty.call(value,part))return key;
      value=(value as Record<string,unknown>)[part];
    }
    return typeof value==='string'?value:key;
  };
  return <TranslationContext.Provider value={{language,setLanguage:changeLanguage,t}}>
    {children}
  </TranslationContext.Provider>;
};
export const useTranslation=()=>{
  const value=useContext(TranslationContext);
  if(!value)throw new Error('TranslationProvider required');
  return value;
};
