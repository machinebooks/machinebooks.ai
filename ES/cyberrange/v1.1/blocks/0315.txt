# Limited lexical heuristic; not comprehensive injection detection
def coaching_text(input_data):
    if isinstance(input_data, str):
        if not 1 <= len(input_data) <= 4096:
            raise ValueError("bounded coaching text required")
        return input_data
    if not isinstance(input_data, list) or not 1 <= len(input_data) <= 64:
        raise ValueError("unsupported coaching input")
    parts = []
    for message in input_data:
        if not isinstance(message, dict):
            raise ValueError("unsupported message")
        content = message.get("content")
        if isinstance(content, str):
            if not 1 <= len(content) <= 4096:
                raise ValueError("bounded content required")
            parts.append(content)
        elif isinstance(content, list):
            if not 1 <= len(content) <= 16:
                raise ValueError("bounded content list required")
            for item in content:
                if not isinstance(item, dict) or item.get("type") != "input_text" or not isinstance(item.get("text"), str):
                    raise ValueError("unsupported content")
                if not 1 <= len(item["text"]) <= 4096:
                    raise ValueError("bounded content required")
                parts.append(item["text"])
        else:
            raise ValueError("unsupported content")
    value = "\n".join(parts)
    if len(value.encode("utf-8")) > 65536:
        raise ValueError("bounded total input required")
    return value

def coaching_input_guardrail(ctx, agent, input_data):
    from agents import GuardrailFunctionOutput
    try:
        value = coaching_text(input_data).casefold()
    except ValueError:
        return GuardrailFunctionOutput(output_info={"reason": "unsupported_input"},
                                       tripwire_triggered=True)
    # Bilingual fixtures cover selected phrases, not all injection variants.
    patterns = ("dame la flag", "ignora tus instrucciones", "give me the flag", "ignore your instructions", "reveal your instructions")
    matched = any(pattern in value for pattern in patterns)
    return GuardrailFunctionOutput(output_info={"heuristic_match": matched},
                                   tripwire_triggered=matched)

def build_secure_coach(configured_model):
    from agents import Agent, InputGuardrail
    return Agent(name="secure_coach", model=configured_model,
        instructions="Guide the authorized learner without solutions. "
                     "Secret flags are never included in model context.",
        input_guardrails=[InputGuardrail(guardrail_function=coaching_input_guardrail,
                                        run_in_parallel=False)])
