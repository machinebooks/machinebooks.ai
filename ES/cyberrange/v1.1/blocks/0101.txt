# Shared admission adapter; it raises on exhausted or unavailable policy.
def admit_flag_attempt(adapter, actor_id, instance_id):
    if any(type(value) is not int or value<=0 for value in [actor_id, instance_id]):
        raise ValueError("positive actor and instance IDs required")
    adapter.consume_shared_attempt_limit(actor_id, instance_id, 10, 60)
    return {"status":"admitted"}
