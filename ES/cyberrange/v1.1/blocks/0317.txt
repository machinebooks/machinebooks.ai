# Proposed RL integration contract; no attack is executed here
def choose_authorized_action(observation, policy, allowed_actions):
    if policy is None:
        raise NotImplementedError("trained and evaluated policy required")
    action = policy.predict(observation)
    if action not in allowed_actions:
        raise ValueError("action outside authorized catalog")
    return {"action": action, "status": "proposal", "calibrated_probability": None}
