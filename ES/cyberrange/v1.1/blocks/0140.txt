# Didactic optional proposals; validators and instructor review are independent.
import asyncio
import json

async def generate_challenge_with_ai(request, provider, validator, catalog):
    verified_request = validator.validate_request(request, catalog)
    if provider is None:
        return {"status": "manual_review_required"}
    raw = await asyncio.wait_for(provider.propose_challenge(verified_request), 20)
    if not isinstance(raw, str) or len(raw.encode("utf-8")) > 65536:
        raise ValueError("invalid provider response")
    proposal = validator.validate_proposal(json.loads(raw), catalog)
    return {"status": "pending_review", "proposal": proposal}

async def generate_adaptive_hint(actor, instance_ref, coach):
    # Coach owns authorization, minimized context, level, secrets, budgets and penalty ledger.
    return await asyncio.wait_for(coach.validated_recorded_hint(actor, instance_ref), 20)
