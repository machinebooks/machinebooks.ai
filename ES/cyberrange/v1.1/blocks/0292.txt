# NGINX: nginx/default.conf; certificate/key provisioned separately
upstream backend { server backend:8000; }
upstream frontend { server frontend:80; }
server {
    listen 443 ssl;
    server_name _;
    ssl_certificate /etc/nginx/certs/server.crt;
    ssl_certificate_key /etc/nginx/certs/server.key;
    ssl_protocols TLSv1.2 TLSv1.3;
    client_max_body_size 50M;
    location /api/ {
        proxy_pass http://backend/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $remote_addr;
    }
    location ~ ^/(ws|console)/ {
        proxy_pass http://backend;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $remote_addr;
        proxy_read_timeout 60s;
        proxy_buffering off;
    }
    location / { proxy_pass http://frontend; }
}
# No generic management-API proxy. Backend console gateways must authorize
# resources, consume opaque tickets and validate upstream certificates.
# Only trust forwarding headers from this configured ingress boundary.
