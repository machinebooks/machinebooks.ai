# Didactic SQLAlchemy 2.x fragment; external secret store retains credential values.
from sqlalchemy import Column, Integer, String, BigInteger, DateTime, ForeignKey

def cluster_models(Base):
    class ProxmoxCluster(Base):
        __tablename__ = "proxmox_cluster"
        id = Column(Integer, primary_key=True)
        name = Column(String(128), nullable=False, unique=True)
        host = Column(String(255), nullable=False)
        port = Column(Integer, default=8006)
        credential_ref = Column(String(128), nullable=False)
        trusted_ca_ref = Column(String(128), nullable=False)
        status = Column(String(32), default="unknown")
        last_sync = Column(DateTime(timezone=True))
    class ProxmoxNode(Base):
        __tablename__ = "proxmox_node"
        id = Column(Integer, primary_key=True)
        cluster_id = Column(Integer, ForeignKey("proxmox_cluster.id"), nullable=False)
        name = Column(String(128), nullable=False)
        memory_total_bytes = Column(BigInteger)
        ip_address = Column(String(45))
        status = Column(String(32), default="unknown")
        last_sync = Column(DateTime(timezone=True))
    return ProxmoxCluster, ProxmoxNode
# Adapter normalizes UTC and applies per-cluster/node uniqueness in the private migration.
