# Authorized command telemetry supplied by a PTY/guest instrumentation adapter
from collections import deque
from datetime import datetime, timedelta, timezone

class ActionTracker:
    def __init__(self,max_sessions=100,max_actions=500,ttl_minutes=120):
        if any(type(value) is not int or value<1 for value in (max_sessions,max_actions,ttl_minutes)):
            raise ValueError("positive telemetry bounds required")
        self.sessions={};self.max_sessions=max_sessions
        self.max_actions=max_actions;self.ttl=timedelta(minutes=ttl_minutes)
    def purge(self,now):
        for key,state in list(self.sessions.items()):
            if state["expires_at"]<=now:
                del self.sessions[key]
    def track(self,actor_id,instance_id,command,authorize,redact):
        authorize(actor_id,"command_telemetry",instance_id)
        now=datetime.now(timezone.utc);self.purge(now)
        key=(actor_id,instance_id)
        if key not in self.sessions:
            if len(self.sessions)>=self.max_sessions:
                raise ValueError("telemetry capacity reached")
            self.sessions[key]={"expires_at":now+self.ttl,"actions":deque(maxlen=self.max_actions)}
        sanitized=redact(command)
        if not isinstance(sanitized,str) or len(sanitized)>500:
            raise ValueError("bounded redacted command required")
        action={"timestamp":now,"command":sanitized,"category":"unclassified"}
        self.sessions[key]["actions"].append(action)
        return action

# No command extraction from RFB/VNC is implemented. The collector supplies
# authorized structured events and verified redaction before storage. Pattern
# categories are heuristics, not MITRE technique IDs or proof of execution.
