// Didactic bounded sessionStorage cache; backend owns durable evidence.
export const LOG_TTL_MS=7_200_000;
export function loadPersistedLogs(storage,key,ownerId,sessionId,now=Date.now()) {
  try {
    const raw=storage.getItem(key);
    if(!raw)return [];
    if(raw.length>131072)throw new Error('oversized');
    const data=JSON.parse(raw);
    if(data.ownerId!==ownerId||data.sessionId!==sessionId||
       !Number.isFinite(data.timestamp)||data.timestamp>now||now-data.timestamp>=LOG_TTL_MS||
       !Array.isArray(data.logs)||data.logs.length>500)throw new Error('invalid');
    if(!data.logs.every(log=>typeof log.line==='string'&&log.line.length<=1000&&
       typeof log.timestamp==='string'&&['output','error','status'].includes(log.type)))throw new Error('invalid');
    return data.logs;
  }catch {try{storage.removeItem(key);}catch{}return [];}
}
export function persistLogs(storage,key,ownerId,sessionId,logs,redact,now=Date.now()) {
  if(!Array.isArray(logs))throw new Error('invalid_logs');
  const data={ownerId,sessionId,timestamp:now,logs:logs.slice(-500).map(redact)};
  const raw=JSON.stringify(data);
  if(raw.length>131072)throw new Error('cache_limit');
  try{storage.setItem(key,raw);}catch {try{storage.removeItem(key);}catch{}}
}
// Use actor-scoped keys and remove them on logout/user change. Redaction and
// schema validation happen before writing. Scripts of this origin can read it;
// storage availability, retention and replay are not guaranteed by this cache.
