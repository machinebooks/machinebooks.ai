# Didactic example: shared REST/WebSocket identity verification
import asyncio
import hmac
from fastapi import Depends, HTTPException, Request
from jose import jwt, JWTError
from backend.models import User
from backend.database import get_db
from backend.config import settings

async def verified_user(token, db, secret, issuer, audience):
    try:
        if hasattr(secret, "get_secret_value"):
            secret = secret.get_secret_value()
        payload = jwt.decode(
            token, secret, algorithms=["HS256"], issuer=issuer, audience=audience,
            options={"require_exp": True, "require_iat": True,
                     "require_sub": True, "require_iss": True, "require_aud": True},
        )
        subject, sid = payload["sub"], payload.get("sid")
        if not isinstance(subject, str) or not subject.isascii() or not subject.isdecimal() or len(subject) > 20:
            raise ValueError("invalid subject")
        if not isinstance(sid, str) or not sid:
            raise ValueError("missing session")
        user = await asyncio.wait_for(db.get_current_account(int(subject)), 5)
        if user is None or not user.is_active:
            raise ValueError("inactive user")
        current_sid = user.session_token
        if not isinstance(current_sid, str) or not current_sid:
            raise ValueError("revoked session")
        if not hmac.compare_digest(sid.encode("utf-8"), current_sid.encode("utf-8")):
            raise ValueError("invalidated session")
    except (JWTError, KeyError, ValueError, TypeError):
        raise HTTPException(status_code=401, detail="Invalid session") from None
    return user

async def get_current_user(request: Request, db=Depends(get_db)):
    parts = request.headers.get("Authorization", "").split()
    if len(parts) != 2 or parts[0].lower() != "bearer":
        raise HTTPException(status_code=401, detail="Authentication required")
    user = await verified_user(parts[1], db, settings.jwt_secret,
                         settings.jwt_issuer, settings.jwt_audience)
    request.state.user = user
    return user
