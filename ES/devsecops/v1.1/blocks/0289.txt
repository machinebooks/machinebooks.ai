from datetime import datetime,timezone
def generar_informe_auditoria(resultados_controles: dict,
                               client: anthropic.Anthropic):
    """Genera borrador parcial ENS en formato Markdown
    legible para auditores no técnicos."""

    prompt = f"""Genera un borrador técnico parcial ENS; conserva REVIEW/INCOMPLETE
a partir de los siguientes resultados de evaluación de controles.

{json.dumps(resultados_controles, indent=2, ensure_ascii=False)}

El informe debe seguir esta estructura:
1. Resumen ejecutivo: estados REVIEW / INCOMPLETE y cobertura
2. Marco operacional: tabla con cada control, estado y evidencia
3. Marco de protección: tabla con cada control, estado y evidencia
4. Hallazgos y acciones pendientes: propuestas y omisiones pendientes de revisión
5. Pendientes de revisión competente; no emitir dictamen de conformidad

Tono: técnico pero accesible para un auditor de seguridad
que no es desarrollador. Referenciar los artefactos por nombre
y fecha. NO inventar evidencias ni hallazgos que no estén
en los datos proporcionados."""

    message = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=4096,
        messages=[{"role": "user", "content": prompt}]
    )
    return "REVIEW: partial ENS draft; no conformity determination\n" + message.content[0].text


def ejecutar_compliance_completo(bucket_path: str):
    """Orquesta cuatro controles mapeados; conserva cobertura parcial."""
    client = anthropic.Anthropic()

    # 1. Recopilar evidencias de los últimos 90 días
    evidencias = recopilar_evidencias(bucket_path, periodo_dias=90)

    # 2. Evaluar cada control
    resultados = {}
    for control_id, control_def in ENS_CONTROL_MAP.items():
        resultados[control_id] = evaluar_control(
            control_id, control_def, evidencias, client
        )

    if not resultados or any(not isinstance(row,dict) or row.get("status") not in {"REVIEW","INCOMPLETE"} for row in resultados.values()):
        raise ValueError("INCOMPLETE: missing or invalid mapped-control results")

    # 3. Generar informe
    informe = generar_informe_auditoria(resultados, client)

    # 4. Guardar con metadatos
    fecha = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    Path(f"informes/compliance-ens-{fecha}.md").write_text(
        informe, encoding="utf-8"
    )

    # 5. Resumen técnico parcial, sin dictamen normativo
    review_count = sum(r.get("status")=="REVIEW" for r in resultados.values())
    incomplete_count = sum(r.get("status")=="INCOMPLETE" for r in resultados.values())
    return {"status":"INCOMPLETE" if incomplete_count else "REVIEW",
            "coverage":"partial","mapped_controls":len(resultados),
            "review_count":review_count,"incomplete_count":incomplete_count,
            "report":informe}
