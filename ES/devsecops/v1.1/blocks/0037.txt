# .github/workflows/security-pipeline.yml (ampliación SCA)
jobs:
  sca-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Instalar Syft
        uses: anchore/sbom-action/download-syft@v0
        with:
          syft-version: v1.5.0

      - name: Generar SBOM (CycloneDX)
        run: syft dir:. -o cyclonedx-json --file sbom.cdx.json

      - name: Almacenar SBOM como artefacto
        uses: actions/upload-artifact@v4
        with:
          name: sbom-cyclonedx
          path: sbom.cdx.json
          retention-days: 90

      - name: Instalar Grype
        uses: anchore/scan-action/download-grype@v4
        with:
          grype-version: v0.82.0

      - name: Escanear vulnerabilidades
        id: grype-scan
        run: |
          grype sbom:sbom.cdx.json -o json --file grype-results.json
          grype sbom:sbom.cdx.json -o table --fail-on critical

      - name: Triaje inteligente de hallazgos SCA
        if: always()
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: python scripts/triage_sca.py grype-results.json sbom.cdx.json
