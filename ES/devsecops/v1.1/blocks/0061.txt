from illustrative_host import Agent, tool
import subprocess
import json

@tool
def get_current_dockerfile(path: str = "Dockerfile") -> str:
    """Lee el Dockerfile actual del repositorio."""
    with open(path) as f:
        return f.read()

@tool
def scan_dockerfile(content: str) -> dict:
    """Escanea un Dockerfile con Trivy config mode."""
    import tempfile
    directory = tempfile.TemporaryDirectory()
    tmp_path = directory.name + "/Dockerfile.candidate"
    with open(tmp_path, "w") as f:
        f.write(content)
    result = subprocess.run(
        ["trivy", "config", "--format", "json", tmp_path],
        capture_output=True, text=True, timeout=30
    )
    if result.returncode != 0:
        return {"status":"INCOMPLETE", "reason":"Trivy config execution failed"}
    parsed = json.loads(result.stdout)
    if not isinstance(parsed,dict) or not isinstance(parsed.get("Results"),list):
        return {"status":"INCOMPLETE", "reason":"Invalid Trivy schema"}
    return {"status":"REVIEW", "coverage":"partial", "config_scan":parsed}
