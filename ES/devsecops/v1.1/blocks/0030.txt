import os
import argparse
import json
import sys

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True, help="Fichero JSON de Semgrep")
    parser.add_argument("--repo", required=True, help="owner/repo")
    parser.add_argument("--pr", required=True, type=int, help="Número del PR")
    args = parser.parse_args()

    with open(args.input) as stream:
        data = json.load(stream)
    if not isinstance(data,dict) or not isinstance(data.get("results"),list) or data.get("errors"):
        raise ValueError("INCOMPLETE: scanner schema or errors")
    findings = data["results"]
    retained = triaje_escalonado(findings)
    if len(retained) != len(findings):
        raise ValueError("INCOMPLETE: lost original findings")
    print(json.dumps({"status":"REVIEW", "coverage":"partial", "findings":retained}))
    sys.exit(2)
