# runtime_analyzer/sbom_correlator.py
"""
Correlaciona alertas de Falco con vulnerabilidades
conocidas del SBOM para enriquecer el contexto de análisis.
"""
import json
from pathlib import Path


def load_sbom(container_image: str) -> dict:
    """Carga el SBOM CycloneDX generado por Syft."""
    import hashlib
    if not isinstance(container_image,str) or "@sha256:" not in container_image:
        raise ValueError("INCOMPLETE: image digest identity required")
    sbom_path = Path("sboms") / (hashlib.sha256(container_image.encode()).hexdigest()+".cdx.json")
    if not sbom_path.exists():
        raise ValueError("INCOMPLETE: SBOM artifact missing")
    parsed = json.loads(sbom_path.read_text())
    if not isinstance(parsed,dict) or not isinstance(parsed.get("components"),list) or not isinstance(parsed.get("vulnerabilities"),list):
        raise ValueError("INCOMPLETE: SBOM coverage or schema")
    return parsed


def find_relevant_cves(sbom: dict, alert_type: str) -> list[dict]:
    records=sbom.get("vulnerabilities") if isinstance(sbom,dict) else None
    if not isinstance(records,list):raise ValueError("INCOMPLETE: missing vulnerability coverage")
    mapping={"network":{918,200,94},"filesystem":{22,73},"process":{78,269}}
    candidates=[]
    for record in records:
        if not isinstance(record,dict) or not isinstance(record.get("id"),str):
            raise ValueError("INCOMPLETE: invalid vulnerability identity")
        cwes,affects,ratings=record.get("cwes",[]),record.get("affects",[]),record.get("ratings",[])
        if not isinstance(cwes,list) or any(type(c) is not int or c<=0 for c in cwes):
            raise ValueError("INCOMPLETE: CycloneDX CWEs must be positive integers")
        if not isinstance(affects,list) or any(not isinstance(a,dict) or not isinstance(a.get("ref"),str) or not a["ref"] for a in affects):
            raise ValueError("INCOMPLETE: invalid affected references")
        if not isinstance(ratings,list) or any(not isinstance(a,dict) or not isinstance(a.get("severity","unknown"),str) for a in ratings):
            raise ValueError("INCOMPLETE: invalid ratings")
        description=record.get("description","")
        if not isinstance(description,str):raise ValueError("INCOMPLETE: invalid description")
        if set(cwes).intersection(mapping.get(alert_type,set())):
            candidates.append({"id":record["id"],"severity":ratings[0].get("severity","unknown") if ratings else "unknown",
                               "component":", ".join(a["ref"] for a in affects) or "unknown",
                               "description":description,"status":"REVIEW","coverage":"partial"})
    return candidates



def enrich_alert_context(
    container_name: str,
    alert_tags: list[str]
) -> str:
    """Genera contexto adicional para el agente Claude."""
    # Determinar tipo de alerta desde los tags de Falco
    alert_type = "unknown"
    if "network" in alert_tags or "mitre_exfiltration" in alert_tags:
        alert_type = "network"
    elif "filesystem" in alert_tags or "mitre_persistence" in alert_tags:
        alert_type = "filesystem"
    elif "mitre_execution" in alert_tags:
        alert_type = "process"

    sbom = load_sbom(image)
    cves = find_relevant_cves(sbom, alert_type)
