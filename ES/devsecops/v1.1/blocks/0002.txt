# .github/workflows/security-pipeline.yml
name: Security Pipeline
on:
  pull_request:
    branches: [main, develop]

permissions:
  contents: read
  pull-requests: write  # Para comentar hallazgos en el PR

jobs:
  sast-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Ejecutar Semgrep SAST
        id: semgrep
        uses: returntocorp/semgrep-action@v1
        with:
          config: >-
            p/security-audit
            p/owasp-top-ten
            p/python
          generateSarif: true

      - name: Guardar resultados SARIF
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: semgrep-results
          path: semgrep.sarif

      - name: Enviar hallazgos a triaje inteligente
        if: always()
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: python scripts/triage_findings.py semgrep.sarif
