def read_file(file_path: str,
              start_line: int = None,
              end_line: int = None) -> dict:
    """Lee contenido de un fichero del repo."""
    path = PurePosixPath(file_path)
    if (not isinstance(file_path,str) or path.is_absolute() or ".." in path.parts
            or "\\" in file_path or not re.fullmatch(r"[0-9a-fA-F]{40}|[0-9a-fA-F]{64}",TARGET_COMMIT_SHA)):
        return {"status":"INCOMPLETE","reason":"Relative path and immutable reviewed commit required"}
    if any(value is not None and (type(value) is not int or value < 1) for value in (start_line,end_line)):
        return {"status":"INCOMPLETE","reason":"Invalid line range"}
    if (start_line is None) != (end_line is None) or (start_line is not None and end_line < start_line):
        return {"status":"INCOMPLETE","reason":"Incomplete line range"}
    try:
        content = repo.get_contents(file_path, ref=TARGET_COMMIT_SHA)
        decoded = content.decoded_content.decode("utf-8")
        lines = decoded.split("\n")

        if start_line and end_line:
            lines = lines[start_line - 1:end_line]

        return {
            "content": "\n".join(lines),
            "total_lines": len(decoded.split("\n")),
            "sha": content.sha
        }
    except GithubException as e:
        return {"error": f"No se pudo leer {file_path}: {e}"}
