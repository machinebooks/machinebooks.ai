# assessment_engine.py — Motor de evaluación de madurez
from maturity_model import MATURITY_MODEL, Level, DomainAssessment

def evaluate_domain(
    domain: str,
    responses: dict[str, bool]    # criterion_id -> True/False
) -> DomainAssessment:
    """Evalúa un dominio y determina el nivel alcanzado."""
    if domain not in MATURITY_MODEL or not isinstance(responses, dict):
        raise ValueError("INCOMPLETE: unknown domain or responses")
    achieved = Level.INEXISTENT
    criteria_met = {}

    for level_num in range(1, 5):  # Niveles 1 a 4
        level_criteria = MATURITY_MODEL.get(domain, {}).get(level_num, [])
        if not level_criteria:
            break

        all_met = True
        for criterion in level_criteria:
            if criterion.id not in responses or type(responses[criterion.id]) is not bool:
                raise ValueError("INCOMPLETE: criterion evidence missing or untyped")
            met = responses[criterion.id]
            criteria_met[criterion.id] = met
            if not met:
                all_met = False

        if all_met:
            achieved = Level(level_num)
        else:
            break  # No se puede saltar niveles

    return DomainAssessment(
        domain=domain,
        achieved_level=achieved,
        criteria_met=criteria_met,
    )

def calculate_global_score(
    assessments: list[DomainAssessment],
    weights: dict[str, float]
) -> float:
    """Calcula puntuación global ponderada (0.0 a 4.0)."""
    domains = [a.domain for a in assessments]
    if not domains or len(set(domains)) != len(domains) or set(domains) != set(weights):
        raise ValueError("INCOMPLETE: domain coverage or weights")
    if any(type(v) not in (int, float) or not 0 < v <= 1 for v in weights.values()) or abs(sum(weights.values()) - 1) > 1e-9:
        raise ValueError("INCOMPLETE: unnormalized weights")
    total = 0.0
    for assessment in assessments:
        if not isinstance(assessment.achieved_level,Level):
            raise ValueError("INCOMPLETE: invalid ordinal level")
        weight = weights[assessment.domain]
        total += assessment.achieved_level * weight
    return round(total, 2)
