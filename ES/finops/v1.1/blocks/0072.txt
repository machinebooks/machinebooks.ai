from decimal import Decimal

def calculate_cost_with_cache(model, input_tokens, output_tokens,
        cache_read_tokens=0, cache_creation_tokens=0, is_batch=False):
    if type(is_batch) is not bool:
        raise ValueError("batch flag")
    counts = (input_tokens, output_tokens, cache_read_tokens, cache_creation_tokens)
    if any(type(v) is not int or not 0 <= v <= 2147483647 for v in counts):
        raise ValueError("native disjoint integer counts required")
    rates = {"claude-sonnet-4-6":(Decimal(3),Decimal(15)),
             "claude-haiku-4-5":(Decimal(1),Decimal(5)),
             "claude-opus-4-6":(Decimal(5),Decimal(25))}
    if model not in rates:
        return None
    inp,out = rates[model]
    million = Decimal(1000000)
    actual = (input_tokens*inp + output_tokens*out + cache_read_tokens*inp*Decimal('.1')
              + cache_creation_tokens*inp*Decimal('1.25'))/million
    actual *= Decimal('.5') if is_batch else Decimal(1)
    baseline = ((input_tokens+cache_read_tokens+cache_creation_tokens)*inp + output_tokens*out)/million
    return {"reference_cost_usd":actual, "counterfactual_standard_usd":baseline,
            "difference_usd":baseline-actual, "invoice_reconciled":False,
            "price_reference":"2026-10-04-standard-global-5m"}
