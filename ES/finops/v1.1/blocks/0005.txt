# Append-only workflow decision, priced from owned attempt ledger, including rejects.
from sqlalchemy import Column, String, Numeric, Boolean, DateTime, ForeignKey, JSON, UniqueConstraint
from database import Base

class TaskCompletionLog(Base):
    __tablename__ = "task_completion_log"
    id = Column(String(36), primary_key=True)
    tenant_id = Column(String(36), nullable=False)
    workflow_id = Column(String(36), nullable=False)
    decision_version = Column(String(100), nullable=False)
    baseline_config_id = Column(String(36), ForeignKey("human_baseline_config.id"), nullable=True)
    baseline_snapshot = Column(JSON, nullable=True)
    accepted = Column(Boolean, nullable=False)
    known_attempt_cost = Column(Numeric(24, 8), nullable=True)
    cost_currency = Column(String(3), nullable=False)
    fx_version = Column(String(100), nullable=True)
    captured_capacity_value = Column(Numeric(24, 8), nullable=True)
    completed_at = Column(DateTime(timezone=True), nullable=False)
    __table_args__ = (UniqueConstraint("tenant_id", "workflow_id", "decision_version"),)

class TaskAttemptLink(Base):
    __tablename__ = "task_attempt_links"
    task_id = Column(String(36), ForeignKey("task_completion_log.id"), primary_key=True)
    usage_id = Column(String(36), ForeignKey("llm_usage_logs.id"), primary_key=True)
