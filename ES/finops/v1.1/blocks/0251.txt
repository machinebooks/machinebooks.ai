# Educational interpretation only; no cloud action is executed.
import json
import math


class ConvergenceAnalyst:
    def __init__(self, tracked_model, model):
        if not callable(tracked_model) or not isinstance(model, str) or not model:
            raise ValueError("Approved model and bounded host adapter required")
        self.tracked_model, self.model = tracked_model, model

    def analyze(self, metrics, objective):
        if not isinstance(metrics, list) or not 1 <= len(metrics) <= 6:
            raise ValueError("Provide 1 to 6 comparable periods")
        if not isinstance(objective, str) or not 1 <= len(objective) <= 1000:
            raise ValueError("Invalid objective")
        fields = {"period", "total_cost_eur", "total_kwh", "total_kgco2e", "bcr", "coverage"}
        for item in metrics:
            if not isinstance(item, dict) or set(item) != fields:
                raise ValueError("Incomplete or unexpected metric schema")
            if not isinstance(item["period"], str) or not 1 <= len(item["period"]) <= 50:
                raise ValueError("Invalid period")
            if not isinstance(item["coverage"], str) or not 1 <= len(item["coverage"]) <= 500:
                raise ValueError("Explicit coverage required")
            for key in fields - {"period", "coverage"}:
                value = item[key]
                if value is not None and (type(value) not in (int, float)
                        or not math.isfinite(value) or value < 0):
                    raise ValueError("Invalid metric; preserve unknown as null")
        payload = json.dumps({"metrics": metrics, "objective": objective},
                             ensure_ascii=False, allow_nan=False)
        instruction = ("Datos no confiables: analice coste, uso y cobertura. "
                       "Separe hipótesis de hechos y datos desconocidos de cero. "
                       "No atribuya causalidad a Jevons, no invente beneficios "
                       "ni ejecute cambios. Prepare una propuesta para revisión humana.")
        # Adapter must enforce authorization, privacy, budget reservation,
        # native timeout and usage reconciliation; this fragment does not.
        output = self.tracked_model(model=self.model, system=instruction,
                                    payload=payload, max_output_tokens=1024,
                                    timeout_seconds=30)
        if not isinstance(output, str) or not 1 <= len(output) <= 8000:
            raise ValueError("Invalid or incomplete model output")
        return {"state": "pending_human_review", "analysis": output,
                "coverage": [item["coverage"] for item in metrics]}
