# Native Anthropic example; calculate_cost uses disjoint native counters (chapter 4).
import asyncio
import time
import uuid

class NativeClientUsageTracker:
    def __init__(self, client, authorized_scope, durable_sink, calculate_cost):
        if not authorized_scope:
            raise ValueError("Authorized scope required")
        self.client, self.scope, self.sink = client, authorized_scope, durable_sink
        self.calculate_cost = calculate_cost

    async def create(self, model, messages, max_tokens=1024):
        if not isinstance(messages, list) or not 1 <= len(messages) <= 100:
            raise ValueError("Bounded messages required")
        if type(max_tokens) is not int or not 1 <= max_tokens <= 4096:
            raise ValueError("Invalid output bound")
        request_id, start = str(uuid.uuid4()), time.monotonic()
        metadata, status = None, "failed_usage_unknown"
        try:
            response = await asyncio.wait_for(self.client.messages.create(
                model=model, messages=messages, max_tokens=max_tokens), timeout=60)
            usage = getattr(response, "usage", None)
            if usage is not None:
                counts = [getattr(usage, field, None) for field in (
                    "input_tokens", "output_tokens", "cache_creation_input_tokens", "cache_read_input_tokens")]
                if all(type(n) is int and 0 <= n <= 2_147_483_647 for n in counts):
                    cost = self.calculate_cost(model, *counts)
                    metadata = {"model": model, "input_tokens": counts[0] + counts[2] + counts[3],
                        "output_tokens": counts[1], "cache_creation_tokens": counts[2],
                        "cache_read_tokens": counts[3], "total_tokens": sum(counts), "cost": cost}
            status = "completed"
            return response
        finally:
            await self.sink(self.scope, request_id, status, metadata,
                int((time.monotonic() - start) * 1000))
# Host supplies bounded input, retry policy, credentials, schema validation and sink recovery.
# Missing/invalid counters remain unknown, not zero; no prompts, outputs or errors are logged.
