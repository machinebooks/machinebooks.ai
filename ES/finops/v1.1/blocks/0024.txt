# Canonical chapter-4 educational schema; chapter-3 views use an explicit adapter.
from sqlalchemy import Column, String, Integer, Numeric, DateTime, Index
from datetime import datetime, timezone
import uuid
from .base import Base

class LLMUsageLog(Base):
    __tablename__ = "llm_usage_logs"
    id = Column(String(36), primary_key=True, default=lambda: str(uuid.uuid4()))
    timestamp = Column(DateTime, nullable=False,
        default=lambda: datetime.now(timezone.utc).replace(tzinfo=None))
    tenant_id = Column(String(64), nullable=False)
    request_id = Column(String(64), nullable=False)
    calling_app = Column(String(128), nullable=False)
    service_name = Column(String(128), nullable=False)
    user_id = Column(String(64), nullable=True)
    prompt_key = Column(String(128), nullable=True)
    model = Column(String(128), nullable=False)
    provider = Column(String(64), nullable=False)
    input_tokens = Column(Integer, nullable=True)
    output_tokens = Column(Integer, nullable=True)
    cache_creation_tokens = Column(Integer, nullable=True)
    cache_read_tokens = Column(Integer, nullable=True)
    total_tokens = Column(Integer, nullable=True)
    total_cost_usd = Column(Numeric(24, 12), nullable=True)
    tariff_version = Column(String(64), nullable=True)
    status = Column(String(32), nullable=False)
    latency_ms = Column(Integer, nullable=True)
    environment = Column(String(32), nullable=False)
    __table_args__ = (Index("idx_scope_service_ts", "tenant_id", "service_name", "timestamp"),)
# input_tokens is normalized total input (including cache partitions); NULL is unknown.
# Host validates counters, sets MySQL/session UTC, uses idempotent attempt identity,
# and durably persists tariff provenance. No raw prompts/responses/errors are stored.
