# Append-only educational tariff rows; no deployed migration is shown.
from sqlalchemy import Column, Integer, String, Numeric, DateTime, UniqueConstraint
from datetime import datetime, timezone
from app.database import Base

class LLMModelPricing(Base):
    __tablename__ = "llm_model_pricing"
    id = Column(Integer, primary_key=True)
    provider = Column(String(50), nullable=False)
    model_id = Column(String(100), nullable=False)
    model_display_name = Column(String(200), nullable=True)
    model_type = Column(String(30), nullable=False)
    currency = Column(String(3), nullable=False)
    tariff_scope = Column(String(128), nullable=False)
    version = Column(String(64), nullable=False)
    input_price_per_1m = Column(Numeric(20, 10), nullable=False)
    output_price_per_1m = Column(Numeric(20, 10), nullable=False)
    cached_input_price_per_1m = Column(Numeric(20, 10), nullable=True)
    cache_write_price_per_1m = Column(Numeric(20, 10), nullable=True)
    effective_date = Column(DateTime, nullable=False)
    source_url = Column(String(1024), nullable=False)
    created_at = Column(DateTime, nullable=False,
        default=lambda: datetime.now(timezone.utc).replace(tzinfo=None))
    __table_args__ = (UniqueConstraint("provider", "model_id", "currency",
        "tariff_scope", "version"),)
# Host validates non-negative Decimal prices, UTC dates and non-overlapping validity.
# Keep previous versions; correction creates a new row, never rewrites priced history.
