createSession(workspacePath, parentMessages): CoordinatorSession {
  const session: CoordinatorSession = {
    id: crypto.randomUUID(),
    agents: new Map(),
    phase: 'research',
    workspacePath,
    parentMessages,
  };
  this.sessions.set(session.id, session);
  return session;
}
