@startuml
!theme plain
skinparam backgroundColor transparent
skinparam classAttributeIconSize 0
skinparam classFontStyle bold

interface ContentBlock <<union>> {
  type: 'text' | 'thinking' | 'tool_use' | 'tool_result'
}

class TextBlock {
  type: 'text'
  text: string
}

class ThinkingBlock {
  type: 'thinking'
  thinking: string
}

class ToolUseBlock {
  type: 'tool_use'
  id: string
  name: string
  input: Record<string, unknown>
}

class ToolResultBlock {
  type: 'tool_result'
  tool_use_id: string
  content: string
  is_error?: boolean
}

class Message {
  role: 'user' | 'assistant' | 'system' | 'tool'
  content: string | ContentBlock[]
}

class AssistantMessage {
  role: 'assistant'
  content: ContentBlock[]
  usage?: { prompt_tokens, completion_tokens }
}

ContentBlock <|.. TextBlock
ContentBlock <|.. ThinkingBlock
ContentBlock <|.. ToolUseBlock
ContentBlock <|.. ToolResultBlock
Message <|-- AssistantMessage
Message "1" *-- "many" ContentBlock : content

note right of ToolUseBlock
  Generado por el modelo.
  El `id` lo asigna el provider
  y persiste para empatarlo con
  el ToolResultBlock correspondiente.
end note

note right of ToolResultBlock
  Lo genera el código tras
  ejecutar la tool. El
  `tool_use_id` debe coincidir
  con el ToolUseBlock anterior.
end note

note bottom of Message
  Proyección histórica simplificada.
  No conserva todos los bloques nativos
  ni las firmas de thinking.
  El adaptador debe preservarlos aparte
  o rechazar una continuación incompatible.
end note
@enduml
