if (llmProvider === 'azure') {
  const { streamAzureCall } = await import('../providers/azure.js');
  const deployment = serverState.azureDeployment || model || llmModel;
  const result = await streamAzureCall(
    {
      endpoint: llmBaseUrl,
      deployment,
      apiVersion: serverState.azureApiVersion,
      apiKey: llmApiKey,
    },
    callMessages,
    withTools ? AGENT_TOOLS : undefined,
    callbacks,
    { maxTokens: 8192 },
  );
  return result;
}
