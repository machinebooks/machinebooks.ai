const llmApiKey = serverState.llmApiKey;

if (llmProvider === 'anthropic') {
  // Dynamic import for the selected provider.
  const { streamAnthropicCall } = await import('../providers/anthropic.js');
  const result = await streamAnthropicCall(
    llmApiKey,
    llmModel,
    callMessages,
    withTools ? AGENT_TOOLS : undefined,
    {
      onContent:    (text) => { sendSSE(res, 'content', { text }); },
      onThinking:   (text) => { sendSSE(res, 'thinking', { text }); },
      onToolCall:   (idx, id, name) => { sendSSE(res, 'status', { state: 'proposed', message: `Tool: ${name}` }); },
      onToolCallDone: () => {},
      onUsage:      (inT, outT) => { lastObservedUsage = { inputTokens: inT, outputTokens: outT }; },
      onError:      (err) => { sendSSE(res, 'error', { message: err }); },
    },
    { maxTokens: 8192, thinking: true, thinkingBudget: 4096 },
  );
  return result;
}
