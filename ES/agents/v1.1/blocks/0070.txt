// Educational read schema; the host authorizes the effective path separately.
inputSchema: {
  type: 'object',
  properties: {
    file_path: { type: 'string', minLength: 1, description: 'Path within the host-authorized scope' },
    offset: { type: 'integer', minimum: 0, description: 'Zero-based first line' },
    limit: { type: 'integer', minimum: 1, maximum: 1000, description: 'Maximum lines to return' },
  },
  required: ['file_path'],
  additionalProperties: false,
},
