const bundle = new VmBundleManager();
let status = bundle.getStatus();
if (!status.ready) {
  const copied = await bundle.provisionFromResources();
  status = bundle.getStatus();
  if (!copied || !status.ready) {
    throw new Error('Required VM bundle is unavailable');
  }
}
