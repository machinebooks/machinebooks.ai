-- Runner preconditions: authorized schema; column and index absent.

SET NAMES utf8mb4;

ALTER TABLE existing_table
    ADD COLUMN new_col VARCHAR(100) NULL;
UPDATE existing_table SET new_col = 'default' WHERE new_col IS NULL;
CREATE INDEX idx_table_col ON existing_table (new_col);

-- Separate trigger example for the mysql client (not driver SQL):
DELIMITER $$
