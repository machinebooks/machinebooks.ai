    # Cap de longitud: limita backtracking catastrophic en regex
    MAX_SCAN_CHARS = 50_000
    if len(text) > MAX_SCAN_CHARS:
        text = text[:MAX_SCAN_CHARS]

    # Length alone does not bound regex runtime.
    ranking = {"public": 0, "internal": 1, "confidential": 2, "secret": 3}
    level = "public"
    indicators = []
    # Partial example: only the secret rules are shown.
    for regex, tag in _SECRET_RE:
        if regex.search(text):
            if ranking["secret"] > ranking[level]:
                level = "secret"
            indicators.append(tag)
