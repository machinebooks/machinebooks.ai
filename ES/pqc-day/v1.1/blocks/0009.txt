def run_crypto_analysis_agent(
    repo_path: str,
    organization_id: int,
    max_iterations: int = 15
) -> dict:
    """Ejecuta el agente autónomo de análisis criptográfico.

    El agente decide qué ficheros analizar, en qué orden,
    y cuándo tiene suficiente información para generar un informe.
    Máximo 15 iteraciones para controlar consumo de tokens.
    """
    conversation = [{
        "role": "user",
        "content": (
            f"Analiza el repositorio en '{repo_path}' buscando "
            "algoritmos criptográficos vulnerables ante computación "
            "cuántica. Prioriza ficheros de configuración TLS, "
            "módulos de autenticación y cualquier uso de RSA, ECDSA, "
            "ECDH, DH o DSA. Registra cada hallazgo con su contexto "
            "de uso y recomendación de migración PQC."
        )
    }]

    findings_count = 0

    for iteration in range(max_iterations):
        response = client.messages.create(
            model="claude-sonnet-4-6",
            max_tokens=4096,
            tools=tools,
            messages=conversation,
            system=(
                "Eres un agente experto en análisis criptográfico "
                "post-cuántico. Tu objetivo es encontrar y clasificar "
                "todos los usos de criptografía vulnerable en el "
                "repositorio. Sé metódico: primero lista ficheros, "
                "luego analiza los más críticos, y registra cada "
                "hallazgo con contexto preciso."
            )
        )

        # Si el agente ha terminado su análisis
        if response.stop_reason == "end_turn":
            break

        # Añadir la respuesta del asistente (una sola vez)
        conversation.append({
            "role": "assistant",
            "content": response.content
        })

        # Procesar llamadas a herramientas y recopilar resultados
        tool_results = []
        for block in response.content:
            if block.type == "tool_use":
                result = execute_tool(
                    block.name,
                    block.input,
                    organization_id
                )
                if block.name == "register_finding":
                    findings_count += 1
                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": str(result)
                })

        # Devolver todos los resultados de herramientas
        if tool_results:
            conversation.append({
                "role": "user",
                "content": tool_results
            })

    return {
        "iterations_used": iteration + 1,
        "findings_registered": findings_count,
        "completed": response.stop_reason == "end_turn"
    }
