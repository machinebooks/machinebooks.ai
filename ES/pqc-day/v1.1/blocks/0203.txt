def to_dict(self):
    return {'id': self.id, 'organization_id': self.organization_id,
            'user_id': self.user_id,
            'actor_status': 'recorded_user' if self.user_id is not None else 'unavailable_or_system',
            'action': self.action, 'entity_type': self.entity_type,
            'entity_id': self.entity_id, 'details': self.details,
            'created_at': None if self.created_at is None else self.created_at.isoformat() + 'Z'}
# A deleted or unavailable user is not relabelled as a known System actor.
# Actor snapshots/system identity require a separately reviewed provenance schema.
