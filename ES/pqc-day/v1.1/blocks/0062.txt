FINAL_FIPS_ALGORITHMS = {
    "key_encapsulation": {"ML-KEM": "FIPS 203"},
    "digital_signatures": {"ML-DSA": "FIPS 204", "SLH-DSA": "FIPS 205"},
}
CLASSICAL_SHOR_CANDIDATES = {"RSA", "ECDSA", "ECDH", "Ed25519", "X25519"}
HISTORICAL_OR_PENDING_NAMES = {"Kyber", "Dilithium", "SPHINCS+", "Falcon", "FN-DSA", "HQC"}

def classify_algorithm(name, function):
    if name in FINAL_FIPS_ALGORITHMS.get(function, {}):
        return {"classification": "standardized_candidate", "severity": None,
                "requires": ["parameters", "protocol", "implementation", "validation_scope"]}
    if name in CLASSICAL_SHOR_CANDIDATES:
        return {"classification": "shor_susceptible", "severity": None,
                "requires": ["function", "exposure", "key_lifetime", "data_lifetime"]}
    if name in HISTORICAL_OR_PENDING_NAMES:
        return {"classification": "version_and_status_required", "severity": None}
    # AES and hashes require their own property, parameter and usage assessment.
    return {"classification": "not_assessed", "severity": None}
