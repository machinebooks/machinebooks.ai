class AnthropicProvider(BaseAIProvider):
    """Anthropic provider (Claude)"""

    def __init__(self, config: Dict, analysis_type: str = 'pqc'):
        super().__init__(config, analysis_type)
        self.api_key = config.get('api_key') or os.getenv('ANTHROPIC_API_KEY')
        self.base_url = 'https://api.anthropic.com/v1'
        self.model = config.get('model', 'claude-sonnet-4-6')

    def analyze_code(self, code: str, filename: str,
                     context: Dict = None) -> Dict:
        if (not isinstance(context, dict)
                or context.get('source_export_authorized') is not True):
            return {'status': 'not_authorized', 'findings': None, 'risk_score': None}
        if not isinstance(code, str) or len(code) > 100_000:
            return {'status': 'invalid_input', 'findings': None, 'risk_score': None}
        language = context.get('language') if context else None

        # Select prompt based on analysis type
        if self.analysis_type == 'owasp':
            prompt = self._build_owasp_prompt(code, filename, language)
        else:
            prompt = self._build_pqc_prompt(code, filename, language)

        response = requests.post(
            f"{self.base_url}/messages",
            headers={
                'x-api-key': self.api_key,
                'Content-Type': 'application/json',
                'anthropic-version': '2023-06-01'
            },
            json={
                'model': self.model,
                'max_tokens': 4096,
                'messages': [{'role': 'user', 'content': prompt}]
            },
            timeout=self.timeout
        )

        if response.status_code == 200:
            data = response.json()
            if data.get('stop_reason') != 'end_turn':
                return {'status': 'incomplete_response', 'findings': None, 'risk_score': None}
            content = ''.join(block.get('text', '') for block in data.get('content', [])
                              if isinstance(block, dict) and block.get('type') == 'text')
            result = parse_ai_response(content, self.analysis_type, code.count('\n') + 1)
            # Record tokens for governance (ch. 14)
            usage = data.get('usage', {})
            counts = [usage.get('input_tokens'), usage.get('output_tokens')]
            result['tokens_used'] = sum(counts) if all(
                type(value) is int and value >= 0 for value in counts) else None
            return result
        else:
            return {'status': 'error', 'http_status': response.status_code,
                    'findings': None, 'risk_score': None, 'tokens_used': None}

# Imports and source export approval are application dependencies. The server
# supplies context; request data cannot authorize itself. No response body is
# logged. The facade normalizes transport exceptions; timeout is not a total
# deadline and an execution owner must impose cancellation/resource budgets.
