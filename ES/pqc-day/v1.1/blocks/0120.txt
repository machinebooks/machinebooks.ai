from flask_jwt_extended import get_current_user
import time
from flask import current_app

def test_provider(provider_id):
    user = get_current_user()
    if not user or not user.is_active or not user.has_permission('manage_ai'):
        return jsonify({'error': 'Access denied'}), 403
    policy = current_app.extensions.get('ai_provider_authorization')

        return jsonify({'error': 'Provider authorization unavailable'}), 503
