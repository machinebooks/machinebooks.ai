class DoraReportGenerator:
    """Legacy class name; scores are didactic risk bands, not DORA grades."""

    def __init__(self, assessment: dict, entity: dict):
        controls = assessment.get("controls", [])
        self.assessment = {"controls": list(controls),
                           "index": calculate_control_index(controls)}
        self.entity = dict(entity)
        self.styles = getSampleStyleSheet()
        self.styles.add(ParagraphStyle(
            "SectionHeader", parent=self.styles["Heading2"],
            fontSize=14, spaceBefore=15, spaceAfter=10,
        ))

    def generate(self, output_path: str):
        """Require the omitted builders before creating any output."""
        required = ("_build_cover", "_build_executive_summary",
                    "_build_findings_section", "_build_recommendations")
        missing = [name for name in required
                   if not callable(getattr(self, name, None))]
        if missing:
            raise NotImplementedError("Implement section builders: " + ", ".join(missing))
        doc = SimpleDocTemplate(
            output_path, pagesize=A4,
            leftMargin=2.5*cm, rightMargin=2.5*cm,
            topMargin=2*cm, bottomMargin=2*cm,
        )
        elements = self._build_cover() + [PageBreak()]
        elements += self._build_executive_summary() + [PageBreak()]
        elements += self._build_scoring_section()
        elements += self._build_findings_section()
        elements += self._build_regulatory_timeline()
        elements += self._build_recommendations()
        doc.build(elements)

    def _build_scoring_section(self) -> list:
        index = self.assessment["index"]
        rows = [["Control", "State", "Weight"]]
        rows.extend([[str(c.get("control_id", "Unknown")), c["state"], str(c["weight"])]
                     for c in self.assessment["controls"]])
        rows = [[Paragraph(escape(str(cell)), self.styles["Normal"]) for cell in row]
                for row in rows]
        table = Table(rows, colWidths=[6*cm, 6*cm, 4*cm], repeatRows=1)
        table.setStyle(TableStyle([("GRID", (0, 0), (-1, -1), 0.5, colors.grey),
                                   ("VALIGN", (0, 0), (-1, -1), "TOP")]))
        value = "Unknown" if index["score"] is None else str(index["score"])
        note = ("Internal assessed-control index: " + value
                + "; status: " + index["status"]
                + ". Unknown controls are disclosed separately. No legal compliance decision.")
        return [Paragraph("Reviewed internal control evidence", self.styles["SectionHeader"]),
                table, Spacer(1, 15), Paragraph(escape(note), self.styles["Normal"])]

    def _build_regulatory_timeline(self) -> list:
        data = [[LABELS["date"], LABELS["milestone"],
                 LABELS["source"], LABELS["nature"]], *MILESTONES]
        data = [[Paragraph(escape(str(cell)), self.styles["Normal"])
                 for cell in row] for row in data]
        table = Table(data, colWidths=[2.4*cm, 4.5*cm, 2.6*cm, 6.5*cm], repeatRows=1)
        table.setStyle(TableStyle([
            ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#e8eaf6")),
            ("GRID", (0, 0), (-1, -1), 0.5, colors.grey),
            ("VALIGN", (0, 0), (-1, -1), "TOP"),
        ]))
        return [Paragraph(LABELS["timeline_title"], self.styles["SectionHeader"]), table]

    @staticmethod
    def _risk_label(score: float) -> str:
        if score >= 75:
            return LABELS["critical"]
        if score >= 50:
            return LABELS["high"]
        if score >= 25:
            return LABELS["medium"]
        return LABELS["low"]
