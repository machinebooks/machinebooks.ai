import json
from typing import List, Dict

class RAGSearchService:
    """Bounded retrieval from an authorized organization corpus."""
    def __init__(self, *, scope, retriever, reranker=None):
        if not isinstance(scope, dict) or scope.get('authorized') is not True:
            raise PermissionError('Authorized retrieval scope required')
        self.scope = scope
        self.retriever = retriever
        self.reranker = reranker

    def search(self, query, collection_types=None, max_chunks=5):
        if not isinstance(query, str) or not query.strip() or len(query) > 4000:
            raise ValueError('Bounded nonempty query required')
        if type(max_chunks) is not int or not 1 <= max_chunks <= 12:
            raise ValueError('max_chunks must be 1 to 12')
        kinds = collection_types or ['documentation', 'framework']
        if (not isinstance(kinds, list) or len(kinds) > 4
                or any(k not in {'documentation', 'framework', 'policy', 'custom'} for k in kinds)):
            raise ValueError('Invalid collection types')
        # Retriever verifies collection/document organization, active state and
        # access policy, applies a bounded indexed search and retains exclusions.
        candidates = self.retriever.search_authorized(
            self.scope, query=self._expand_crypto_synonyms(query),
            collection_types=kinds, limit=min(36, max_chunks * 3))
        if not isinstance(candidates, list) or len(candidates) > 36:
            raise ValueError('Invalid retriever contract')
        for item in candidates:
            if not isinstance(item, dict) or not isinstance(item.get('text'), str):
                raise ValueError('Invalid excerpt')
        if self.reranker is not None and len(candidates) > max_chunks:
            candidates = self._rerank_with_llm(query, candidates)
        return candidates[:max_chunks]

    def _expand_crypto_synonyms(self, query):
        aliases = {'Kyber': 'ML-KEM FIPS-203', 'Dilithium': 'ML-DSA FIPS-204',
                   'SPHINCS+': 'SLH-DSA FIPS-205'}
        return query + ''.join(' ' + value for name, value in aliases.items()
                               if name.lower() in query.lower())
        # Retrieval aliases do not classify pre-standard schemes as final FIPS.

    def _rerank_with_llm(self, query, candidates):
        if self.scope.get('source_export_authorized') is not True:
            return candidates
        # Adapter enforces configured provider, complete token/egress budgets,
        # no source logging and data-as-untrusted prompt construction.
        result = self.reranker.rank(self.scope, query, candidates)
        order = result.get('order') if isinstance(result, dict) else None
        if (not isinstance(order, list) or len(order) != len(candidates)
                or any(type(v) is not int for v in order)
                or sorted(order) != list(range(len(candidates)))):
            return candidates
        return [candidates[i] for i in order]
# The retriever/reranker implementations are omitted; no FULLTEXT, vector-store,
# freshness, truth, injection resistance or whole-corpus coverage is proved here.
