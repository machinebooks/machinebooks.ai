import tiktoken
import zlib


class RAGChunker:
    def __init__(self, chunk_size=1000, chunk_overlap=200):
        if not isinstance(chunk_size, int) or isinstance(chunk_size, bool):
            raise ValueError("chunk_size must be an integer")
        if not isinstance(chunk_overlap, int) or isinstance(chunk_overlap, bool):
            raise ValueError("chunk_overlap must be an integer")
        if chunk_size <= 0 or not 0 <= chunk_overlap < chunk_size:
            raise ValueError("require 0 <= chunk_overlap < chunk_size")
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.encoder = tiktoken.get_encoding("cl100k_base")

    def chunk_document(self, content, title):
        text = zlib.decompress(content).decode("utf-8")
        tokens = self.encoder.encode(text, disallowed_special=())
        raw = text.encode("utf-8")
        offsets = [0]
        for token in tokens:
            offsets.append(offsets[-1] + len(self.encoder.decode_single_token_bytes(token)))
        safe = {
            index for index, offset in enumerate(offsets)
            if offset == len(raw) or raw[offset] & 0xC0 != 0x80
        }
        chunks = []
        start = 0
        while start < len(tokens):
            end = min(start + self.chunk_size, len(tokens))
            while end > start and end not in safe:
                end -= 1
            if end == start:
                raise ValueError("chunk_size cannot contain the next Unicode character")
            chunks.append({
                "index": len(chunks),
                "text": raw[offsets[start]:offsets[end]].decode("utf-8", errors="strict"),
                "token_count": end - start,
                "source_title": title,
                "start_token": start,
                "end_token": end,
            })
            if end == len(tokens):
                break
            next_start = max(start + 1, end - self.chunk_overlap)
            while next_start < end and next_start not in safe:
                next_start += 1
            start = next_start
        return chunks
