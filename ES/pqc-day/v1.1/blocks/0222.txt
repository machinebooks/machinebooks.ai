def policy_recommendation(rule):
    if rule.get("function") not in {"signature", "key_establishment"}:
        raise ValueError("Function required")
    phase = rule.get("phase")
    if phase not in {"hybrid", "pure_pqc"}:
        raise ValueError("Unknown migration phase")
    target = rule.get("hybrid_target") if phase == "hybrid" else rule.get("target")
    if not target:
        raise ValueError("Missing target")
    return {"algorithm": target, "phase": phase, "scope": rule["scope"],
            "policy_version": rule["policy_version"]}
