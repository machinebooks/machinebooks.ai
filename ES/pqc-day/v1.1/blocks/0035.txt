from app.extensions import db
from datetime import datetime


class AuditLog(db.Model):
    """Audit record for user actions.
    Each row answers: who did what, when, from where
    and on what system entity."""
    __tablename__ = 'audit_logs'

    id = db.Column(db.Integer, primary_key=True)
    organization_id = db.Column(db.Integer, db.ForeignKey("organizations.id"), nullable=False)
    # Who: relationship with users table
    user_id = db.Column(
        db.Integer,
        db.ForeignKey('users.id', ondelete='SET NULL'),
        nullable=True  # Allows system actions without user
    )
    # What: action executed (login, create_client, run_scan, etc.)
    action = db.Column(db.String(100), nullable=False)
    # On what: entity type and its ID
    entity_type = db.Column(db.String(50))  # 'client', 'scan', 'report'
    entity_id = db.Column(db.Integer)
    # Additional context: free JSON for action-specific metadata
    details = db.Column(db.JSON)
    # From where: user's IP and browser
    ip_address = db.Column(db.String(45))   # IPv6 supported
    user_agent = db.Column(db.Text)
    # When
    created_at = db.Column(db.DateTime, default=datetime.utcnow)

    # Relationship to resolve user name in queries
    user = db.relationship(
        'User',
        backref=db.backref('audit_logs', lazy='dynamic'),
        foreign_keys=[user_id]
    )
    @staticmethod
    def log(user_id, action, entity_type=None, entity_id=None, details=None,
            ip_address=None, user_agent=None, *, organization_id):
        # IDs and action originate in an authorized server operation; caller owns
        # the transaction. Details are an approved metadata allowlist, not free JSON.
        if type(organization_id) is not int or organization_id <= 0:
            raise ValueError('Trusted organization required')
        if not isinstance(action, str) or not 1 <= len(action) <= 100:
            raise ValueError('Invalid action')
        if user_id is not None and (type(user_id) is not int or not User.query.filter_by(
                id=user_id, organization_id=organization_id, is_active=True).first()):
            raise ValueError('Actor does not belong to organization')
        allowed = {'organization_id', 'records_created', 'job_id', 'target_id'}
        data = details or {}
        if not isinstance(data, dict) or set(data) - allowed:
            raise ValueError('Unapproved audit metadata')
        if any(type(v) is not int or v < 0 for v in data.values()):
            raise ValueError('Invalid audit metadata values')
        entry = AuditLog(organization_id=organization_id, user_id=user_id,
                         action=action, entity_type=entity_type, entity_id=entity_id,
                         details=data, ip_address=None, user_agent=None)
        db.session.add(entry)
        db.session.flush()
        return entry
    # Commit/rollback belongs to the domain transaction. Retention, actor lifecycle,
    # append-only protection, completeness and protected export are separate contracts.
