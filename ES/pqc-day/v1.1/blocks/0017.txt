# Ejemplo didáctico: patrones/routes/pqc_analysis/resources.py
from flask import request
from flask_restful import Resource
from flask_jwt_extended import jwt_required, get_jwt_identity
from app.models import User, Client, CryptoFinding, AnalysisJob
from app.schemas import findings_schema
from functools import wraps
from sqlalchemy import case

def require_permission(*permissions):
    """Decorador: verifica que el usuario tiene el permiso requerido."""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            user = User.query.get(get_jwt_identity())
            if (not user or not user.is_active or user.organization is None or not user.organization.is_active
                    or not any(user.has_permission(p) for p in permissions)):
                return {"error": "Permisos insuficientes"}, 403
            return f(*args, **kwargs)
        return decorated_function
    return decorator

class FindingsResource(Resource):
    """Hallazgos criptográficos — siempre filtrados por organización."""

    @jwt_required()
    @require_permission('view_findings', 'view_own_findings')
    def get(self, client_id=None):
        """Lista hallazgos. Tres niveles de filtrado:
        1. JWT verifica identidad del usuario
        2. organization_id aísla los datos del tenant
        3. client_id opcionalmente filtra por cliente
        """
        user = User.query.get(get_jwt_identity())

        # REGLA CARDINAL: siempre filtrar por organización
        query = CryptoFinding.query.join(
            AnalysisJob, CryptoFinding.job_id == AnalysisJob.id
        ).join(
            Client, AnalysisJob.client_id == Client.id
        ).filter(Client.organization_id == user.organization_id)

        # Si es client_user, filtrar además por su cliente
        if user.role == 'client_user':
            if user.client_id is None or (client_id is not None and client_id != user.client_id):
                return {"error": "Client not found"}, 404
            authorized_client = Client.query.filter_by(
                id=user.client_id, organization_id=user.organization_id
            ).first()
            if not authorized_client:
                return {"error": "Client not found"}, 404
            query = query.filter(AnalysisJob.client_id == user.client_id)
        elif client_id:
            # Verificar que el cliente pertenece a la organización
            client = Client.query.filter_by(
                id=client_id,
                organization_id=user.organization_id
            ).first()
            if not client:
                return {"error": "Cliente no encontrado"}, 404
            query = query.filter(AnalysisJob.client_id == client_id)

        # Filtros opcionales por query string
        risk_level = request.args.get('risk_level')
        if risk_level:
            if risk_level not in {'critical', 'high', 'medium', 'low', 'info'}:
                return {"error": "Invalid risk_level"}, 400
            query = query.filter(CryptoFinding.risk_level == risk_level)

        pqc_compliant = request.args.get('pqc_compliant')
        if pqc_compliant is not None:
            if pqc_compliant.lower() not in {'true', 'false'}:
                return {"error": "Invalid pqc_compliant"}, 400
            query = query.filter(
                CryptoFinding.pqc_compliant == (pqc_compliant.lower() == 'true')
            )

        findings = query.order_by(
            case(
                (CryptoFinding.risk_level == "critical", 0),
                (CryptoFinding.risk_level == "high", 1),
                (CryptoFinding.risk_level == "medium", 2),
                (CryptoFinding.risk_level == "low", 3), else_=4
            ),
            CryptoFinding.created_at.desc()
        ).all()

        return findings_schema.dump(findings), 200
