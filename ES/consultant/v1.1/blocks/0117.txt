from dataclasses import dataclass

@dataclass
class DimensionScore:
    dimension: Dimension
    level: float          # 1.0 - 5.0, con decimales
    confidence: float     # 0.0 - 1.0
    evidence_ratio: float # Proporción de respuestas con evidencia
    key_findings: list[str]
    gaps: list[str]
    strengths: list[str]

    def __post_init__(self):
        from math import isfinite
        if not isinstance(self.dimension, Dimension):
            raise ValueError("Unknown dimension")
        if any(type(v) not in (int, float) or not isfinite(v) for v in (self.level, self.confidence, self.evidence_ratio)):
            raise ValueError("Dimension values must be finite numbers")
        if not 1 <= self.level <= 5 or not 0 <= self.confidence <= 1 or not 0 < self.evidence_ratio <= 1:
            raise ValueError("Invalid level or evidence; absent evidence is not level 1")


@dataclass
class MaturityAssessment:
    organization: str
    sector: str
    size_band: str          # "200-500", "500-2000", "2000+"
    assessment_date: str
    dimensions: list[DimensionScore]
    overall_level: float
    stakeholders_interviewed: int

    @property
    def overall_level_weighted(self) -> float:
        """Nivel global ponderado — datos y personas pesan más."""
        if not isinstance(self.dimensions, list) or len(self.dimensions) != len(Dimension) or {d.dimension for d in self.dimensions} != set(Dimension):
            raise ValueError("Weighted maturity requires exactly one evidenced record for each dimension")
        for dimension in self.dimensions:
            dimension.__post_init__()

        weights = {
            Dimension.STRATEGY: 0.15,
            Dimension.DATA: 0.25,      # Peso mayor: sin datos no hay IA
            Dimension.TECHNOLOGY: 0.20,
            Dimension.PEOPLE: 0.25,    # Peso mayor: sin personas no hay cambio
            Dimension.GOVERNANCE: 0.15,
        }
        total = sum(
            d.level * weights[d.dimension] for d in self.dimensions
        )
        return round(total, 1)
