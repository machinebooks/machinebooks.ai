@dataclass
class ScoreGoNoGo:
    """Resultado del análisis go/no-go de un RFP."""
    puntuacion_global: float          # 0-100
    recomendacion: str                # "go" | "no-go" | "go-condicional"
    requisitos_criticos_cumplidos: int
    requisitos_criticos_total: int
    brechas_bloqueantes: list[str]
    brechas_mitigables: list[str]
    fortalezas_competitivas: list[str]
    riesgos_principales: list[str]
    esfuerzo_propuesta_horas: int     # Estimación de horas para preparar
    coste_estimado_propuesta: float   # En EUR
    probabilidad_estimada_ganar: str  # "alta" | "media" | "baja"
    justificacion: str                # Narrativa de 3-5 frases

    def __post_init__(self):
        from math import isfinite
        if type(self.puntuacion_global) not in (int, float) or not isfinite(self.puntuacion_global) or not 0 <= self.puntuacion_global <= 100:
            raise ValueError("Invalid go/no-go score")
        counts = (self.requisitos_criticos_cumplidos, self.requisitos_criticos_total, self.esfuerzo_propuesta_horas)
        if any(type(v) is not int or v < 0 for v in counts) or self.requisitos_criticos_cumplidos > self.requisitos_criticos_total:
            raise ValueError("Invalid requirement counts or proposal effort")
        if type(self.coste_estimado_propuesta) not in (int, float) or not isfinite(self.coste_estimado_propuesta) or self.coste_estimado_propuesta < 0:
            raise ValueError("Invalid proposal cost")
        if self.recomendacion not in ("go", "no-go", "go-condicional") or self.probabilidad_estimada_ganar not in ("alta", "media", "baja"):
            raise ValueError("Use canonical recommendation and probability enums")
        for values in (self.brechas_bloqueantes, self.brechas_mitigables, self.fortalezas_competitivas, self.riesgos_principales):
            if not isinstance(values, list) or any(type(v) is not str for v in values):
                raise ValueError("Invalid go/no-go lists")
        expected = "no-go" if self.puntuacion_global < 30 else "go-condicional" if self.puntuacion_global < 65 else "go"
        if self.recomendacion != expected or (self.brechas_bloqueantes and self.recomendacion != "no-go"):
            raise ValueError("Recommendation contradicts score or blocking gaps")
        if self.recomendacion == "go" and (self.requisitos_criticos_total == 0 or self.requisitos_criticos_cumplidos != self.requisitos_criticos_total):
            raise ValueError("Go requires a known, fully satisfied critical requirement set")
