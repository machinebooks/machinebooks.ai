from hashlib import sha256
from pathlib import Path

def audit_deliverable_transparency(document_path, review_record, authorize_reviewer):
    artifact = Path(document_path).read_bytes()
    if review_record["artifact_hash"] != sha256(artifact).hexdigest():
        raise ValueError("Review does not match artifact")
    if not authorize_reviewer(review_record["reviewer_id"], review_record["version"]):
        raise ValueError("Unauthorized reviewer")
    checks = review_record["checks"]
    required = {"methodology_note", "attribution", "factual_review", "privacy_review"}
    if set(checks) != required or any(value is not True for value in checks.values()):
        raise ValueError("Transparency checks incomplete")
    if review_record["decision"] != "approved":
        raise ValueError("Human decision pending")
    return {"artifact_hash": review_record["artifact_hash"], "status": "approved"}
