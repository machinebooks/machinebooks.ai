from datetime import datetime, timezone

@tool
def verify_control_evidence(
    control_id: str,
    framework: str,
    evidence_sources: list[str]
) -> dict:
    """Verifica si la evidencia de un control sigue siendo válida."""
    # Recopilar evidencias (APIs, documentos, configuraciones)
    evidence_data = collect_evidence(evidence_sources)

    if not evidence_data or isinstance(evidence_data, str) and not evidence_data.strip():
        return {
            "control_id": control_id, "framework": framework,
            "status": "NO_EVALUADO",
            "justification": "Insufficient evidence; human review required",
            "verified_at": datetime.now(timezone.utc).isoformat(),
            "cost_usd": 0.0,
        }

    message = client.messages.create(
        model="claude-haiku-4-5",  # Haiku para verificaciones frecuentes
        max_tokens=512,
        system=(
            f"Eres un auditor de {framework}. Evalúa si la evidencia "
            "presentada demuestra cumplimiento del control. "
            "Si falta evidencia suficiente, usa NO_EVALUADO. "
            "Responde con: CUMPLE, CUMPLE_PARCIAL, NO_CUMPLE o NO_EVALUADO, "
            "seguido de una justificación en una frase."
        ),
        messages=[{
            "role": "user",
            "content": (
                f"Control: {control_id}\n"
                f"Evidencia recopilada:\n{evidence_data}\n"
                f"Fecha de verificación: {datetime.now(timezone.utc).isoformat()}\n"
                "Evaluación:"
            )
        }]
    )

    response = message.content[0].text.strip()
    token = response.split(",", 1)[0].split()
    status = token[0] if token else "NO_EVALUADO"
    if status not in ['CUMPLE', 'CUMPLE_PARCIAL', 'NO_CUMPLE', 'NO_EVALUADO']:
        status = "NO_EVALUADO"
        response = "Invalid verification response; human review required"
    if not response:
        response = "Empty verification response; human review required"

    return {
        "control_id": control_id,
        "framework": framework,
        "status": status,
        "justification": response,
        "verified_at": datetime.now(timezone.utc).isoformat(),
        "cost_usd": 0.003,  # Coste estimado por verificación con Haiku
    }
