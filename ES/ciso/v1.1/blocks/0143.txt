from math import ceil
from enum import Enum
from fastapi import APIRouter, Depends, Query, HTTPException
from sqlalchemy import select, func
from app.database import get_db
from app.auth.dependencies import get_current_user
from app.auth.permissions import authorize_privacy_read
from app.models.privacy import Treatment
from app.schemas.privacy import PaginatedResponse, TreatmentResponse

router = APIRouter(prefix="/privacy", tags=["privacy"])

class SortOrder(str, Enum):
    ASC = "asc"
    DESC = "desc"

@router.get("/treatments", response_model=PaginatedResponse[TreatmentResponse])
async def list_treatments(
    page: int = Query(1, ge=1, le=10000),
    page_size: int = Query(25, ge=1, le=100),
    sort_by: str = Query("last_review"),
    sort_order: SortOrder = SortOrder.DESC,
    legal_basis: str | None = Query(None, max_length=100),
    dpia_status: str | None = Query(None, max_length=50),
    db=Depends(get_db), user=Depends(get_current_user),
):
    authorize_privacy_read(user)
    columns = {"last_review": Treatment.last_review, "name": Treatment.name, "id": Treatment.id}
    if sort_by not in columns:
        raise HTTPException(422, "Campo de ordenación no permitido")
    # Domain schemas/adapters validate legal-basis and DPIA-status vocabularies.
    query = select(Treatment).where(
        Treatment.corporate_id == user.corporate_id, Treatment.is_deleted.is_(False))
    if legal_basis is not None:
        query = query.where(Treatment.legal_basis == legal_basis)
    if dpia_status is not None:
        query = query.where(Treatment.dpia_status == dpia_status)
    total = await db.scalar(select(func.count()).select_from(query.subquery()))
    column = columns[sort_by]
    order = column.desc() if sort_order == SortOrder.DESC else column.asc()
    items = (await db.execute(
        query.order_by(order, Treatment.id.asc()).offset((page - 1) * page_size)
        .limit(page_size))).scalars().all()
    return {"items": items, "total": total, "page": page, "page_size": page_size,
            "total_pages": ceil(total / page_size)}
