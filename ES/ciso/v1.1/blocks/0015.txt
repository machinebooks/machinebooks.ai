def seed_sector_catalog(session, corporate_id: int, sector: str,
                        created_by: int, controller_name: str):
    if type(corporate_id) is not int or corporate_id < 1:
        raise ValueError("Invalid tenant")
    if type(created_by) is not int or created_by < 1:
        raise ValueError("Invalid actor")
    if sector != "municipal":
        raise ValueError("Only the illustrated municipal templates are supported")
    if not isinstance(controller_name, str) or not controller_name.strip() or len(controller_name) > 255:
        raise ValueError("Invalid controller name")
    created = 0
    with session.begin():
        for number, template in enumerate(MUNICIPAL_DRAFT_TEMPLATES, start=1):
            code = f"RAT-SEED-MUNICIPAL-{number:03d}"
            existing = session.scalar(select(DataProcessingActivity).where(
                DataProcessingActivity.corporate_id == corporate_id,
                DataProcessingActivity.code == code,
            ))
            if existing is not None:
                # Preserve human changes and archived entries; restoration is a separate decision.
                continue
            data = dict(template)
            data["legal_basis"] = LegalBasis(data["legal_basis"])
            session.add(DataProcessingActivity(
                **data, corporate_id=corporate_id, created_by=created_by,
                status="draft", code=code, controller_name=controller_name.strip(),
            ))
            created += 1
    # (corporate_id, code) must be unique in the schema: a concurrent insert rolls back.
    return {"sector": sector, "treatments_created": created,
            "status": "draft_pending_controller_review"}
