from datetime import datetime, timedelta, timezone
from fastapi import APIRouter, Depends, Query
from app.database import get_db
from app.auth import require_role
from app.models import AIMonitoringMetric
from app.services.ai_budget import approved_monthly_spend

router = APIRouter(prefix="/api/v1/admin/ai", tags=["ai-monitoring"])

@router.get("/monitoring/dashboard")
def get_ai_monitoring_dashboard(
    period: str = Query("7d", pattern="^(24h|7d|30d|90d)$"),
    service_type: str | None = Query(None, max_length=50),
    aggregation: str = Query("daily", pattern="^(daily|weekly)$"),
    current_user=Depends(require_role("admin")), db=Depends(get_db),
):
    days = {"24h": 1, "7d": 7, "30d": 30, "90d": 90}[period]
    now = datetime.now(timezone.utc)
    query = db.query(AIMonitoringMetric).filter(
        AIMonitoringMetric.corporate_id == current_user.corporate_id,
        AIMonitoringMetric.period_type == aggregation,
        AIMonitoringMetric.period_start >= now - timedelta(days=days),
        AIMonitoringMetric.period_end <= now)
    if service_type is not None:
        query = query.filter(AIMonitoringMetric.service_type == service_type)
    rows = query.order_by(AIMonitoringMetric.period_start.desc(),
                          AIMonitoringMetric.id.desc()).limit(201).all()
    if len(rows) > 200:
        return {"status": "incomplete_requires_pagination", "period": period}
    # Required adapter authorizes the tenant and reports meter/FX coverage.
    # Cost is null until comparable USD records and reservations are reconciled.
    cost = approved_monthly_spend(db, current_user, now)
    return {
        "period": period, "coverage": "stored_bounded_aggregates",
        "metrics": [{"metric_name": row.metric_name,
                     "service_type": row.service_type, "value": row.value,
                     "status": row.status, "trend": row.trend,
                     "trend_delta_pct": row.trend_delta_pct,
                     "sample_size": row.sample_size,
                     "period_start": row.period_start.isoformat()}
                    for row in rows],
        "cost": cost,
        "legal_conformity": "not_assessed",
    }
# No quality, fairness or drift evaluation is generated by this endpoint.
