# Continuation of policy model imports; collect event COUNTS only in an authorized simulation.
from sqlalchemy import CheckConstraint

class PhishingSimulation(BaseModel):
    __tablename__ = "pa_phishing_simulations"
    title: Mapped[str] = mapped_column(String(500), nullable=False)
    template_type: Mapped[str | None] = mapped_column(String(50), nullable=True)
    status: Mapped[str] = mapped_column(String(20), nullable=False, default="planned")
    emails_sent: Mapped[int | None] = mapped_column(Integer, nullable=True)
    emails_opened: Mapped[int | None] = mapped_column(Integer, nullable=True)
    links_clicked: Mapped[int | None] = mapped_column(Integer, nullable=True)
    credentials_submitted: Mapped[int | None] = mapped_column(Integer, nullable=True)
    reported_by_users: Mapped[int | None] = mapped_column(Integer, nullable=True)
    click_rate: Mapped[float | None] = mapped_column(Float, nullable=True)
    report_rate: Mapped[float | None] = mapped_column(Float, nullable=True)
    target_departments: Mapped[dict | None] = mapped_column(JSON, nullable=True)
    ai_adaptive: Mapped[bool] = mapped_column(Boolean, default=False, nullable=False)
    __table_args__ = (
        CheckConstraint("emails_sent IS NULL OR emails_sent >= 0"),
        CheckConstraint("links_clicked IS NULL OR links_clicked >= 0"),
        CheckConstraint("reported_by_users IS NULL OR reported_by_users >= 0"),
        CheckConstraint("click_rate IS NULL OR (click_rate >= 0 AND click_rate <= 100)"),
        CheckConstraint("report_rate IS NULL OR (report_rate >= 0 AND report_rate <= 100)"),
    )
# Rate uses unique recipients in the same approved cohort/window; zero sent or unknown
# counts produce NULL, not a 0% success claim. Validate event deduplication and denominators.
# credentials_submitted counts a benign simulation event; never collect/store real passwords.
# AI-adaptive is a requested option, not an implemented/adaptive-learning guarantee.
# Employee/privacy/consent-or-other-basis, permitted targets and retention require review.
