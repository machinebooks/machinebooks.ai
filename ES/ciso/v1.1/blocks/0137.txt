import { vi, expect, test } from "vitest";
import { render, screen } from '@testing-library/react'
import { axe, toHaveNoViolations } from 'jest-axe'
import { DPIAForm } from '@/modules/privacy/DPIAForm'

expect.extend(toHaveNoViolations)
test('Etiquetas del estado inicial y comprobación DOM limitada', async () => {
  const { container } = render(<DPIAForm onSubmit={vi.fn()} />)
  expect(screen.getByRole('textbox', { name: 'Descripción del tratamiento' })).toBeTruthy()
  expect(screen.getByRole('button', { name: 'Guardar evaluación' })).toBeTruthy()
  expect(await axe(container)).toHaveNoViolations()
})
