# Alternative worker fragment, merged with the chapter's approved production topology.
# It is not added to the shared-worker version without removing overlapping consumers.
x-worker: &worker
  build:
    context: ./backend
    dockerfile: Dockerfile.prod
  environment:
    DATABASE_URL_FILE: /run/secrets/worker_database_url
    REDIS_URL_FILE: /run/secrets/redis_url
    RUNTIME_CONFIG_FILE: /run/secrets/runtime_config
    AI_SECRET_CONFIG_FILE: /run/secrets/ai_secret_config
  secrets: [worker_database_url, redis_url, runtime_config, ai_secret_config]
  networks: [spd_backend]
  depends_on:
    redis: {condition: service_healthy}
    mysql: {condition: service_healthy}
  restart: unless-stopped
services:
  worker-ai:
    <<: *worker
    command: celery -A app.celery_app worker -Q ai --concurrency=2 --max-tasks-per-child=50 --loglevel=info --hostname=worker-ai@%h
    deploy: {resources: {limits: {memory: 2G}}}
  worker-email:
    <<: *worker
    command: celery -A app.celery_app worker -Q email --concurrency=4 --max-tasks-per-child=200 --loglevel=info --hostname=worker-email@%h
    deploy: {resources: {limits: {memory: 512M}}}
  worker-reports:
    <<: *worker
    command: celery -A app.celery_app worker -Q reports --concurrency=2 --max-tasks-per-child=20 --loglevel=info --hostname=worker-reports@%h
    deploy: {resources: {limits: {memory: 1G}}}
  worker-ops:
    <<: *worker
    command: celery -A app.celery_app worker -Q maintenance,notifications --concurrency=2 --max-tasks-per-child=100 --loglevel=info --hostname=worker-ops@%h
    deploy: {resources: {limits: {memory: 512M}}}
  celery-beat:
    <<: *worker
    command: celery -A app.celery_app beat --loglevel=info --pidfile=/tmp/celerybeat.pid --schedule=/var/run/celery/celerybeat-schedule
    volumes: [celerybeat-data:/var/run/celery]
  flower:
    <<: *worker
    command: celery -A app.celery_app flower --port=5555 --persistent=True --db=/var/flower/flower.db --conf=/run/secrets/flower_config
    secrets: [worker_database_url, redis_url, runtime_config, ai_secret_config, flower_config]
    ports: ["127.0.0.1:5555:5555"]
    volumes: [flower-data:/var/flower]
    profiles: [monitoring]
volumes:
  celerybeat-data: {}
  flower-data: {}
secrets:
  flower_config:
    file: ${FLOWER_CONFIG_FILE:?Set the protected Flower config file}
# spd_backend, shared secrets, mysql/redis, protected build assets and runtime readers
# are inherited requirements from the coherent base topology. Flower auth belongs in its
# protected config, not process argv. Test the merged Compose model before any deployment.
# Four pools serve five queues; ops shares maintenance/notifications and reserves no capacity.
