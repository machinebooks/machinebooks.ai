import { test, expect } from "./fixtures"
// Project fixtures provide synthetic role identities, valid DTOs and document extraction.
test.use({ timezoneId: "UTC" })

test("DPO creates a breach draft with an awareness-based 72h reference", async ({ page, roleFixtures }, testInfo) => {
  await roleFixtures.login(page, "dpo")
  await page.goto("/privacy/breaches/new")
  const awareness = "2026-10-04T10:00:00Z"
  await page.fill('[data-testid="titulo"]', "Caso sintético: acceso no autorizado")
  await page.fill('[data-testid="fecha-deteccion"]', "2026-10-04T09:50")
  await page.fill('[data-testid="fecha-constancia"]', "2026-10-04T10:00")
  await roleFixtures.fill_valid_breach_fields(page)
  const createdResponse = page.waitForResponse(response =>
    response.url().endsWith("/api/v1/breaches/") &&
    response.request().method() === "POST")
  await page.click('[data-testid="btn-guardar"]')
  const response = await createdResponse
  expect(response.status()).toBe(201)
  const created = await response.json()
  expect(new Date(created.awareness_at).toISOString()).toBe(awareness)
  expect(new Date(created.notification_deadline).getTime()).toBe(
    new Date(awareness).getTime() + 72 * 60 * 60 * 1000)
  await expect(page.locator('[data-testid="plazo-72h-badge"]')).toBeVisible()

  const downloadPromise = page.waitForEvent("download")
  await page.click('[data-testid="btn-exportar-aepd"]')
  const download = await downloadPromise
  expect(await download.failure()).toBeNull()
  expect(download.suggestedFilename()).toMatch(/\.(pdf|docx)$/)
  const output = testInfo.outputPath("breach-fixture-export")
  await download.saveAs(output)
  const text = await roleFixtures.read_docx_or_pdf_for_test(output)
  expect(text).toContain("Caso sintético: acceso no autorizado")
  expect(text).toContain(created.code)
})
test("Analyst is denied a valid breach-write request", async ({ page, roleFixtures }) => {
  await roleFixtures.login(page, "analyst")
  await page.goto("/privacy/breaches/new")
  await expect(page.locator('[data-testid="error-permisos"]')).toBeVisible()
  const payload = await roleFixtures.valid_breach_payload()
  const response = await page.request.post("/api/v1/breaches/", { data: payload })
  expect(response.status()).toBe(403)
})
