class MappingType(str, enum.Enum):
    EQUIVALENT = "equivalent"    # Mismo requisito, distinto marco
    RELATED = "related"          # Requisito similar pero no idéntico
    EXTENDS = "extends"          # Un control extiende a otro (27701 → 27001)

class ControlMapping(BaseModel):
    """Mapeo cruzado entre controles de diferentes marcos."""
    __tablename__ = "compliance_control_mappings"

    id = Column(BigInteger, primary_key=True)
    source_control_id = Column(
        BigInteger,
        nullable=False
    )
    target_control_id = Column(
        BigInteger,
        nullable=False
    )
    mapping_type = Column(SAEnum(MappingType), nullable=False)
    notes = Column(Text)  # Matices del mapeo

    # Relaciones
    source_control = relationship(
        "ComplianceControl",
        foreign_keys="[ControlMapping.corporate_id, ControlMapping.source_control_id]",
        viewonly=True
    )
    target_control = relationship(
        "ComplianceControl",
        foreign_keys="[ControlMapping.corporate_id, ControlMapping.target_control_id]",
        viewonly=True
    )

    __table_args__ = (
        UniqueConstraint(
            "corporate_id", "source_control_id", "target_control_id",
            name="uq_control_mapping"
        ),
        ForeignKeyConstraint(["corporate_id", "source_control_id"],
            ["compliance_controls.corporate_id", "compliance_controls.id"]),
        ForeignKeyConstraint(["corporate_id", "target_control_id"],
            ["compliance_controls.corporate_id", "compliance_controls.id"]),
    )
