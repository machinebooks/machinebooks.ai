# Example operational thresholds, to validate against the actual emitted metric series.
groups:
  - name: grc_security_alerts
    rules:
      - alert: HighAuthFailureRate
        expr: sum(rate(grc_auth_failures_total[5m])) > 0.5
        for: 2m
        labels: {severity: critical}
        annotations:
          summary: "Tasa alta de respuestas HTTP 401 en login"
          description: "Más de 30 respuestas 401 por minuto durante 2 minutos"
      - alert: LLMLatencyHigh
        expr: histogram_quantile(0.95, sum by (le, provider, model) (rate(grc_llm_latency_seconds_bucket[5m]))) > 30
        for: 5m
        labels: {severity: warning}
        annotations:
          summary: "p95 de intentos LLM por encima de 30 segundos"
      - alert: CeleryQueueBacklog
        expr: grc_celery_tasks_pending{queue="ai",job="grc-celery-exporter"} > 50
        for: 10m
        labels: {severity: warning}
        annotations:
          summary: "Cola IA con tareas pendientes"
          description: "Más de 50 tareas pendientes durante 10 minutos; se requiere exporter válido"
      - alert: HighErrorRate
        expr: (sum(rate(http_requests_total{status="5xx"}[5m])) / sum(rate(http_requests_total[5m])) > 0.05) and (sum(rate(http_requests_total[5m])) > 0)
        for: 5m
        labels: {severity: critical}
        annotations:
          summary: "Tasa 5xx superior al 5% con tráfico observado"
      - alert: MetricsTargetUnavailable
        expr: up{job=~"grc-backend|grc-celery-exporter"} == 0
        for: 2m
        labels: {severity: warning}
        annotations:
          summary: "Destino de métricas no disponible"
# Pending is broker backlog, not the active-worker gauge. Exporter must implement/validate
# separate pending/active series, inspection freshness and unknown/unavailable states.
# Thresholds are case policy, not ENS/NIS2/AI Act obligations or evidence of legal compliance.
# Missing series are not zero. Test rule inputs, restarts, no traffic and collector outages.
