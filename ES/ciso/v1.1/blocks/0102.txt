# CAs de confianza cargadas al arranque
TRUSTED_CA_CERTS = load_trusted_cas("/etc/ssl/trusted-cas/")


def parse_x509_certificate(cert_pem_urlencoded: str) -> dict:
    """Parsea el certificado que Nginx pasa como header URL-encoded.
    Extrae CN, serial, emisor, fechas y extensiones relevantes."""
    cert_pem = urllib.parse.unquote(cert_pem_urlencoded)
    cert = x509.load_pem_x509_certificate(cert_pem.encode())

    return {
        "subject_cn": cert.subject.get_attributes_for_oid(
            x509.oid.NameOID.COMMON_NAME
        )[0].value,
        "issuer_cn": cert.issuer.get_attributes_for_oid(
            x509.oid.NameOID.COMMON_NAME
        )[0].value,
        "serial_number": cert.serial_number,
        "not_before": cert.not_valid_before_utc,
        "not_after": cert.not_valid_after_utc,
        "cert_object": cert,
    }


def verify_certificate_chain(cert_info: dict) -> bool:
    """Delegate client path validation to cryptography; revocation is separate."""
    cert = cert_info["cert_object"]
    try:
        store = Store(list(TRUSTED_CA_CERTS.values()))
        verifier = (PolicyBuilder().store(store)
                    .time(datetime.now(timezone.utc)).build_client_verifier())
        verifier.verify(cert, cert_info.get("intermediates", []))
        return True
    except (VerificationError, ValueError, TypeError):
        return False

    """Verifica revocación contra OCSP y CRL (fallback).

async def check_ocsp(cert: x509.Certificate, ocsp_url: str) -> Optional[bool]:
    """Use an integration adapter; never accept an unverified GOOD response."""
    from app.services.pki_verifier import validated_ocsp_status
    # Contract: approved destination, signed authorized responder, matching
    # CertID, response status, freshness, nonce policy and bounded response.
    status = await validated_ocsp_status(cert, ocsp_url, timeout=5.0)
    if status == "GOOD":
        return False
    if status == "REVOKED":
        return True
    return None  # UNKNOWN/unverified is not evidence of a valid certificate.
