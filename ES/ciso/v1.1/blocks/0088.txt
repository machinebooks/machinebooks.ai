# Integration must invoke this boundary for every authorized terminal state.
# Required audit policy binds request IDs to the current actor/tenant; no prompts are retained.
from math import isfinite
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.audit import AuditTrail

def optional_metric(value, integer=False):
    if value is None:
        return None
    if (isinstance(value, bool) or not isinstance(value, (int, float))
            or not isfinite(value) or value < 0 or (integer and type(value) is not int)):
        raise ValueError("Invalid nonnegative metric")
    return value if integer else round(value, 4)

async def log_copilot_execution(db: AsyncSession, execution_id, request, mode, steps,
                                total_tokens, total_cost_eur, duration_ms, status, policy):
    if status not in {"completed", "partial", "failed", "blocked", "cancelled"} or len(steps) > 256:
        raise ValueError("Invalid bounded terminal audit")
    await policy.authorize_audit(request, execution_id)
    entry = AuditTrail(
        action="copilot_execution", entity_type="copilot", entity_id=execution_id,
        user_id=request.user_id, corporate_id=request.tenant_id,
        details={"module_context": request.module_context, "mode": mode.value if mode else None,
                 "status": status,
                 "steps": [{"agent": s.agent_name, "action": s.action, "status": s.status,
                            "tokens": optional_metric(s.tokens_used, True),
                            "cost_eur": optional_metric(s.cost_eur),
                            "duration_ms": optional_metric(s.duration_ms, True)} for s in steps],
                 "totals": {"tokens": optional_metric(total_tokens, True),
                            "cost_eur": optional_metric(total_cost_eur),
                            "duration_ms": optional_metric(duration_ms, True)}})
    db.add(entry)
    await db.flush()
    # The owning transaction must commit or roll back; successful insertion is not immutable custody.
