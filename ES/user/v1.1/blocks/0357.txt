#!/usr/bin/env python3
# Read-only parser for nginx combined logs; run on approved, non-overlapping copies.
import gzip
import json
import locale
import re
import sys
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path

locale.setlocale(locale.LC_TIME, "C")
directory = Path(sys.argv[1])
start = datetime.fromisoformat("2026-03-15T00:00:00+00:00")
end = datetime.fromisoformat("2026-03-22T00:00:00+00:00")
files = sorted(path for path in directory.glob("access.log*")
               if path.is_file() and not path.is_symlink()
               and re.fullmatch(r"access\.log(?:\.[0-9]+(?:\.gz)?)?", path.name))
if not files:
    raise ValueError("No matching log files")
pattern = re.compile(r'^\S+ \S+ \S+ \[([^\]]+)\] "([^"]*)" ([0-9]{3}) ')
codes, targets, days = Counter(), Counter(), Counter()
for path in files:
    opener = gzip.open if path.suffix == ".gz" else open
    with opener(path, "rt", encoding="utf-8") as handle:
        for line_number, line in enumerate(handle, 1):
            match = pattern.match(line)
            if not match:
                raise ValueError(f"Unsupported log format: {path.name}:{line_number}")
            stamp = datetime.strptime(match[1], "%d/%b/%Y:%H:%M:%S %z")
            stamp = stamp.astimezone(timezone.utc)
            request = match[2].split()
            if len(request) != 3:
                raise ValueError(f"Unsupported request: {path.name}:{line_number}")
            status = int(match[3])
            if start <= stamp < end and status in (500, 502, 503, 504):
                codes[status] += 1
                targets[request[1]] += 1
                days[stamp.date().isoformat()] += 1
print(json.dumps({"files": len(files), "period_utc": [start.isoformat(), end.isoformat()],
                  "codes": dict(sorted(codes.items())),
                  "top_targets": targets.most_common(10),
                  "days": dict(sorted(days.items()))}, ensure_ascii=False, indent=2))
