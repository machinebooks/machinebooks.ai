#!/usr/bin/env python3
# Read-only Lambda invocation totals for the fixed illustrative UTC period.
import json
import os
import subprocess

profile = os.environ["AWS_PROFILE"]
region = os.environ["AWS_REGION"]
expected = os.environ["EXPECTED_AWS_ACCOUNT"]
def aws(*args):
    return json.loads(subprocess.check_output(
        ["aws", "--profile", profile, "--region", region, "--no-cli-pager",
         *args, "--output", "json"], text=True))
if aws("sts", "get-caller-identity")["Account"] != expected:
    raise ValueError("Unexpected AWS account")
rows = []
for function in aws("lambda", "list-functions")["Functions"]:
    name = function["FunctionName"]
    data = aws("cloudwatch", "get-metric-statistics",
               "--namespace", "AWS/Lambda", "--metric-name", "Invocations",
               "--dimensions", f"Name=FunctionName,Value={name}",
               "--start-time", "2026-03-01T00:00:00Z",
               "--end-time", "2026-03-22T00:00:00Z",
               "--period", "86400", "--statistics", "Sum")
    points = data["Datapoints"]
    rows.append({"function": name, "invocations_observed":
                 sum(p["Sum"] for p in points) if points else None,
                 "days_with_points": len(points)})
rows.sort(key=lambda r: (r["invocations_observed"] is None,
                        -(r["invocations_observed"] or 0), r["function"]))
print(json.dumps({"account": expected, "region": region,
                  "period_utc": ["2026-03-01T00:00:00Z", "2026-03-22T00:00:00Z"],
                  "rows": rows[:10], "missing_points_are_not_zero": True,
                  "function_costs": "NOT_ASSESSED"}, indent=2))
