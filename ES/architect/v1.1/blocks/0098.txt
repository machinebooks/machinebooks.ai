    def extract_project_list(self) -> list[dict]:
        """Extracts the list of active projects from the portal."""
        # Navigate to the projects section
        projects_link = self.wait_for_element(
            (By.LINK_TEXT, "Active Projects")
        )
        projects_link.click()

        projects = []
        page = 1
        seen={}
        deadline=time.monotonic()+120

        while True:
            if page>500 or time.monotonic()>=deadline:
                raise TimeoutError("Extraction incomplete: page/deadline limit")
            # Wait for the table to load
            self.wait_for_element((By.CSS_SELECTOR, "table.projects-table tbody tr"))

            rows = self.driver.find_elements(
                By.CSS_SELECTOR, "table.projects-table tbody tr"
            )

            for row in rows:
                cells = row.find_elements(By.TAG_NAME, "td")
                if len(cells) >= 4:
                    record={
                        "id": cells[0].text.strip(),
                        "name": cells[1].text.strip(),
                        "status": cells[2].text.strip(),
                        "team": cells[3].text.strip(),
                    }
                    if not record["id"]:raise ValueError("Project identifier required")
                    if record["id"] in seen and seen[record["id"]]!=record:
                        raise ValueError("Conflicting repeated project row")
                    if record["id"] not in seen:
                        seen[record["id"]]=record
                        projects.append(record)
                    if len(projects)>100000:raise ValueError("Extraction row limit")

            # Check if there is a next page
            try:
                next_btn = self.driver.find_element(
                    By.CSS_SELECTOR, "button.pagination-next:not([disabled])"
                )
            except NoSuchElementException:
                break  # Only an absent enabled next control ends this example.
            next_btn.click()
            if not rows:raise ValueError("Page transition lacks a row witness")
            WebDriverWait(self.driver,self.DEFAULT_WAIT).until(EC.staleness_of(rows[0]))
            page += 1

        return projects
