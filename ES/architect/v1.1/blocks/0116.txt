from datetime import datetime, timezone
from sqlalchemy import (Column, String, Integer, Numeric, DateTime,
                        Boolean, Float, Text, ForeignKey, JSON)

class LLMUsageLog(Base):
    __tablename__ = "llm_usage_logs"
    id = Column(Integer, primary_key=True)
    reservation_id = Column(String(64), nullable=False, unique=True)
    timestamp = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc),
                       nullable=False, index=True)
    tenant_id = Column(String(64), nullable=False, index=True)
    service_type = Column(String(64), nullable=False, index=True)
    provider = Column(String(32), nullable=False)
    model_id = Column(String(100))  # Observed model; unknown on failed calls
    requested_model = Column(String(100), nullable=False)
    input_tokens = Column(Integer)  # Missing usage stays NULL, not zero
    output_tokens = Column(Integer)
    total_tokens = Column(Integer)
    latency_ms = Column(Integer)
    cost_eur = Column(Numeric(18, 8))
    pricing_version_id = Column(String(64))
    user_id = Column(String(64), index=True)
    correlation_id = Column(String(64))
    prompt_id = Column(String(64))
    prompt_hash = Column(String(64))  # Full SHA-256; not anonymization/authenticity
    contained_pii = Column(Boolean)  # NULL means not assessed
    pii_redacted = Column(Boolean)
    privacy_restricted = Column(Boolean, nullable=False, default=False)
    success = Column(Boolean, nullable=False)
    error_type = Column(String(100))

# Base is the application's declarative base. This is a didactic schema projection,
# not a migration for the chapter-5 Flask model. UTC conversion, row authorization,
# constraints and durable reservation/usage transactions require integration tests.
