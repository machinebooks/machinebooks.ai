# Didactic example: patterns/automation/base_bot.py

import os
import time
import logging
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.common.exceptions import TimeoutException, WebDriverException

logger = logging.getLogger(__name__)

class BaseBot:
    """Base class for all RPA automation bots.

    Manages the driver lifecycle, retries, and logging
    in a centralized manner to avoid duplication between bots.
    """

    SELENIUM_HUB_URL = os.environ["SELENIUM_HUB_URL"]
    DEFAULT_WAIT = 15    # seconds to wait for elements
    PAGE_LOAD_TIMEOUT = 30

    def __init__(self, task_id: str, headless: bool = True):
        self.task_id = task_id
        self.headless = headless
        self.driver = None
        self.wait = None

    def _create_driver(self) -> webdriver.Remote:
        """Creates a Chrome session in the Grid with standard options."""
        options = Options()
        if self.headless:
            options.add_argument("--headless=new")

        # Critical stability options for containerized environments
        options.add_argument("--disable-dev-shm-usage")
        options.add_argument("--disable-gpu")
        options.add_argument("--window-size=1920,1080")
        options.add_argument("--disable-notifications")

        driver = webdriver.Remote(
            command_executor=self.SELENIUM_HUB_URL,
            options=options
        )
        try:
            driver.set_page_load_timeout(self.PAGE_LOAD_TIMEOUT)
            return driver
        except BaseException:
            try:driver.quit()
            except Exception:logger.warning("Driver setup cleanup unconfirmed")
            raise
