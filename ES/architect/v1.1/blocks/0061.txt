# Correlation is observational, never an authorization credential.
import logging
import time
import uuid
from flask import g, request

logger = logging.getLogger(__name__)


def init_request_context(app):
    @app.before_request
    def set_request_context():
        raw = request.headers.get("X-Request-ID", "")
        try:
            if not isinstance(raw, str) or len(raw) != 36:
                raise ValueError("Invalid correlation ID")
            request_id = str(uuid.UUID(raw))
        except (ValueError,TypeError,AttributeError):
            request_id = str(uuid.uuid4())
        g.request_id = request_id
        g.request_start_time = time.perf_counter()

    @app.after_request
    def add_correlation_id(response):
        request_id = getattr(g, "request_id", str(uuid.uuid4()))
        response.headers["X-Request-ID"] = request_id
        started = getattr(g, "request_start_time", None)
        duration = None if started is None else (time.perf_counter()-started)*1000
        logger.info("request_completed", extra={
            "request_id": request_id, "endpoint": request.endpoint,
            "status_code": response.status_code, "duration_ms": duration,
        })
        return response
