import { useState } from 'react';

export default function AIDisclaimerBanner() {
  const [dismissed, setDismissed] = useState(false);
  if (dismissed) return null;
  return (
    <div className="border-b border-gray-300 bg-gray-50 px-4 py-3">
      <p className="text-sm text-gray-800">
        <strong>Aviso IA:</strong> Esta aplicación utiliza inteligencia artificial
        para análisis, evaluaciones y recomendaciones. Verifique la información
        antes de tomar decisiones.
      </p>
      <button type="button" onClick={() => setDismissed(true)}
              className="mt-2 rounded px-3 py-2 text-sm underline focus-visible:outline">
        Cerrar aviso en esta sesión
      </button>
    </div>
  );
}
