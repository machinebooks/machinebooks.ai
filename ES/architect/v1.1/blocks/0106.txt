    def list_changed_files(self,start_url):
        self._validate_link(start_url)
        token=self.get_token(self.context)
        if not isinstance(token,str) or not token:raise PermissionError("Authorized token required")
        url=start_url;changes=[];seen_links=set();deadline=time.monotonic()+120
        for page in range(2000):
            if url in seen_links or time.monotonic()>=deadline:
                raise TimeoutError("Delta incomplete or cyclic")
            self._validate_link(url);seen_links.add(url)
            response=self.client.get(url,headers={"Authorization":"Bearer "+token},
                                     timeout=10,follow_redirects=False)
            response.raise_for_status();data=response.json()
            if not isinstance(data,dict) or not isinstance(data.get("value"),list):
                raise ValueError("Delta page required")
            for item in data["value"]:
                if not isinstance(item,dict) or not isinstance(item.get("id"),str) or not item["id"]:
                    raise ValueError("Delta identifier required")
                if "deleted" in item or "@removed" in item:
                    changes.append({"id":item["id"],"deleted":True})
                elif "file" in item:
                    modified=datetime.fromisoformat(item["lastModifiedDateTime"].replace("Z","+00:00"))
                    if modified.tzinfo is None:raise ValueError("Aware modification time required")
                    changes.append({"id":item["id"],"name":item.get("name"),
                      "modified_at":modified.astimezone(timezone.utc).isoformat().replace("+00:00","Z"),
                      "size":item.get("size"),"deleted":False})
                if len(changes)>100000:raise ValueError("Delta change limit exceeded")
            next_link=data.get("@odata.nextLink");delta_link=data.get("@odata.deltaLink")
            if next_link and delta_link:raise ValueError("Ambiguous delta cursor")
            if next_link:
                if not isinstance(next_link,str):raise ValueError("Next link required")
                url=next_link;continue
            if not isinstance(delta_link,str) or not delta_link:raise ValueError("Terminal delta link required")
            self._validate_link(delta_link)
            return {"changes":changes,"delta_link":delta_link}
        raise TimeoutError("Delta page limit exceeded")
