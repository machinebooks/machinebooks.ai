from functools import wraps
from collections.abc import Mapping
from types import MappingProxyType, SimpleNamespace
import inspect
import time

class NativeClientUsageTracker:
    def __init__(self, client, model_name: str, service_type: str,
                 provider_type: str, audit_sink, context: dict):
        if not callable(audit_sink):
            raise TypeError("A durable audit sink is required")
        if not isinstance(context, Mapping):
            raise TypeError("Invalid context")
        self._client = client
        self._audit_sink = audit_sink
        fields = ("user_id", "tenant_id", "app_code", "request_id", "entity_type", "entity_id")
        if any(context.get(key) is not None and type(context.get(key)) not in (str, int) for key in fields):
            raise TypeError("Context must contain only scalar identifiers")
        self._context = MappingProxyType({key: context.get(key) for key in fields})
        original = client.messages.create
        if inspect.iscoroutinefunction(original) or inspect.iscoroutinefunction(audit_sink):
            raise TypeError("This adapter requires a synchronous client and sink")

        def deliver(record):
            if self._audit_sink(record) is not True:
                raise RuntimeError("The sink did not confirm persistence")

        @wraps(original)
        def tracked_create(*args, **kwargs):
            if kwargs.get("stream"):
                raise NotImplementedError("Streaming requires a separately instrumented adapter")
            start = time.perf_counter()
            base = {**dict(self._context), "model_name": kwargs.get("model", model_name),
                    "service_type": service_type, "provider_type": provider_type}
            try:
                response = original(*args, **kwargs)
            except BaseException as error:
                record = {**base, "success": False, "prompt_tokens": None,
                          "completion_tokens": None,
                          "latency_ms": int((time.perf_counter() - start) * 1000)}
                try:
                    deliver(record)
                except Exception:
                    error.add_note("Audit delivery also failed")
                raise
            usage = getattr(response, "usage", None)
            record = {**base, "success": True,
                      "observed_model_name": getattr(response, "model", None),
                      "prompt_tokens": getattr(usage, "input_tokens", None),
                      "completion_tokens": getattr(usage, "output_tokens", None),
                      "latency_ms": int((time.perf_counter() - start) * 1000)}
            deliver(record)
            return response

        self.messages = SimpleNamespace(create=tracked_create)

    def __getattr__(self, name):
        return getattr(self._client, name)
