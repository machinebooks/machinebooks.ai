from decimal import Decimal, InvalidOperation

class LLMModelPricing(Base):
    __tablename__ = "llm_model_pricing"
    id = Column(Integer, primary_key=True)
    provider = Column(String(32), nullable=False)
    model_id = Column(String(100), nullable=False)
    input_price_per_million = Column(Numeric(18, 8), nullable=False)
    output_price_per_million = Column(Numeric(18, 8), nullable=False)
    cached_input_price_per_million = Column(Numeric(18, 8))
    supports_cache_pricing = Column(Boolean, nullable=False, default=False)
    effective_from = Column(DateTime(timezone=True), nullable=False)
    effective_to = Column(DateTime(timezone=True))
    updated_by = Column(String(64))

    def calculate_cost(self, input_tokens, output_tokens, cached_tokens=0):
        # Adapter contract: input_tokens INCLUDES cached_tokens in this example.
        for value in (input_tokens, output_tokens, cached_tokens):
            if value is not None and (type(value) is not int or not 0 <= value <= 10**12):
                raise ValueError("Invalid token count")
        if input_tokens is None or output_tokens is None or cached_tokens is None:
            return None
        if cached_tokens > input_tokens:
            raise ValueError("Cached tokens exceed input tokens")
        try:
            regular = Decimal(str(self.input_price_per_million))
            output = Decimal(str(self.output_price_per_million))
            cache = (Decimal(str(self.cached_input_price_per_million))
                     if self.supports_cache_pricing
                     and self.cached_input_price_per_million is not None else regular)
        except (InvalidOperation, ValueError):
            raise ValueError("Invalid pricing snapshot") from None
        if any(not price.is_finite() or price < 0 for price in (regular, output, cache)):
            raise ValueError("Invalid pricing snapshot")
        return ((input_tokens - cached_tokens) * regular
                + cached_tokens * cache + output_tokens * output) / Decimal(1_000_000)
