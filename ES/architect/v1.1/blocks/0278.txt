    location /api/ {
        proxy_pass http://backend_api;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_read_timeout 120s;   # Endpoints de generación pueden tardar
    }

    # CRÍTICO: deshabilitar buffering para que el streaming llegue al cliente
    location /api/ai/stream {
        proxy_pass http://ai_service_api;
        proxy_buffering off;              # Sin buffering en SSE
        proxy_cache off;
        proxy_set_header Connection '';   # HTTP/1.1 para keep-alive
        proxy_http_version 1.1;
        proxy_read_timeout 300s;          # El streaming puede durar varios minutos
        chunked_transfer_encoding on;
    }
